\begin{lexlean}{GradeOracle}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{ReasoningOracle}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: Graded,
  parameters: [v : ReasoningOracle::Vitals, r : Nat],
  result: Prop,
  body: ((r <= 3) /\ ((r = 3) -> (ReasoningOracle::Hypotensive v)))
}

definition{
  name: gradedCheck,
  parameters: [v : ReasoningOracle::Vitals, r : Nat],
  result: Bool,
  body: ((r <=? 3) && ((not (r =? 3)) || (ReasoningOracle::hypotensive v))),
  executable: true
}

theorem{
  name: graded_sound,
  parameters: [v : ReasoningOracle::Vitals, r : Nat],
  statement: (((gradedCheck v r) = true) -> (Graded v r)),
  proof: linear_arithmetic{
    definitions: [Graded, ReasoningOracle::Hypotensive, ReasoningOracle::hypotensive, gradedCheck]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

verifier{
  name: Scale,
  axioms: [Classical.choice, Quot.sound, propext],
  candidate: {name: r, type: Nat},
  check: gradedCheck,
  sound: graded_sound,
  specification: Graded,
  subject: {name: v, type: ReasoningOracle::Vitals}
}

theorem{
  name: grade_admitted,
  parameters: [v : ReasoningOracle::Vitals],
  statement: (ReasoningOracle::Consistent (record ReasoningOracle::Chart {vitals := v, fever := 0, tachycardia := 0, tachypnea := 0, leukocytosis := 0, sirs := 0, sepsis := 0, shock := 0, level := 0})),
  proof: apply{arguments: [v], theorem: ReasoningOracle::admitted_consistent},
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: grade_pending,
  parameters: [v : ReasoningOracle::Vitals],
  statement: ((ReasoningOracle::Pending (record ReasoningOracle::Chart {vitals := v, fever := 0, tachycardia := 0, tachypnea := 0, leukocytosis := 0, sirs := 0, sepsis := 0, shock := 0, level := 0})) < 11),
  proof: apply{arguments: [v], theorem: ReasoningOracle::admitted_pending},
  axioms: [Quot.sound, propext]
}

definition{
  name: Related,
  parameters: [v : ReasoningOracle::Vitals, c : ReasoningOracle::Chart],
  result: Prop,
  body: ((ReasoningOracle::Consistent c) /\ ((ReasoningOracle::Warranted c) /\ ((((project c shock) = 1) -> (ReasoningOracle::Hypotensive (project c vitals))) /\ ((project (project c vitals) systolic) = (project v systolic)))))
}

theorem{
  name: related_admitted,
  parameters: [v : ReasoningOracle::Vitals],
  statement: (Related v (record ReasoningOracle::Chart {vitals := v, fever := 0, tachycardia := 0, tachypnea := 0, leukocytosis := 0, sirs := 0, sepsis := 0, shock := 0, level := 0})),
  proof: linear_arithmetic{
    definitions: [
      ReasoningOracle::Consistent,
      ReasoningOracle::Hypotensive,
      ReasoningOracle::Warranted,
      Related
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: related_preserved,
  parameters: [v : ReasoningOracle::Vitals, s : ReasoningOracle::Chart, t : ReasoningOracle::Chart],
  statement: ((Related v s) -> ((ReasoningOracle::Justified s t) -> (Related v t))),
  proof: linear_arithmetic{
    definitions: [
      ReasoningOracle::Consistent,
      ReasoningOracle::Febrile,
      ReasoningOracle::Hypotensive,
      ReasoningOracle::Justified,
      ReasoningOracle::Leukocytic,
      ReasoningOracle::Tachycardic,
      ReasoningOracle::Tachypneic,
      ReasoningOracle::Warranted,
      Related
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

theorem{
  name: grade_correct,
  parameters: [v : ReasoningOracle::Vitals, chart : ReasoningOracle::Chart, r : Nat],
  statement: ((Related v chart) -> (((ctor Option.some [Nat] (project chart level)) = (ctor Option.some [Nat] r)) -> (Graded v r))),
  proof: linear_arithmetic{
    definitions: [
      Graded,
      ReasoningOracle::Consistent,
      ReasoningOracle::Hypotensive,
      ReasoningOracle::Warranted,
      Related
    ]
  },
  axioms: [Classical.choice, Quot.sound, propext]
}

reasoner{
  name: Grade,
  axioms: [Classical.choice, Quot.sound, propext],
  answer: {name: chart, type: Nat, value: (ctor Option.some [Nat] (project chart level))},
  claims: [
    initial_invariant{theorem: grade_admitted},
    terminates{theorem: grade_pending},
    observation_invariant{
      initial: related_admitted,
      predicate: Related,
      preserved: related_preserved
    },
    answer_correct{theorem: grade_correct}
  ],
  executable: true,
  logic: {member: ReasoningOracle::Findings},
  observation: {name: v, type: ReasoningOracle::Vitals},
  observe: (record ReasoningOracle::Chart {vitals := v, fever := 0, tachycardia := 0, tachypnea := 0, leukocytosis := 0, sirs := 0, sepsis := 0, shock := 0, level := 0}),
  rules: [
    {member: ReasoningOracle::Fever},
    {member: ReasoningOracle::Tachycardia},
    {member: ReasoningOracle::Tachypnea},
    {member: ReasoningOracle::Leukocytosis},
    {member: ReasoningOracle::Sirs},
    {member: ReasoningOracle::Sepsis},
    {member: ReasoningOracle::Shock},
    {member: ReasoningOracle::Escalate}
  ],
  strategy: forward{fuel: 11},
  verifier: {member: Scale}
}
\end{semanticmodule}
\end{lexlean}
