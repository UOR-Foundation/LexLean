\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
inductive{
  name: Rose,
  type_parameters: [],
  parameters: [],
  constructors: [{name: node, fields: [Nat, (List Rose)]}]
}

definition{
  name: roseLabel,
  parameters: [rose : Rose],
  result: Nat,
  body: (match rose | Rose.node label children => (forestLabel children children)),
  mutual: Labels,
  recursive_argument: rose
}

definition{
  name: forestLabel,
  parameters: [forest : (List Rose), spare : (List Rose)],
  result: Nat,
  body: (match forest | List.nil => 0 | List.cons head tail => (forestLabel forest tail)),
  mutual: Labels,
  recursive_argument: forest
}
\end{semanticmodule}
\end{lexlean}
