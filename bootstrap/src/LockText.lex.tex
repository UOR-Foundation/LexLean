definition{
  name: escapeByte,
  parameters: [value : Nat],
  result: Bytes,
  body: (if (value =? 34) then 0x5c22 else (if (value =? 92) then 0x5c5c else (if (if (value <? 32) then true else (value =? 127)) then (cat (cat 0x5c753030 (upperHex (prim quotient value 16 0 : Nat))) (upperHex (prim remainder value 16 0 : Nat))) else (LexLeanStd.Numeral::pick (LexLeanStd.Numeral::byteValues) value)))),
  executable: true
}

definition{
  name: escapeStep,
  parameters: [subject : Bytes, state : (Nat * Bytes)],
  result: (Option (Nat * Bytes)),
  body: (match (LexLeanStd.Text::byteAt subject (first state))
    | Option.some byte => (ctor Option.some [(Nat * Bytes)] (((first state) + 1), (cat (second state) (escapeByte (prim to_nat byte : Nat)))))
    | Option.none => (ctor Option.none [(Nat * Bytes)])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{

  axioms: [Quot.sound, propext],
  executable: true,
