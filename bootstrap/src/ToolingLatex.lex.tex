definition{
  name: putTy,
  parameters: [out : LexLeanStd.Buffer::Buffer, reserved : (List Bytes), ty : ToolingAst::Ty, mode : Nat, lead : Bool],
  result: LexLeanStd.Buffer::Buffer,
  body: (match ty
    | ToolingAst::Ty.TyConst head args => (if (tyEmpty args) then (putName out reserved head) else (putS (putTy (putS (putName (putS out "(") reserved head) " ") reserved args 1 true) ")"))
    | ToolingAst::Ty.TyParam head args => (if (tyEmpty args) then (putName out reserved head) else (putS (putTy (putS (putName (putS out "(") reserved head) " ") reserved args 1 true) ")"))
    | ToolingAst::Ty.TyFn params result => (putS (putTy (putArrow (putTy (putS out "(") reserved params 2 true)) reserved result 0 true) ")")
    | ToolingAst::Ty.TyIndex owner => (putName out reserved owner)
    | ToolingAst::Ty.TyNil => out
    | ToolingAst::Ty.TyCons head tail => (putTy (putTy (if lead then out else (if (mode =? 2) then (putArrow out) else (putS out " "))) reserved head 0 true) reserved tail mode false)),
  executable: true,
  recursive_argument: ty
}

definition{
  name: putSig,
  parameters: [out : LexLeanStd.Buffer::Buffer, ctx : Ctx, sig : ToolingAst::Sig, lead : Bool],
  result: LexLeanStd.Buffer::Buffer,
  body: (match sig
    | ToolingAst::Sig.SgConst title => (putName out (project ctx reserved) title)
    | ToolingAst::Sig.SgVar title => (putName out (project ctx reserved) title)
    | ToolingAst::Sig.SgApp callee args => (let o1 : LexLeanStd.Buffer::Buffer := (if (explicitApp ctx callee (sigLength args)) then (putS out "(@") else (putS out "(")); (putS (putSig (putS (putSig o1 ctx callee true) " ") ctx args true) ")"))
    | ToolingAst::Sig.SgArrow domain codomain => (putS (putSig (putArrow (putSig (putS out "(") ctx domain true)) ctx codomain true) ")")
    | ToolingAst::Sig.SgPi label info domain codomain => (let o1 : LexLeanStd.Buffer::Buffer := (putS out "("); (putS (putSig (putArrow (match label
      | Option.some title => (match info
        | ToolingAst::BinderKind.BkExplicit => (putS (putSig (putS (putName (putS o1 "(") (project ctx reserved) title) " : ") ctx domain true) ")")
        | ToolingAst::BinderKind.BkImplicit => (putS (putSig (putS (putName (putS o1 "{") (project ctx reserved) title) " : ") ctx domain true) "}")
        | ToolingAst::BinderKind.BkInstance => (putS (putSig (putS (putName (putS o1 "[") (project ctx reserved) title) " : ") ctx domain true) "]"))
      | Option.none => (putS (putSig (putS o1 "[") ctx domain true) "]"))) ctx codomain true) ")"))
    | ToolingAst::Sig.SgType level => (match level
      | Option.some title => (putS (putB (putS out "(Type ") title) ")")
      | Option.none => (putS out "Type"))
    | ToolingAst::Sig.SgOpt domain fallback => (putS (putSig (putS (putSig (putS out "(optParam ") ctx domain true) " ") ctx fallback true) ")")
    | ToolingAst::Sig.SgEmpty => (putS out "{ }")
    | ToolingAst::Sig.SgNat digits => (putB out digits)
    | ToolingAst::Sig.SgNil => out
    | ToolingAst::Sig.SgCons head tail => (putSig (putSig (if lead then out else (putS out " ")) ctx head true) ctx tail false)),
  executable: true,
  recursive_argument: sig
}

definition{
  name: putPars,

  result: LexLeanStd.Buffer::Buffer,
  body: (match pars
    | ToolingAst::Pars.PrNil => out
    | ToolingAst::Pars.PrCons title ty rest => (putPars (putS (putTy (putS (putName (putS (if lead then (if pre then (putS out " ") else out) else (putS out " ")) "(") reserved title) " : ") reserved ty 0 true) ")") reserved rest pre false)),
  executable: true,
  recursive_argument: pars
}

definition{
  name: putBinders,

definition{
  name: putTerm,
  parameters: [out : LexLeanStd.Buffer::Buffer, ctx : Ctx, term : ToolingAst::Term, mode : Nat, lead : Bool],
  result: LexLeanStd.Buffer::Buffer,
  body: (match term
    | ToolingAst::Term.TmVar title => (putName out (project ctx reserved) title)
    | ToolingAst::Term.TmNat digits => (putB out digits)
    | ToolingAst::Term.TmStr payload => (putLit out payload)
    | ToolingAst::Term.TmChar code => (putChar out code)
    | ToolingAst::Term.TmName title => (putName (putS out "``") (project ctx reserved) title)
    | ToolingAst::Term.TmCall callee targs args => (putS (putTerm (putNamed (putName (putS out "(") (project ctx reserved) callee) (project ctx reserved) (typingOf (project ctx typings) callee) targs) ctx args 1 true) ")")
    | ToolingAst::Term.TmConstruct callee targs args => (putS (putTerm (putNamed (putName (putS out "(") (project ctx reserved) callee) (project ctx reserved) (paramNames (project ctx tys) (ownerName callee)) targs) ctx args 1 true) ")")
    | ToolingAst::Term.TmField subject label => (putName (putS (putTerm (putS out "(") ctx subject 0 true) ").") (project ctx reserved) label)
    | ToolingAst::Term.TmList element elements => (putS (putTy (putS (putTerm (putS out "(([") ctx elements 2 true) "] : List ") (project ctx reserved) element 0 true) "))")
    | ToolingAst::Term.TmArray element elements => (putS (putTy (putS (putTerm (putS out "((#[") ctx elements 2 true) "] : Array ") (project ctx reserved) element 0 true) "))")
    | ToolingAst::Term.TmLet label ty bound body => (putS (putTerm (putS (putTerm (putS (putTy (putS (putName (putS out "(let ") (project ctx reserved) label) " : ") (project ctx reserved) ty 0 true) " := ") ctx bound 0 true) "; ") ctx body 0 true) ")")
    | ToolingAst::Term.TmLambda params body => (putS (putTerm (putS (putPars (putS out "(fun ") (project ctx reserved) params false true) " => ") ctx body 0 true) ")")
    | ToolingAst::Term.TmApply callee args => (putS (putTerm (putTerm (putS out "(") ctx callee 0 true) ctx args 1 true) ")")
    | ToolingAst::Term.TmIf condition thenTerm elseTerm => (putS (putTerm (putS (putTerm (putS (putTerm (putS out "(if ") ctx condition 0 true) " then ") ctx thenTerm 0 true) " else ") ctx elseTerm 0 true) ")")
    | ToolingAst::Term.TmMatch scrutinee branches => (putS (putTerm (putS (putTerm (putS out "(match ") ctx scrutinee 0 true) " with") ctx branches 0 true) ")")
    | ToolingAst::Term.TmNil => out
    | ToolingAst::Term.TmCons head tail => (putTerm (putTerm (if (mode =? 1) then (putS out " ") else (if lead then out else (putS out ", "))) ctx head 0 true) ctx tail mode false)
    | ToolingAst::Term.TmBranchNil => out
    | ToolingAst::Term.TmBranch label binders body tail => (putTerm (putTerm (putS (match label
      | Option.some full => (putBinders (putName (putS (putS out " | ") ".") (project ctx reserved) (shortName full)) (project ctx reserved) binders)
      | Option.none => (putS out " | _")) " => ") ctx body 0 true) ctx tail 0 true)),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: term
}

definition{
  name: putSyntax,
  parameters: [out : LexLeanStd.Buffer::Buffer, atoms : (List ToolingAst::Atom)],
  result: LexLeanStd.Buffer::Buffer,
  body: (prim list_fold (fun (written : LexLeanStd.Buffer::Buffer) (atom : ToolingAst::Atom) => (putS (match atom
    | ToolingAst::Atom.AtToken text => (putLit written text)
    | ToolingAst::Atom.AtIdent => (putS written "ident")
    | ToolingAst::Atom.AtType => (putS written "term")) " ")) out atoms : LexLeanStd.Buffer::Buffer),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: nextArgument,
  parameters: [ctx : Ctx, state : ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer)],
  result: ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer),
  body: (match (first state)
    | List.nil => state
    | List.cons argument rest => (rest, (match argument
      | ToolingAst::InvokeArg.IaIdent title => (putName (second state) (project ctx reserved) title)
      | ToolingAst::InvokeArg.IaType sig => (putSig (second state) ctx sig true)))),
  executable: true
}

definition{
  name: invokeStep,
  parameters: [ctx : Ctx, state : ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer), atom : ToolingAst::Atom],
  result: ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer),
  body: (match atom
    | ToolingAst::Atom.AtToken text => ((first state), (putB (second state) text))
    | ToolingAst::Atom.AtIdent => (nextArgument ctx state)
    | ToolingAst::Atom.AtType => (nextArgument ctx state)),
  executable: true
}

definition{
  name: leanDecl,
  parameters: [out : LexLeanStd.Buffer::Buffer, ctx : Ctx, declaration : ToolingAst::Decl],
  result: LexLeanStd.Buffer::Buffer,
  body: (let reserved : (List Bytes) := (project ctx reserved); (match declaration
    | ToolingAst::Decl.DType d0 d1 d2 d3 d4 d5 => out
    | ToolingAst::Decl.DCall d0 d1 d2 d3 => out
    | ToolingAst::Decl.DPlumbing d0 => out
    | ToolingAst::Decl.DFunction title isPublic isPartial label typeNames params resultTy bodyTerm => (let o1 : LexLeanStd.Buffer::Buffer := (if isPublic then (putS out "public ") else out); (let o2 : LexLeanStd.Buffer::Buffer := (putS o1 (if isPartial then "meta partial def " else "meta def ")); (let o3 : LexLeanStd.Buffer::Buffer := (putTypeParams (putName o2 reserved title) reserved typeNames); (let o4 : LexLeanStd.Buffer::Buffer := (putS (putTy (putS (putPars o3 reserved params true true) " : ") reserved resultTy 0 true) " :=\n  "); (putS (putTerm o4 ctx bodyTerm 0 true) "\n")))))
    | ToolingAst::Decl.DCommand title atoms elaborator => (putS (putName (putS (putName (putS (putSyntax (putS (putName (putS out "syntax (name := ") reserved title) ") ") atoms) ": command\nattribute [command_elab ") reserved title) "] ") reserved elaborator) "\n")
    | ToolingAst::Decl.DInvoke title args => (putS (second (prim list_fold (fun [ctx] (state : ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer)) (atom : ToolingAst::Atom) => (invokeStep ctx state atom)) (args, out) (commandAtoms (project ctx decls) title) : ((List ToolingAst::InvokeArg) * LexLeanStd.Buffer::Buffer))) "\n")
    | ToolingAst::Decl.DEvaluate title args => (putS (putTerm (putName (putS out "#eval ") reserved title) ctx args 1 true) "\n"))),
  axioms: [Quot.sound, propext],
  executable: true
}
