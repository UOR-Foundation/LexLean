  name: treeSum,
  parameters: [tree : Tree],
  result: Nat,
  body: (match tree | Tree.leaf => 0 | Tree.node l v r => (((treeSum l) + v) + (treeSum r))),
  executable: true,
  recursive_argument: tree
}

definition{
  name: treeDepth,
  parameters: [tree : Tree],
  result: Nat,
  body: (match tree
    | Tree.leaf => 0
    | Tree.node l v r => (let dl : Nat := (treeDepth l); (let dr : Nat := (treeDepth r); (if (dl <? dr) then (dr + 1) else (dl + 1))))),
  executable: true,
  recursive_argument: tree
}

definition{
  name: treeStats,
  parameters: [tree : Tree],
  result: (Nat * Nat),

  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}

theorem{
  name: bisect_decreases,
  parameters: [target : Nat, low : Nat, high : Nat],
  statement: (((low <? high) = true) -> (((target <=? (low + low)) = false) -> ((prim subtract high (low + 1) : Nat) < (prim subtract high low : Nat)))),

}

definition{
  name: bisect,

  production: {effects: [overflow], targets: [rust-core, rust-std]},
