\begin{lexlean}{Boundary}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
inductive{
  name: Grove,
  type_parameters: [],
  parameters: [],
  constructors: [
    {
      name: node,
      fields: [(Set Nat), (List Grove), (Map String Int), (Option Grove), (List (Nat * Grove))]
    },
    {name: leaf, fields: [(Map Nat Int)]},
    {name: settled, fields: [(Result Grove Nat)]}
  ]
}

structure{
  name: Ledger,
  type_parameters: [],
  parameters: [],
  fields: [{name: entries, type: (Map String Int)}, {name: owner, type: String}]
}

definition{
  name: groveTags,
  parameters: [g : Grove],
  result: Nat,
  body: (match g
    | Grove.node tags children labels spare paired => (prim set_size tags : Nat)
    | Grove.leaf weights => (prim map_size weights : Nat)
    | Grove.settled outcome => 0),
  executable: true
}

definition{
  name: groveRoot,
  parameters: [
    g : Grove,
    ledger : Ledger,
    picks : (Option (Set Nat)),
    spare : (List (Map Nat Bool))
  ],
  result: (Nat * (Ledger * ((Option (Set Nat)) * (List (Map Nat Bool))))),
  body: ((groveTags g), (ledger, (picks, spare))),
  executable: true,
  production: {effects: [allocation], targets: [rust-std]}
}
\end{semanticmodule}
\end{lexlean}
