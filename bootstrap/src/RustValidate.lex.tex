definition{
  name: tyDebug,
  parameters: [ty : Ty],
  result: Bytes,
  body: (match ty
    | Ty.TyLeaf k => (leafName k)
    | Ty.TyFixed k => (cat (text "Fixed(") (cat (kindUpper k) (text ")")))
    | Ty.TyUnary k inner => (cat (unaryName k) (cat (text "(") (cat (tyDebug inner) (text ")"))))
    | Ty.TyBinary k left right => (cat (binaryName k) (cat (text "(") (cat (tyDebug left) (cat (text ", ") (cat (tyDebug right) (text ")"))))))
    | Ty.TyNamed k n => (cat (nameName k) (cat (text "(") (cat (decimal n) (text ")"))))
    | Ty.TyNil => (text "unreachable")
    | Ty.TyCons d0 d1 => (text "unreachable")),
  executable: true,
  recursive_argument: ty
}

definition{
  name: tyHeap,
  parameters: [ty : Ty],
  result: Bool,
  body: (match ty
    | Ty.TyLeaf k => (leafHeap k)
    | Ty.TyFixed d0 => false
    | Ty.TyUnary k inner => (match (unaryHeap k)
      | Bool.true => true
      | Bool.false => (tyHeap inner))
    | Ty.TyBinary k left right => (let l : Bool := (tyHeap left); (let r : Bool := (tyHeap right); (l || r)))
    | Ty.TyNamed d0 d1 => false
    | Ty.TyNil => false
    | Ty.TyCons d0 d1 => false),
  executable: true,
  recursive_argument: ty
}

definition{

  parameters: [decl : Decl, defs : Def],

    | Def.DApply d0 d1 d2 d3 d4 d5 => (ctor Result.ok [Decl, Bytes] decl)

      | Id.IdExport d0 => (declareDefs decl tail))
