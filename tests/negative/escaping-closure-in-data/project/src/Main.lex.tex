\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: keep,
  parameters: [step : (Fn (Nat) Nat), value : Nat],
  result: Nat,
  body: (match (ctor Option.some [(Fn (Nat) Nat)] step)
    | Option.none => value
    | Option.some held => (apply held value)),
  executable: true
}
\end{semanticmodule}
\end{lexlean}
