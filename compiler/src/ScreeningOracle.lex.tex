\begin{lexlean}{ScreeningOracle}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: Short,
  parameters: [w : Nat, doses : (List Nat)],
  result: Prop,
  body: ((prim length doses : Nat) <= 4)
}

definition{
  name: shortCheck,
  parameters: [w : Nat, doses : (List Nat)],
  result: Bool,
  body: ((prim length doses : Nat) <=? 4),
  executable: true
}

theorem{
  name: short_sound,
  parameters: [w : Nat, doses : (List Nat)],
  statement: (((shortCheck w doses) = true) -> (Short w doses)),
  proof: linear_arithmetic{definitions: [Short, shortCheck]},
  axioms: [propext]
}

contract{
  name: DoseContract,
  axioms: [propext],
  input: {name: w, type: Nat},
  output: {name: doses, type: (List Nat)},
  postcondition: Short,
  validators: [{predicate: postcondition, sound: short_sound, validator: shortCheck}]
}

realization{
  name: DoseCandidates,
  descriptor: deterministic{
    body: (cons (prim subtract w 10 : Nat) (cons (prim subtract w 20 : Nat) (cons (prim subtract w 30 : Nat) (cons (prim subtract w 50 : Nat) (nil Nat)))))
  },
  executable: true,
  input: {name: w, type: Nat},
  output: (List Nat)
}

model{name: DoseModel, contract: {member: DoseContract}, realization: {member: DoseCandidates}}

definition{
  name: Proposed,
  parameters: [s : (Nat * Nat), t : (Nat * Nat)],
  result: Prop,
  body: (((second s) = 0) /\ (((first t) = (first s)) /\ (0 < (second t))))
}

logic{name: Prescribing, relation: Proposed, state: {name: s, type: (Nat * Nat), next: t}}

theorem{
  name: propose_sound,
  parameters: [s : (Nat * Nat), d : Nat],
  statement: (((((second s) =? 0) && (0 <? d)) = true) -> (Proposed s ((first s), d))),
  proof: linear_arithmetic{definitions: [Proposed]},
  axioms: [propext]
}

inference_rule{
  name: Propose,
  axioms: [propext],
  binding: {
    name: d,
    type: Nat,
    candidates: (match (checked DoseModel (first s) with [postcondition])
      | Result.ok proposed => proposed
      | Result.error refusal => (nil Nat))
  },
  conclusion: ((first s), d),
  executable: true,
  guard: (((second s) =? 0) && (0 <? d)),
  logic: {member: Prescribing},
  soundness: propose_sound
}

definition{name: Safe, parameters: [w : Nat, r : Nat], result: Prop, body: ((r + r) <= w)}

definition{
  name: safeCheck,
  parameters: [w : Nat, r : Nat],
  result: Bool,
  body: ((r + r) <=? w),
  executable: true
}

theorem{
  name: safe_sound,
  parameters: [w : Nat, r : Nat],
  statement: (((safeCheck w r) = true) -> (Safe w r)),
  proof: linear_arithmetic{definitions: [Safe, safeCheck]},
  axioms: [propext]
}

verifier{
  name: SafeDose,
  axioms: [propext],
  candidate: {name: r, type: Nat},
  check: safeCheck,
  sound: safe_sound,
  specification: Safe,
  subject: {name: w, type: Nat}
}

reasoner{
  name: Dose,
  axioms: [propext],
  executable: true,
  observation: {name: w, type: Nat},
  rules: [],
  strategy: generate_and_verify{
    budget: 3,
    generator: (match (checked DoseModel w with [postcondition])
      | Result.ok proposed => proposed
      | Result.error refusal => (nil Nat))
  },
  verifier: {member: SafeDose}
}

reasoner{
  name: Screen,
  axioms: [propext],
  answer: {
    name: pick,
    type: Nat,
    value: (if (0 <? (second pick)) then (ctor Option.some [Nat] (second pick)) else (ctor Option.none [Nat]))
  },
  executable: true,
  logic: {member: Prescribing},
  observation: {name: w, type: Nat},
  observe: (w, 0),
  rules: [{member: Propose}],
  strategy: search{frontier: 3, fuel: 8, order: depth_first},
  verifier: {member: SafeDose}
}
\end{semanticmodule}
\end{lexlean}
