\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{name: Positive, parameters: [x : Nat], result: Prop, body: (1 <= x)}

definition{
  name: positiveCheck,
  parameters: [x : Nat],
  result: Bool,
  body: (1 <=? x),
  executable: true
}

theorem{
  name: positive_check_sound,
  parameters: [x : Nat],
  statement: (((positiveCheck x) = true) -> (Positive x)),
  proof: linear_arithmetic{definitions: [Positive, positiveCheck]}
}

definition{name: Doubles, parameters: [y : Nat], result: Prop, body: (0 <= y)}

contract{
  name: DoubleContract,
  input: {name: x, type: Nat},
  output: {name: y, type: Nat},
  postcondition: Doubles,
  precondition: Positive,
  validators: [{predicate: precondition, sound: positive_check_sound, validator: positiveCheck}]
}

realization{
  name: Doubler,
  descriptor: deterministic{body: (x + x)},
  executable: true,
  input: {name: x, type: Nat},
  output: Nat
}
\end{semanticmodule}
\end{lexlean}
