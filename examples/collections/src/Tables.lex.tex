\begin{lexlean}{Tables}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: declare,
  parameters: [table : (Map String Nat), name : String, slot : Nat],
  result: (Map String Nat),
  body: (prim map_insert table name slot : (Map String Nat)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: buildTable,
  parameters: [names : (List String)],
  result: (Map String Nat),
  body: (prim list_fold (fun (table : (Map String Nat)) (name : String) => (declare table name (prim map_size table : Nat))) map_literal{
    entries: [],
    key: String,
    value: Nat
  } names : (Map String Nat)),
  axioms: [Classical.choice, Quot.sound, propext],
  executable: true
}

definition{
  name: callGraph,
  parameters: [],
  result: (Map String (Set String)),
  body: graph_literal{
    edges: [
      {source: "main", target: "parse"},
      {source: "main", target: "emit"},
      {source: "parse", target: "lex"},
      {source: "emit", target: "report"},
      {source: "report", target: "lex"}
    ],
    node: String,
    nodes: ["main", "parse", "lex", "emit", "report"]
  }
}

definition{
  name: callOrder,
  parameters: [],
  result: (Option (List String)),
  body: (prim graph_topological (callGraph) : (Option (List String))),
  axioms: [Classical.choice, Quot.sound, propext]
}

definition{
  name: reachableFromParse,
  parameters: [],
  result: (Set String),
  body: (prim graph_reachable (callGraph) "parse" : (Set String)),
  axioms: [Classical.choice, Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
