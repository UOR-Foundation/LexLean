  name: halve_decreases,
  parameters: [number : Nat],
  statement: (((number <? 2) = false) -> ((prim subtract number 2 : Nat) < number)),
  proof: linear_arithmetic{}
}

  name: halve,
  parameters: [number : Nat],
  result: Nat,
  body: (if (number <? 2) then 0 else ((halve (prim subtract number 2 : Nat)) + (apply (ref halve) number))),
  termination: {evidence: [halve_decreases], measure: number}
}
