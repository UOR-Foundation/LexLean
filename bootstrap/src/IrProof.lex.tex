definition{
  name: proofJson,
  parameters: [proof : Proof, state : IrTerm::Renumber],
  result: (LexLeanStd.Json::Json * IrTerm::Renumber),
  body: (match proof
    | Proof.PSeq steps => (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson steps state); ((tagged "seq" (IrTerm::member "s" (jarr (first inner)) (jnil))), (second inner)))
    | Proof.PIntro locals => (let bound : (LexLeanStd.Json::Json * IrTerm::Renumber) := (bindAll locals state); ((tagged "intro" (IrTerm::member "l" (jarr (first bound)) (jnil))), (second bound)))
    | Proof.PExact term => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (termWith "t" term state (jnil)); ((tagged "exact" (first written)), (second written)))
    | Proof.PApplyOne term => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (termWith "t" term state (jnil)); ((tagged "apply-one" (first written)), (second written)))
    | Proof.PApply function premises => (let head : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson function state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson premises (second head)); ((tagged "apply" (IrTerm::member "f" (first head) (IrTerm::member "p" (jarr (first inner)) (jnil)))), (second inner))))
    | Proof.PRfl => ((tagged "rfl" (jnil)), state)
    | Proof.PWitness term => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (termWith "t" term state (jnil)); ((tagged "witness" (first written)), (second written)))
    | Proof.PLeft => ((tagged "left" (jnil)), state)
    | Proof.PRight => ((tagged "right" (jnil)), state)
    | Proof.PHave localId proposition nested => (let stated : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson proposition state); (let index : (Nat * IrTerm::Renumber) := (IrTerm::bind (second stated) localId); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson nested (second index)); ((tagged "have" (IrTerm::member "l" (IrTerm::jnat (first index)) (IrTerm::member "p" (first stated) (IrTerm::member "q" (first inner) (jnil))))), (second inner)))))
    | Proof.PRewrite target rules => (let aimed : (LexLeanStd.Json::Json * IrTerm::Renumber) := (targetJson target state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson rules (second aimed)); ((tagged "rw" (IrTerm::member "at" (first aimed) (IrTerm::member "r" (jarr (first inner)) (jnil)))), (second inner))))
    | Proof.PSimp target rules => (let aimed : (LexLeanStd.Json::Json * IrTerm::Renumber) := (targetJson target state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson rules (second aimed)); ((tagged "simp-only" (IrTerm::member "at" (first aimed) (IrTerm::member "r" (jarr (first inner)) (jnil)))), (second inner))))
    | Proof.PConstructor branches => (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson branches state); ((tagged "constructor" (IrTerm::member "b" (jarr (first inner)) (jnil))), (second inner)))
    | Proof.PCases scrutinee branchesOf => (let head : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson scrutinee state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson branchesOf (second head)); ((tagged "cases" (IrTerm::member "t" (first head) (IrTerm::member "c" (jarr (first inner)) (jnil)))), (second inner))))
    | Proof.PInduction scrutinee branchesOf => (let head : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson scrutinee state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson branchesOf (second head)); ((tagged "induction" (IrTerm::member "t" (first head) (IrTerm::member "c" (jarr (first inner)) (jnil)))), (second inner))))
    | Proof.PCalc relation start steps => (let begun : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson start state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson steps (second begun)); ((tagged "calc" (IrTerm::member "rel" (IrTerm::globalTermJson relation (ctor LexiconLse::Universe.UNil)) (IrTerm::member "s" (first begun) (IrTerm::member "steps" (jarr (first inner)) (jnil))))), (second inner))))
    | Proof.PNil => ((jnil), state)
    | Proof.PItem head tail => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson head state); (let rest : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson tail (second written)); ((ctor LexLeanStd.Json::Json.JItem (first written) (first rest)), (second rest))))
    | Proof.PRule backward term tail => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson term state); (let rest : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson tail (second written)); ((ctor LexLeanStd.Json::Json.JItem (jobj (IrTerm::member "rev" (ctor LexLeanStd.Json::Json.JBool backward) (IrTerm::member "t" (first written) (jnil)))) (first rest)), (second rest))))
    | Proof.PCase ctorName leanName binders branch tail => (let bound : (LexLeanStd.Json::Json * IrTerm::Renumber) := (bindAll (caseLocals binders) state); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson branch (second bound)); (let rest : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson tail (second inner)); ((ctor LexLeanStd.Json::Json.JItem (jobj (IrTerm::member "ctor" (ctor LexLeanStd.Json::Json.JStr ctorName) (IrTerm::member "lean" (ctor LexLeanStd.Json::Json.JStr leanName) (IrTerm::member "b" (jarr (first bound)) (IrTerm::member "p" (first inner) (jnil)))))) (first rest)), (second rest)))))
    | Proof.PStep term justification tail => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson term state); (let justified : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson justification (second written)); (let rest : (LexLeanStd.Json::Json * IrTerm::Renumber) := (proofJson tail (second justified)); ((ctor LexLeanStd.Json::Json.JItem (jobj (IrTerm::member "t" (first written) (IrTerm::member "p" (first justified) (jnil)))) (first rest)), (second rest)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: proof
}

definition{
  name: listed,
  parameters: [names : (List Bytes), named : Bytes],
  result: Bool,
  body: (prim list_fold (fun [named] (found : Bool) (candidate : Bytes) => (if found then true else (prim equal candidate named : Bool))) false names : Bool),
  executable: true

definition{
  name: sortedUnique,
  parameters: [names : (List Bytes)],
  result: (List Bytes),
  body: (LexLeanStd.Order::reverseList [Bytes] (prim list_fold (fun (kept : (List Bytes)) (entry : (Bytes * Bytes)) => (match kept
    | List.nil => (cons (first entry) kept)
    | List.cons head rest => (if (prim equal head (first entry) : Bool) then kept else (cons (first entry) kept)))) (nil Bytes) (LexLeanStd.Order::sortEntries (prim list_fold (fun (acc : (List (Bytes * Bytes))) (named : Bytes) => (cons (named, named) acc)) (nil (Bytes * Bytes)) (LexLeanStd.Order::reverseList [Bytes] names) : (List (Bytes * Bytes)))) : (List Bytes))),
  axioms: [propext],
  executable: true
}

definition{
  name: walkStep,
  parameters: [walk : Walk, named : Bytes],
  result: Walk,
  body: (match (project walk remaining)
    | List.nil => (record Walk {remaining := (nil Bytes), same := false})
    | List.cons head rest => (record Walk {remaining := rest, same := (if (project walk same) then (prim equal head named : Bool) else false)})),
  executable: true
}

definition{
  name: sameNames,
  parameters: [left : (List Bytes), right : (List Bytes)],
  result: Bool,
  body: (let walked : Walk := (prim list_fold (fun (walk : Walk) (named : Bytes) => (walkStep walk named)) (record Walk {remaining := right, same := true}) left : Walk); (if (project walked same) then (match (project walked remaining)
    | List.nil => true
    | List.cons head rest => false) else false)),
  executable: true
}

definition{
  name: permits,
  parameters: [policy : AxiomPolicy, observed : (List Bytes)],
  result: Bool,
  body: (match policy
    | AxiomPolicy.ANone => (match observed | List.nil => true | List.cons head rest => false)
    | AxiomPolicy.AAllow allowed => (prim list_fold (fun [allowed] (allSoFar : Bool) (named : Bytes) => (if allSoFar then (listed allowed named) else false)) true observed : Bool)
    | AxiomPolicy.AExact exactNames => (sameNames (sortedUnique observed) exactNames)),
  axioms: [propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}
