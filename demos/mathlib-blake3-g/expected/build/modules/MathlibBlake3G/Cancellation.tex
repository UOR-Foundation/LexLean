\section{Cancellation}
\label{ll:cancellation:addition}
\begin{theorem}
\label{ll:cancellation:add-sub-cancel}
For every \(\operatorname{bitVec}(32)\) \(a\) and \(\operatorname{bitVec}(32)\) \(b\), \(\operatorname{hSubHSub}(\operatorname{hAddHAdd}(a, b), b) = a\).
\end{theorem}
\begin{proof}
The goal follows from \(\operatorname{lexLeanSpecBitVec32BitVecAddSubCancel}(a, b)\).
\end{proof}
\begin{theorem}
\label{ll:cancellation:add-sub-cancel-group}
For every \(\operatorname{bitVec}(32)\) \(a\) and \(\operatorname{bitVec}(32)\) \(b\), \(\operatorname{hSubHSub}(\operatorname{hAddHAdd}(a, b), b) = a\).
\end{theorem}
\begin{proof}
The goal follows from \(\operatorname{lexLeanSpecBitVec32AddSubCancelRight}(a, b)\).
\end{proof}
\begin{theorem}
\label{ll:cancellation:add-sub-cancel-u32}
For every \(\operatorname{uInt32}\) \(a\) and \(\operatorname{uInt32}\) \(b\), \(\operatorname{hSubHSub}(\operatorname{hAddHAdd}(a, b), b) = a\).
\end{theorem}
\begin{proof}
The goal follows from \(\operatorname{lexLeanSpecU32UInt32AddSubCancel}(a, b)\).
\end{proof}
\begin{lemma}
\label{ll:cancellation:add-sub-sub-cancel}
For every \(\operatorname{bitVec}(32)\) \(p\) and \(\operatorname{bitVec}(32)\) \(q\) and \(\operatorname{bitVec}(32)\) \(r\), \(\operatorname{hSubHSub}(\operatorname{hSubHSub}(\operatorname{hAddHAdd}(\operatorname{hAddHAdd}(p, q), r), q), r) = p\).
\end{lemma}
\begin{proof}
Rewrite the goal using \(\rightarrow \operatorname{lexLeanSpecBitVec32AddRightComm}(p, q, r)\), \(\rightarrow \operatorname{lexLeanSpecBitVec32AddSubCancelRight}(\operatorname{hAddHAdd}(p, r), q)\), \(\rightarrow \operatorname{lexLeanSpecBitVec32AddSubCancelRight}(p, r)\).
\end{proof}
