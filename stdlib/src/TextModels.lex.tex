\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: withIndex,
  parameters: [vocabulary : (List String), start : Nat],
  result: (List (Nat * String)),
  body: (match vocabulary
    | List.nil => (nil (Nat * String))
    | List.cons head tail => (cons (start, head) (withIndex tail (start + 1)))),
  executable: true,
  recursive_argument: vocabulary
}

definition{
  name: takeInts,
  parameters: [count : Nat, values : (List Int)],
  result: (List Int),
  body: (match count
    | Nat.zero => (nil Int)
    | Nat.succ previous => (match values
      | List.nil => (nil Int)
      | List.cons head tail => (cons head (takeInts previous tail)))),
  executable: true,
  recursive_argument: count
}

definition{
  name: zerosInts,
  parameters: [count : Nat],
  result: (List Int),
  body: (match count
    | Nat.zero => (nil Int)
    | Nat.succ previous => (cons (0 : Int) (zerosInts previous))),
  executable: true,
  recursive_argument: count
}

definition{
  name: fitWidth,
  parameters: [width : Nat, values : (List Int)],
  result: (List Int),
  body: (prim append (takeInts width values) (zerosInts (prim subtract width (prim length values : Nat) : Nat)) : (List Int)),
  executable: true
}

definition{
  name: fits,
  parameters: [capacity : Nat, context : (List UInt32)],
  result: Bool,
  body: ((prim length context : Nat) <=? capacity),
  executable: true
}

definition{
  name: dropTokens,
  parameters: [count : Nat, context : (List UInt32)],
  result: (List UInt32),
  body: (match count
    | Nat.zero => context
    | Nat.succ previous => (match context
      | List.nil => (nil UInt32)
      | List.cons head tail => (dropTokens previous tail))),
  executable: true,
  recursive_argument: count
}

definition{
  name: keepLast,
  parameters: [capacity : Nat, context : (List UInt32)],
  result: (List UInt32),
  body: (dropTokens (prim subtract (prim length context : Nat) capacity : Nat) context),
  executable: true
}

definition{
  name: admitRefuse,
  parameters: [capacity : Nat, context : (List UInt32)],
  result: (Option (List UInt32)),
  body: (if (fits capacity context) then (ctor Option.some [(List UInt32)] context) else (ctor Option.none [(List UInt32)])),
  executable: true
}

definition{
  name: admitKeepLast,
  parameters: [capacity : Nat, context : (List UInt32)],
  result: (Option (List UInt32)),
  body: (ctor Option.some [(List UInt32)] (keepLast capacity context)),
  executable: true
}

definition{
  name: pushRefuse,
  parameters: [capacity : Nat, context : (List UInt32), token : UInt32],
  result: (Option (List UInt32)),
  body: (admitRefuse capacity (prim append context (cons token (nil UInt32)) : (List UInt32))),
  executable: true
}

definition{
  name: pushKeepLast,
  parameters: [capacity : Nat, context : (List UInt32), token : UInt32],
  result: (Option (List UInt32)),
  body: (admitKeepLast capacity (prim append context (cons token (nil UInt32)) : (List UInt32))),
  executable: true
}

definition{
  name: admitWith,
  parameters: [keep : Bool, capacity : Nat, context : (List UInt32)],
  result: (Option (List UInt32)),
  body: (if keep then (admitKeepLast capacity context) else (admitRefuse capacity context)),
  executable: true
}

definition{
  name: pushWith,
  parameters: [keep : Bool, capacity : Nat, context : (List UInt32), token : UInt32],
  result: (Option (List UInt32)),
  body: (if keep then (pushKeepLast capacity context token) else (pushRefuse capacity context token)),
  executable: true
}

definition{
  name: tokenIdFrom,
  parameters: [count : Nat, reached : UInt32],
  result: (Option UInt32),
  body: (match count
    | Nat.zero => (ctor Option.some [UInt32] reached)
    | Nat.succ previous => (match (prim checked_add reached (1 : UInt32) : (Option UInt32))
      | Option.some next => (tokenIdFrom previous next)
      | Option.none => (ctor Option.none [UInt32]))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: count
}

definition{
  name: tokenId,
  parameters: [index : Nat],
  result: (Option UInt32),
  body: (tokenIdFrom index (0 : UInt32)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: guard,
  parameters: [capacity : Nat, score : (Fn ((List UInt32)) (List Int)), context : (List UInt32)],
  result: (List Int),
  body: (if (fits capacity context) then (apply score context) else (nil Int)),
  executable: true
}

definition{
  name: firstMaxGo,
  parameters: [scores : (List Int), index : Nat, best : Nat, top : Int],
  result: Nat,
  body: (match scores
    | List.nil => best
    | List.cons score rest => (if (prim less_than top score : Bool) then (firstMaxGo rest (index + 1) index score) else (firstMaxGo rest (index + 1) best top))),
  axioms: [propext],
  executable: true,
  recursive_argument: scores
}

definition{
  name: firstMax,
  parameters: [scores : (List Int)],
  result: (Option Nat),
  body: (match scores
    | List.nil => (ctor Option.none [Nat])
    | List.cons score rest => (ctor Option.some [Nat] (firstMaxGo rest 1 0 score))),
  axioms: [propext],
  executable: true
}

definition{
  name: chooseGreedy,
  parameters: [scores : (List Int), seed : UInt32],
  result: ((Option Nat) * UInt32),
  body: ((firstMax scores), seed),
  axioms: [propext],
  executable: true
}

definition{
  name: genGo,
  parameters: [score : (Fn ((List UInt32)) (List Int)), choose : (Fn ((List Int) UInt32) ((Option Nat) * UInt32)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, context : (List UInt32), output : (List UInt32), seed : UInt32],
  result: (((List UInt32) * Nat) * UInt32),
  body: (match fuel
    | Nat.zero => ((output, 1), seed)
    | Nat.succ previous => (let step : ((Option Nat) * UInt32) := (apply choose (apply score context) seed); (match (first step)
      | Option.none => ((output, 2), (second step))
      | Option.some index => (if (index <? size) then (match (tokenId index)
        | Option.none => ((output, 2), (second step))
        | Option.some token => (if (prim equal token stop : Bool) then ((output, 0), (second step)) else (match (pushWith keep capacity context token)
          | Option.none => ((output, 3), (second step))
          | Option.some grown => (genGo score choose size capacity keep stop previous grown (prim append output (cons token (nil UInt32)) : (List UInt32)) (second step))))) else ((output, 2), (second step)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: generateWith,
  parameters: [score : (Fn ((List UInt32)) (List Int)), choose : (Fn ((List Int) UInt32) ((Option Nat) * UInt32)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, prompt : (List UInt32), seed : UInt32],
  result: (((List UInt32) * Nat) * UInt32),
  body: (match (admitWith keep capacity prompt)
    | Option.none => (((nil UInt32), 3), seed)
    | Option.some context => (genGo score choose size capacity keep stop fuel context (nil UInt32) seed)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: generateGreedy,
  parameters: [score : (Fn ((List UInt32)) (List Int)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, prompt : (List UInt32)],
  result: ((List UInt32) * Nat),
  body: (first (generateWith score (ref chooseGreedy) size capacity keep stop fuel prompt (0 : UInt32))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: byteVocab,
  parameters: [],
  result: Nat,
  body: 256,
  executable: true
}

definition{
  name: vocabSize,
  parameters: [vocabulary : (List String)],
  result: Nat,
  body: ((byteVocab) + (prim length vocabulary : Nat)),
  executable: true
}

definition{
  name: xorshift32,
  parameters: [value : UInt32],
  result: UInt32,
  body: (let stepOne : UInt32 := (match (prim shift_left value (13 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor value shifted : UInt32)
    | Option.none => value); (let stepTwo : UInt32 := (match (prim shift_right stepOne (17 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor stepOne shifted : UInt32)
    | Option.none => stepOne); (match (prim shift_left stepTwo (5 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor stepTwo shifted : UInt32)
    | Option.none => stepTwo))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: seedBits,
  parameters: [value : UInt32, bits : Nat, place : Int, accumulated : Int],
  result: Int,
  body: (match bits
    | Nat.zero => accumulated
    | Nat.succ previous => (let next : UInt32 := (match (prim shift_right value (1 : UInt32) : (Option UInt32))
      | Option.some shifted => shifted
      | Option.none => value); (if (prim equal (prim bit_and value (1 : UInt32) : UInt32) (1 : UInt32) : Bool) then (seedBits next previous (prim multiply place (2 : Int) : Int) (prim subtract accumulated (prim negate place : Int) : Int)) else (seedBits next previous (prim multiply place (2 : Int) : Int) accumulated)))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: bits
}

definition{
  name: seedInt,
  parameters: [value : UInt32],
  result: Int,
  body: (seedBits value 32 (1 : Int) (0 : Int)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: weight,
  parameters: [score : Int],
  result: Int,
  body: (if (prim less_than score (0 : Int) : Bool) then (0 : Int) else score),
  axioms: [propext],
  executable: true
}

definition{
  name: totalWeight,
  parameters: [scores : (List Int)],
  result: Int,
  body: (match scores
    | List.nil => (0 : Int)
    | List.cons score rest => (prim subtract (weight score) (prim negate (totalWeight rest) : Int) : Int)),
  axioms: [propext],
  executable: true,
  recursive_argument: scores
}

definition{
  name: drawGo,
  parameters: [scores : (List Int), index : Nat, remaining : Int],
  result: (Option Nat),
  body: (match scores
    | List.nil => (ctor Option.none [Nat])
    | List.cons score rest => (if (prim less_than remaining (weight score) : Bool) then (ctor Option.some [Nat] index) else (drawGo rest (index + 1) (prim subtract remaining (weight score) : Int)))),
  axioms: [propext],
  executable: true,
  recursive_argument: scores
}

definition{
  name: proportional,
  parameters: [scores : (List Int), seed : UInt32],
  result: (Option Nat),
  body: (if (prim less_than (totalWeight scores) (1 : Int) : Bool) then (firstMax scores) else (drawGo scores 0 (prim remainder (seedInt seed) (totalWeight scores) (0 : Int) : Int))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: chooseSample,
  parameters: [scores : (List Int), seed : UInt32],
  result: ((Option Nat) * UInt32),
  body: (let next : UInt32 := (xorshift32 seed); ((proportional scores next), next)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: generateSample,
  parameters: [score : (Fn ((List UInt32)) (List Int)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, prompt : (List UInt32), seed : UInt32],
  result: (((List UInt32) * Nat) * UInt32),
  body: (generateWith score (ref chooseSample) size capacity keep stop fuel prompt seed),
  axioms: [Quot.sound, propext],
  executable: true
}
