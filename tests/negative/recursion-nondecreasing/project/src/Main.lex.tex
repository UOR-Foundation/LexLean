\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: isEven,
  parameters: [number : Nat],
  result: Bool,
  body: (match number | Nat.zero => true | Nat.succ previous => (isOdd number)),
  mutual: Parity,
  recursive_argument: number
}

definition{
  name: isOdd,
  parameters: [number : Nat],
  result: Bool,
  body: (match number | Nat.zero => false | Nat.succ previous => (isEven previous)),
  mutual: Parity,
  recursive_argument: number
}
\end{semanticmodule}
\end{lexlean}
