definition{
  name: allowedLints,
  parameters: [],
  result: (List LintRow),
  body: (cons (record LintRow {lint := "type_complexity", group := (ctor ClippyGroup.complexity), reason := "asks for a type alias: a generated type is exactly its calculus type, and an alias is a name with no calculus counterpart"}) (cons (record LintRow {lint := "too_many_arguments", group := (ctor ClippyGroup.complexity), reason := "asks to group parameters into a struct: a function takes exactly its calculus parameters, and the struct is a type with no calculus counterpart"}) (cons (record LintRow {lint := "large_enum_variant", group := (ctor ClippyGroup.perf), reason := "asks to box a large variant: boxing changes the representation and allocates, which rust-core cannot and the calculus does not"}) (cons (record LintRow {lint := "result_large_err", group := (ctor ClippyGroup.perf), reason := "asks to box a large error: boxing changes the representation and allocates, which rust-core cannot and the calculus does not"}) (cons (record LintRow {lint := "result_unit_err", group := (ctor ClippyGroup.style), reason := "asks for an error type in place of unit: a function returns exactly its calculus result type, and the error type has no calculus counterpart"}) (cons (record LintRow {lint := "single_match", group := (ctor ClippyGroup.style), reason := "asks for `if let` or `if` in place of a match with an empty arm: every calculus match states an arm for every shape, and `if let` leaves one implicit"}) (cons (record LintRow {lint := "manual_unwrap_or", group := (ctor ClippyGroup.complexity), reason := "asks for a library combinator in place of the program's own match: the combinator is outside the closed runtime"}) (cons (record LintRow {lint := "manual_unwrap_or_default", group := (ctor ClippyGroup.suspicious), reason := "asks for a (cons (record LintRow {lint := "manual_map", group := (ctor ClippyGroup.style), reason := "asks for `Option::map` and a Rust closure in place of the program's own match: a calculus closure is match: the combinator is outside the closed runtime"}) (nil LintRow)))))))))))
}

definition{
  name: outsideCorrectness,
  parameters: [rows : (List LintRow)],
  result: Bool,
  body: (match rows
    | List.nil => true
    | List.cons row rest => (match (project row group)
      | ClippyGroup.correctness => false
      | ClippyGroup.suspicious => (outsideCorrectness rest)
      | ClippyGroup.style => (outsideCorrectness rest)
      | ClippyGroup.complexity => (outsideCorrectness rest)
      | ClippyGroup.perf => (outsideCorrectness rest)
      | ClippyGroup.pedantic => (outsideCorrectness rest)
      | ClippyGroup.restriction => (outsideCorrectness rest)
      | ClippyGroup.nursery => (outsideCorrectness rest)
      | ClippyGroup.cargo => (outsideCorrectness rest))),
  recursive_argument: rows
}

theorem{

  proof: reflexivity{}
}
