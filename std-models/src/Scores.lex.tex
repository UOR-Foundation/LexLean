definition{
  name: leaderStep,
  parameters: [state : (Nat * (Nat * Int)), score : Int],
  result: (Nat * (Nat * Int)),
  body: (if ((first state) =? 0) then (1, (0, score)) else (if (prim less_than (second (second state)) score : Bool) then (((first state) + 1), ((first state), score)) else (((first state) + 1), (second state)))),
  axioms: [propext],
  executable: true
}

definition{
  name: best,
  parameters: [scores : (List Int)],
  result: Nat,
  body: (first (second (prim list_fold (fun (state : (Nat * (Nat * Int))) (score : Int) => (leaderStep state score)) (0, (0, (0 : Int))) scores : (Nat * (Nat * Int))))),
  axioms: [propext],
  executable: true
}

definition{
  name: runnerStep,
  parameters: [state : (Nat * (Int * Int)), score : Int],
  result: (Nat * (Int * Int)),
  body: (if ((first state) =? 0) then (1, (score, (0 : Int))) else (if ((first state) =? 1) then (if (prim less_than (first (second state)) score : Bool) then (2, (score, (first (second state)))) else (2, ((first (second state)), score))) else (if (prim less_than (first (second state)) score : Bool) then (((first state) + 1), (score, (first (second state)))) else (((first state) + 1), ((first (second state)), (if (prim less_than (second (second state)) score : Bool) then score else (second (second state)))))))),
  axioms: [propext],
  executable: true
}

definition{
  name: leaders,
  parameters: [scores : (List Int)],
  result: (Int * Int),
  body: (second (prim list_fold (fun (state : (Nat * (Int * Int))) (score : Int) => (runnerStep state score)) (0, ((0 : Int), (0 : Int))) scores : (Nat * (Int * Int)))),
  axioms: [propext],
  executable: true
}

definition{
  name: margin,
  parameters: [scores : (List Int)],
  result: Int,
  body: (if ((prim length scores : Nat) <? 2) then (0 : Int) else (prim subtract (first (leaders scores)) (second (leaders scores)) : Int)),
  axioms: [propext],
  executable: true
}

definition{
  name: decideOr,
  parameters: [threshold : Int, scores : (List Int)],
  result: (Option Nat),
  body: (if ((prim length scores : Nat) <? 2) then (ctor Option.none [Nat]) else (if (prim less_than (margin scores) threshold : Bool) then (ctor Option.none [Nat]) else (ctor Option.some [Nat] (best scores)))),
  axioms: [propext],
  executable: true
}

definition{
  name: isConfident,
  type_parameters: [T],
  parameters: [candidate : (Candidate 'T)],
  result: Bool,
  body: (match candidate | Candidate.confident value => true | Candidate.unsure value => false),
  executable: true

definition{
  name: verdictOf,
  type_parameters: [T],
  parameters: [candidate : (Candidate 'T)],
  result: (Option 'T),
  body: (match candidate
    | Candidate.confident value => (ctor Option.some ['T] value)
    | Candidate.unsure value => (ctor Option.none ['T])),
  executable: true
}

definition{
  name: isCommitted,
  type_parameters: [T],
  parameters: [verdict : (Option 'T)],
  result: Bool,
  body: (match verdict | Option.some value => true | Option.none => false),
  executable: true
}

definition{
  name: Faithful,
