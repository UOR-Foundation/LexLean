    | RustAst::Expr.ELit literal at => (putLit out literal)

    | RustAst::Expr.EBlockE block => (printNode out block depth family slot)
    | RustAst::Expr.EProbe kind ident at => (match kind
      | RustValidate::ProbeKind.PkIsZero => (putS (putId out ident) " == 0")
      | RustValidate::ProbeKind.PkNonZero => (putS (putId out ident) " != 0")

      | RustValidate::ProbeKind.PkPred => (putS (putId out ident) " - 1"))

    | RustAst::Expr.EArm pat body rest => (let armed : LexLeanStd.Buffer::Buffer := (putS (printNode (putS (putPat (indent out (depth + 1)) pat) " => ") body (depth + 1) 100 0) "\n"); (printNode armed rest depth 100 0))

  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: node
}
