definition{
  name: absRun,
  parameters: [cells : RustMemory::Cells, length : Nat, onto : (List TargetSyntax::Value)],
  result: (List TargetSyntax::Value),
  body: (match length
    | Nat.zero => onto
    | Nat.succ less => (match cells
      | RustMemory::Cells.CNil => onto
      | RustMemory::Cells.CCons head tail => (absRun tail less (cons head onto)))),
  recursive_argument: length
}

definition{
  name: unconsAbs,
  parameters: [
    found : (Option TargetSyntax::Value),
    rest : (List TargetSyntax::Value),
    otherwise : (List TargetSyntax::Value)
  ],
  result: (List TargetSyntax::Value),
  body: (match found | Option.none => otherwise | Option.some head => (cons head rest))
}

definition{
  name: flagPick,
  parameters: [
    flag : RustMemory::Flag,
    whenYes : (List TargetSyntax::Value),
    whenNo : (List TargetSyntax::Value)
  ],
  result: (List TargetSyntax::Value),
  body: (match flag | RustMemory::Flag.yes => whenYes | RustMemory::Flag.no => whenNo)
}

definition{
  name: flipFlag,
  parameters: [flag : RustMemory::Flag],
  result: RustMemory::Flag,
  body: (match flag
    | RustMemory::Flag.yes => (ctor RustMemory::Flag.no)
    | RustMemory::Flag.no => (ctor RustMemory::Flag.yes))
}

definition{
  name: runWalk,
  parameters: [store : Store, address : Nat, length : Nat, onto : (List TargetSyntax::Value)],
  result: (List TargetSyntax::Value),
  body: (match store
    | Store.SEmpty => (absRun (ctor RustMemory::Cells.CNil) length onto)
    | Store.SCons b rest => (match address
      | Nat.zero => (absRun (project b slots) length onto)
      | Nat.succ lower => (runWalk rest lower length onto))),
  recursive_argument: store
}

definition{
  name: headAt,
  parameters: [store : Store, address : Nat, slot : Nat],
  result: (Option TargetSyntax::Value),
  body: (match store
    | Store.SEmpty => (ctor Option.none [TargetSyntax::Value])
    | Store.SCons b rest => (match address
      | Nat.zero => (RustMemory::get (project b slots) slot)
      | Nat.succ lower => (headAt rest lower slot))),
  recursive_argument: store
}

definition{
  name: bufAt,
  parameters: [store : Store, address : Nat],
  result: RustMemory::Buf,
  body: (match store
    | Store.SEmpty => (record RustMemory::Buf {slots := (ctor RustMemory::Cells.CNil), len := 0})
    | Store.SCons b rest => (match address | Nat.zero => b | Nat.succ lower => (bufAt rest lower))),
  recursive_argument: store
}

definition{
  name: keepOr,
  parameters: [result : (Option RustMemory::Buf), old : RustMemory::Buf],
  result: RustMemory::Buf,
  body: (match result | Option.none => old | Option.some got => got)
}

definition{
  name: consIn,
  parameters: [store : Store, address : Nat, length : Nat, item : TargetSyntax::Value],
  result: Store,
  body: (match store
    | Store.SEmpty => (ctor Store.SEmpty)
    | Store.SCons b rest => (match address
      | Nat.zero => (ctor Store.SCons (keepOr (RustMemory::bufPush (RustMemory::bufTruncate b length) item) b) rest)
      | Nat.succ lower => (ctor Store.SCons b (consIn rest lower length item)))),
  recursive_argument: store
}

definition{
  name: isFull,
  parameters: [buf : RustMemory::Buf, length : Nat, item : TargetSyntax::Value],
  result: RustMemory::Flag,
  body: (match (RustMemory::bufPush (RustMemory::bufTruncate buf length) item)
    | Option.none => (ctor RustMemory::Flag.yes)
    | Option.some got => (ctor RustMemory::Flag.no))
}

definition{
  name: storeLen,

  recursive_argument: store
}

  parameters: [store : Store, others : (List Handle), item : TargetSyntax::Value, handle : Handle],
  result: Landing,

      | Terminal.TNil => (ctor Landing.Landed (record World {store := (storeSnoc store (allocBuf item)), handle := (record Handle {spine := (nil TargetSyntax::Value), base := (ctor Terminal.TRun (storeLen store) 0)})}))
      | Terminal.TRun address more => (match (aloneOn address others)
