definition{
  name: leafStep,
  parameters: [
    ctx : Ctx,
    m : Mach,
    rest : (List Frame),
    depth : Nat,
    position : Nat,
    atom : PrimitiveScan::Atom,
    leaves : (List Edge)
  ],
  result: Mach,
  body: (let stop : Nat := (project ctx stop); (let identifier : (Option Composed) := (if ((PrimitiveScan::classCode atom) =? 1) then (let composed : Composed := (composedAt ctx atom position); (if ((project composed next) <=? stop) then (ctor Option.some [Composed] composed) else (ctor Option.none [Composed]))) else (ctor Option.none [Composed])); (let leafEnds : (List Nat) := (prim list_fold (fun (sorted : (List Nat)) (edge : Edge) => (insertNat sorted (project edge stop))) (nil Nat) leaves : (List Nat)); (let ends : (List Nat) := (match identifier
    | Option.some composed => (insertNat leafEnds (project composed next))
    | Option.none => leafEnds); (match (prim index ends 0 : (Option Nat))
    | Option.none => (failed m (placed (lit "LLL1004") (cat (cat (lit "unknown atom `") (project atom text)) (lit "` in a mathematical island")) (placeAt ctx position)))
    | Option.some shortest => (let upto : Nat := (match (chosenAt ctx position)
      | Option.some picked => (if (natIn picked ends) then picked else shortest)
      | Option.none => shortest); (let met : (List Met) := (if (1 <? (prim length ends : Nat)) then (cons (record Met {position := position, ends := ends}) (project m met)) else (project m met)); (let forms : (List LeafKind) := (prim list_fold (fun [upto] (kinds : (List LeafKind)) (edge : Edge) => (if ((project edge stop) =? upto) then (cons (ctor LeafKind.LForm (project edge form)) kinds) else kinds)) (nil LeafKind) leaves : (List LeafKind)); (let kinds : (List LeafKind) := (prim list_fold (fun (turned : (List LeafKind)) (kind : LeafKind) => (cons kind turned)) (match identifier
      | Option.some composed => (if ((project composed next) =? upto) then (cons (ctor LeafKind.LIdent (project composed spelled)) (nil LeafKind)) else (nil LeafKind))
      | Option.none => (nil LeafKind)) forms : (List LeafKind)); (let leaf : Val := (mathVal (ctor Math.MLeaf kinds position upto) position upto (cat (cat (cat (lit "L") (kindCode kinds)) (encN position)) (encN upto))); (record Mach {cursor := upto, stack := (cons (ctor Frame.FPostfix depth) rest), value := (ctor Option.some [Val] leaf), error := (project m error), bud := (project m bud), met := met}))))))))))),
  executable: true
}

    | Option.some problem => (failed charged problem)
    | Option.none => (let stop : Nat := (project ctx stop); (let prefixes : (List Edge) := (keepEdges (project looked found) stop 2 2); (let leaves : (List Edge) := (keepEdges (project looked found) stop 0 1); (match (prim index prefixes 0 : (Option Edge))
