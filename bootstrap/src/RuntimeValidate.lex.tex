definition{

    | Expr.ELit x0 => false
    | Expr.EPath x0 => false
    | Expr.ECall x0 x1 => false
    | Expr.EMethod receiver x1 x2 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.EField receiver x1 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.EFieldNamed receiver x1 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.EBorrow inner => (if (12 <=? (exprPrec inner)) then (exposed inner) else false)
    | Expr.EEquals a b => (if (if (6 <=? (exprPrec a)) then (exposed a) else false) then true else (if (6 <=? (exprPrec b)) then (exposed b) else false))
    | Expr.EMatch x0 x1 => false
    | Expr.ELet x0 x1 x2 => false
    | Expr.EAssign place value => (if (exposed place) then true else (exposed value))
    | Expr.EIf x0 x1 x2 => false
    | Expr.EWhile x0 x1 => false
    | Expr.EWhileLet x0 x1 x2 => false
    | Expr.ELoop x0 => false
    | Expr.EFor x0 x1 x2 => false
    | Expr.EBreak => false
    | Expr.EReturn value => (exposed value)
    | Expr.ETuple x0 => false
    | Expr.EStructLit x0 x1 x2 => true
    | Expr.EBinary op a b => (if (if (leftParenthesized op a) then false else (exposed a)) then true else (if (((opPrec op) + 1) <=? (exprPrec b)) then (exposed b) else false))
    | Expr.ENot inner => (if (12 <=? (exprPrec inner)) then (exposed inner) else false)
    | Expr.ECast inner x1 => (if (11 <=? (exprPrec inner)) then (exposed inner) else false)
    | Expr.EIndex receiver x1 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.ERange receiver x1 x2 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.EDeref inner => (if (12 <=? (exprPrec inner)) then (exposed inner) else false)
    | Expr.EBorrowMut inner => (if (12 <=? (exprPrec inner)) then (exposed inner) else false)
    | Expr.EMethodOf receiver x1 x2 x3 => (if (13 <=? (exprPrec receiver)) then (exposed receiver) else false)
    | Expr.EBlock x0 x1 => false
    | Expr.ENone => false
    | Expr.ENil => false
    | Expr.ECons x0 x1 => false
    | Expr.ELetS x0 x1 x2 x3 => false
    | Expr.EArm x0 x1 x2 x3 => false),
  axioms: [],
  executable: true,
  recursive_argument: e

definition{
  name: unscopedErr,
  parameters: [what : String, ty : Ty],
  result: (Option Bytes),
  body: (match ty
    | Ty.TyUnit => (ctor Option.none [Bytes])
    | Ty.TyBool => (ctor Option.none [Bytes])
    | Ty.TyNat => (ctor Option.none [Bytes])
    | Ty.TyFixed => (ctor Option.none [Bytes])
    | Ty.TyOrdering => (ctor Option.none [Bytes])
    | Ty.TySelf => (ctor Option.none [Bytes])
    | Ty.TyParam x0 => (ctor Option.some [Bytes] (rcat (rtext "runtime support: type-scope: ") (rcat (rtext what) (rtext " names a generic parameter, which the printer writes without the generics in scope"))))
    | Ty.TyRef inner => (unscopedErr what inner)

definition{
  name: exprOk,
  parameters: [sc : Sc, fr : Fr, e : Expr],
  result: (Option Bytes),
  body: (match e
    | Expr.EVar name => (match (findVar (project fr vars) name)
      | Option.some held => (ctor Option.none [Bytes])
      | Option.none => (if (inList (project sc statics) name) then (ctor Option.none [Bytes]) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: variable-scope: ") (rcat (rtick name) (rtext " is not in scope"))))))
    | Expr.ELit scalars => (literalErr scalars)
    | Expr.EPath path => (pathOk sc path)
    | Expr.ECall path args => (match (callCheck sc path (exprLen args))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprsOk sc fr args))
    | Expr.EMethod recv name args => (match (exprOk sc fr recv)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (methodOk sc fr recv name (exprLen args))
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (exprsOk sc fr args)))
    | Expr.EField inner at => (exprOk sc fr inner)
    | Expr.EFieldNamed inner name => (match (identErr "a field access" name)
      | Option.some bad => (ctor Option.some [Bytes] bad)
      | Option.none => (exprOk sc fr inner))
    | Expr.EBorrow inner => (exprOk sc fr inner)
    | Expr.EEquals a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.EMatch scrutinee arms => (match (orElse (headErr scrutinee) (exprOk sc fr scrutinee))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (if (isEmptyChain arms) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: match-arms: ") (rtext "a match with no arm"))) else (armsOk sc fr arms)))
    | Expr.ELet pat ty value => (ctor Option.some [Bytes] (rcat (rtext "runtime support: let-position: ") (rtext "a `let` that is not a statement of a block")))
    | Expr.EAssign place value => (match (assignable fr place)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (exprOk sc fr place)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (exprOk sc fr value)))
    | Expr.EIf cond thenBlock elseBlock => (match (orElse (headErr cond) (exprOk sc fr cond))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (blockOk sc fr thenBlock)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (blockOk sc fr elseBlock)))
    | Expr.EWhile cond body => (match (orElse (headErr cond) (exprOk sc fr cond))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (blockOk sc (withBound fr (nil Var) true) body))
    | Expr.EWhileLet pat value body => (match (orElse (headErr value) (exprOk sc fr value))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (patOk sc pat)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match (bindsOf (nil Var) pat)
          | Result.error problem => (ctor Option.some [Bytes] problem)
          | Result.ok bound => (blockOk sc (withBound fr bound true) body))))
    | Expr.ELoop body => (blockOk sc (withBound fr (nil Var) true) body)
    | Expr.EFor pat value body => (match (orElse (headErr value) (exprOk sc fr value))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (patOk sc pat)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match (bindsOf (nil Var) pat)
          | Result.error problem => (ctor Option.some [Bytes] problem)
          | Result.ok bound => (blockOk sc (withBound fr bound true) body))))
    | Expr.EBreak => (if (project fr inLoop) then (ctor Option.none [Bytes]) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: break-position: ") (rtext "a `break` outside a loop"))))
    | Expr.EReturn value => (if (isNone value) then (ctor Option.none [Bytes]) else (exprOk sc fr value))
    | Expr.ETuple items => (if ((exprLen items) <? 2) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: tuple-arity: ") (rcat (rtext "a tuple of ") (rdec (exprLen items))))) else (exprsOk sc fr items))
    | Expr.EStructLit path names values => (match (pathOk sc path)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (recordFields sc path)
        | Option.none => (ctor Option.some [Bytes] (rcat (rtext "runtime support: struct-fields: ") (rcat (rtick (joinWith path "::")) (rtext " is no record struct or record variant"))))
        | Option.some held => (if (sameFields names held (exprLen values)) then (exprsOk sc fr values) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: struct-fields: ") (rcat (rtick (joinWith path "::")) (rcat (rtext " is given ") (rcat (joinWith names ", ") (rcat (rtext ", not ") (joinWith held ", "))))))))))
    | Expr.EBinary op a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.ENot inner => (exprOk sc fr inner)
    | Expr.ECast inner ty => (match (orElse (tyOk sc ty (prim length (project fr generics) : Nat) (not (isAbsent (project fr selfTy)))) (unscopedErr "a cast type" ty))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr inner))
    | Expr.EIndex a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.ERange inner lower upper => (match (exprOk sc fr inner)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (if (isNone lower) then (ctor Option.none [Bytes]) else (exprOk sc fr lower))
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (if (isNone upper) then (ctor Option.none [Bytes]) else (exprOk sc fr upper))))
    | Expr.EDeref inner => (exprOk sc fr inner)
    | Expr.EBorrowMut inner => (exprOk sc fr inner)
    | Expr.EMethodOf recv name tys args => (match (exprOk sc fr recv)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (turbofishTypes name)
        | Option.none => (ctor Option.some [Bytes] (rcat (rtext "runtime support: method-name: ") (rcat (rtick name) (rtext " takes no type arguments in the closed table"))))
        | Option.some count => (if (if (count =? (tyLen tys)) then ((turbofishArgs name) =? (exprLen args)) else false) then (if (if (project sc core) then (bEq name (rtext "collect")) else false) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: hidden-allocation: ") (rcat (rtext "rust-core renders ") (rcat (rtick name) (rtext ", which requires heap allocation"))))) else (match (orElse (tyOk sc tys (prim length (project fr generics) : Nat) (not (isAbsent (project fr selfTy)))) (unscopedErr "a type argument" tys))
          | Option.some stop => (ctor Option.some [Bytes] stop)
          | Option.none => (exprsOk sc fr args))) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: method-arity: ") (rcat (rtick name) (rcat (rtext " takes ") (rcat (rdec count) (rcat (rtext " type arguments and ") (rcat (rdec (turbofishArgs name)) (rtext " arguments")))))))))))
    | Expr.EBlock stmts tail => (match (stmtsOk sc fr stmts)
      | Result.error problem => (ctor Option.some [Bytes] problem)
      | Result.ok inner => (exprOk sc inner tail))
    | Expr.ENone => (ctor Option.none [Bytes])
    | Expr.ENil => (ctor Option.none [Bytes])
    | Expr.ECons d0 d1 => (ctor Option.none [Bytes])
    | Expr.ELetS d0 d1 d2 d3 => (ctor Option.none [Bytes])
    | Expr.EArm d0 d1 d2 d3 => (ctor Option.none [Bytes])),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: CheckRt,
  recursive_argument: e
}

definition{

  parameters: [sc : Sc, fr : Fr, es : Expr],
  result: (Option Bytes),

        | Option.none => (if (inList (project seen names) (variantName variant)) then (record Seen {names := (project seen names), dup := (ctor Option.some [Bytes] (rcat (rtext "runtime support: variant-name: ") (rcat (rtick enumName) (rcat (rtext " declares ") (rcat (rtick (variantName variant)) (rtext " twice"))))))}) else (record Seen {names := (cons (variantName variant) (project seen names)), dup := (ctor Option.none [Bytes])})))))) (record Seen {names := (nil Bytes), dup := (ctor Option.none [Bytes])}) variants : Seen) dup),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: pass1Item,
  parameters: [sc : Sc, item : Item],
  result: (Result Sc Bytes),
  body: (match item
    | Item.IUse path alias => (postCheck (match path
      | List.nil => (ctor Result.error [Sc, Bytes] (rtext "runtime support: use-path: empty"))
      | List.cons root more => (if (if (bEq root (rtext "core")) then true else (if (project sc core) then false else (bEq root (rtext "std")))) then (ctor Result.ok [Sc, Bytes] (record Sc {core := (project sc core), declared := (project sc declared), traits := (project sc traits), structs := (project sc structs), enums := (project sc enums), aliases := (project sc aliases), statics := (project sc statics), functions := (project sc functions), methods := (project sc methods), imported := (prim append (project sc imported) (cons (match alias
        | List.nil => (lastName path)
        | List.cons renamed rest => renamed) (nil Bytes)) : (List Bytes))})) else (ctor Result.error [Sc, Bytes] (rcat (rtext "runtime support: profile-root: ") (rcat (rtick (rcat (rtext "use ") (joinWith path "::"))) (rcat (rtext " starts at ") (rcat (rtick root) (rtext ", which the profile does not carry")))))))) (orElse (segmentsErr "a `use` segment" path) (identsErr "a `use` alias" alias)))
    | Item.IUnit name derived => (preCheck (identErr "a unit struct name" name) (postCheck (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (prim append (project s2 structs) (cons (record StructE {name := name, fields := (nil Bytes), gcount := 0}) (nil StructE)) : (List StructE)), enums := (project s2 enums), aliases := (project s2 aliases), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}))) (derivesErr "a unit struct" derived)))
    | Item.IAlias name params ty => (preCheck (identErr "an alias name" name) (postCheck (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (project s2 structs), enums := (project s2 enums), aliases := (prim append (project s2 aliases) (cons (record Entry {name := name, n := (prim length params : Nat)}) (nil Entry)) : (List Entry)), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}))) (identsErr "an alias parameter" params)))
    | Item.IStatic name ty init => (preCheck (identErr "a static name" name) (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (project s2 structs), enums := (project s2 enums), aliases := (project s2 aliases), statics := (prim append (project s2 statics) (cons name (nil Bytes)) : (List Bytes)), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}))))
    | Item.ITrait name sigs => (preCheck (identErr "a trait name" name) (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (addMethods (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (prim append (project s2 traits) (cons (record TraitE {name := name, sigs := sigs}) (nil TraitE)) : (List TraitE)), structs := (project s2 structs), enums := (project s2 enums), aliases := (project s2 aliases), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}) sigs))))
    | Item.IImpl generics tr ty fns => (ctor Result.ok [Sc, Bytes] (addFnMethods sc fns))
    | Item.IFn f => (preCheck (identErr "a function name" (project (project f sig) name)) (match (declare sc (project (project f sig) name))
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (project s2 structs), enums := (project s2 enums), aliases := (project s2 aliases), statics := (project s2 statics), functions := (prim append (project s2 functions) (cons (record Entry {name := (project (project f sig) name), n := (prim length (project (project f sig) params) : Nat)}) (nil Entry)) : (List Entry)), methods := (project s2 methods), imported := (project s2 imported)}))))
    | Item.ITuple name generics derived tys => (preCheck (identErr "a tuple struct name" name) (postCheck (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (prim append (project s2 structs) (cons (record StructE {name := name, fields := (indexNames tys 0), gcount := (prim length generics : Nat)}) (nil StructE)) : (List StructE)), enums := (project s2 enums), aliases := (project s2 aliases), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}))) (orElse (identsErr "a tuple struct generic" generics) (derivesErr "a tuple struct" derived))))
    | Item.IRecord name generics derived fields tys => (preCheck (identErr "a record struct name" name) (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (match (orElse (identsErr "a record struct generic" generics) (orElse (derivesErr "a record struct" derived) (identsErr "a record field name" fields)))
        | Option.some stop => (ctor Result.error [Sc, Bytes] stop)
        | Option.none => (if ((prim length fields : Nat) =? (tyLen tys)) then (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (prim append (project s2 structs) (cons (record StructE {name := name, fields := fields, gcount := (prim length generics : Nat)}) (nil StructE)) : (List StructE)), enums := (project s2 enums), aliases := (project s2 aliases), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)})) else (ctor Result.error [Sc, Bytes] (rcat (rtext "runtime support: struct-fields: ") (rcat (rtick name) (rcat (rtext " has ") (rcat (rdec (prim length fields : Nat)) (rcat (rtext " field names and ") (rcat (rdec (tyLen tys)) (rtext " types"))))))))))))
    | Item.IEnum name generics derived variants => (preCheck (identErr "an enum name" name) (match (declare sc name)
      | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
      | Result.ok s2 => (match (orElse (identsErr "an enum generic" generics) (orElse (derivesErr "an enum" derived) (variantProblem name variants)))
        | Option.some stop => (ctor Result.error [Sc, Bytes] stop)
        | Option.none => (match (firstDuplicate variants)
        | Option.some twice => (ctor Result.error [Sc, Bytes] (rcat (rtext "runtime support: variant-name: ") (rcat (rtick name) (rcat (rtext " declares ") (rcat (rtick twice) (rtext " twice"))))))
        | Option.none => (ctor Result.ok [Sc, Bytes] (record Sc {core := (project s2 core), declared := (project s2 declared), traits := (project s2 traits), structs := (project s2 structs), enums := (prim append (project s2 enums) (cons (record EnumE {name := name, variants := variants, gcount := (prim length generics : Nat)}) (nil EnumE)) : (List EnumE)), aliases := (project s2 aliases), statics := (project s2 statics), functions := (project s2 functions), methods := (project s2 methods), imported := (project s2 imported)}))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: pass1,
  parameters: [sc : Sc, items : (List Item)],
  result: (Result Sc Bytes),
  body: (prim list_fold (fun (state : (Result Sc Bytes)) (item : Item) => (match state
    | Result.error problem => (ctor Result.error [Sc, Bytes] problem)
    | Result.ok held => (pass1Item held item))) (ctor Result.ok [Sc, Bytes] sc) items : (Result Sc Bytes)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: pass2Item,
  parameters: [sc : Sc, item : Item],
  result: (Option Bytes),
  body: (match item
    | Item.IUse path alias => (ctor Option.none [Bytes])
    | Item.IUnit name derived => (ctor Option.none [Bytes])
    | Item.IAlias name params ty => (tyOk sc ty (prim length params : Nat) false)
    | Item.IStatic name ty init => (match (tyOk sc ty 0 false)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc (emptyFr ()) init))
    | Item.ITrait name sigs => (match (match (firstDuplicateSig sigs)
        | Option.some twice => (ctor Option.some [Bytes] (rcat (rtext "runtime support: trait-method: ") (rcat (rtick name) (rcat (rtext " declares ") (rcat (rtick twice) (rtext " twice"))))))
        | Option.none => (ctor Option.none [Bytes]))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (prim list_fold (fun [sc] (found : (Option Bytes)) (sig : Sig) => (match found
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (sigOk sc sig 0 true))) (ctor Option.none [Bytes]) sigs : (Option Bytes)))
    | Item.IImpl generics tr ty fns => (match (genericsOk sc generics)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (tyOk sc ty (prim length generics : Nat) false)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match tr
          | List.cons named more => (completes sc named generics ty fns)
          | List.nil => (prim list_fold (fun [generics, sc, ty] (found : (Option Bytes)) (f : FnDef) => (match found
            | Option.some stop => (ctor Option.some [Bytes] stop)
            | Option.none => (fnOk sc generics ty f))) (ctor Option.none [Bytes]) fns : (Option Bytes)))))
    | Item.IFn f => (if (not (project f public)) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: function-visibility: ") (rcat (rtick (project (project f sig) name)) (rtext " is a free function that is not public")))) else (if (project (project f sig) receiver) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: function-receiver: ") (rcat (rtext "free function ") (rcat (rtick (project (project f sig) name)) (rtext " takes `&self`"))))) else (fnOk sc (nil Generic) (ctor Ty.TyAbsent) f)))
    | Item.ITuple name generics derived tys => (tyOk sc tys (prim length generics : Nat) false)
    | Item.IRecord name generics derived fields tys => (tyOk sc tys (prim length generics : Nat) false)
    | Item.IEnum name generics derived variants => (prim list_fold (fun [generics, sc] (found : (Option Bytes)) (variant : Variant) => (match found
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match variant
        | Variant.VUnit n => (ctor Option.none [Bytes])
        | Variant.VTuple n tys => (tyOk sc tys (prim length generics : Nat) false)
        | Variant.VRecord n fs tys => (tyOk sc tys (prim length generics : Nat) false)))) (ctor Option.none [Bytes]) variants : (Option Bytes))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: validateRuntime,
  parameters: [krate : Crate],
  result: (Result Unit Bytes),
  body: (match (pass1 (record Sc {core := (project krate core), declared := (nil Bytes), traits := (nil TraitE), structs := (nil StructE), enums := (nil EnumE), aliases := (nil Entry), statics := (nil Bytes), functions := (nil Entry), methods := (nil Entry), imported := (nil Bytes)}) (project krate items))
    | Result.error problem => (ctor Result.error [Unit, Bytes] problem)
    | Result.ok sc => (match (prim list_fold (fun [sc] (found : (Option Bytes)) (item : Item) => (match found
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (pass2Item sc item))) (ctor Option.none [Bytes]) (project krate items) : (Option Bytes))
      | Option.some problem2 => (ctor Result.error [Unit, Bytes] problem2)
      | Option.none => (ctor Result.ok [Unit, Bytes] ()))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}
