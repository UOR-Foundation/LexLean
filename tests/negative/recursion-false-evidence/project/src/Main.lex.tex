\title{Natural number addition}
\begin{semanticmodule}
theorem{
  name: stall_decreases,
  parameters: [number : Nat],
  statement: (((number <? 2) = false) -> (number < number)),
  proof: linear_arithmetic{},
  axioms: [Quot.sound, propext]
}

definition{
  name: stall,
  parameters: [number : Nat],
  result: Nat,
  body: (if (number <? 2) then 0 else (stall number)),
  axioms: [Quot.sound, propext],
  termination: {evidence: [stall_decreases], measure: number}
}
\end{semanticmodule}
