    | SemanticType.SParameter named => named
    | SemanticType.SOption inner => (cat (lit "Option (") (cat (typeDisplay inner) (lit ")")))
    | SemanticType.SResult good bad => (cat (cat (lit "Result (") (typeDisplay good)) (cat (cat (lit ") (") (typeDisplay bad)) (lit ")")))
    | SemanticType.SList inner => (cat (lit "List (") (cat (typeDisplay inner) (lit ")")))
    | SemanticType.SNamed owner named arguments => (cat (match owner
      | Option.some source => (cat (cat source (lit ".")) named)
      | Option.none => named) (prim list_fold (fun (acc : Bytes) (shown : Bytes) => (cat acc (cat (lit " (") (cat shown (lit ")"))))) 0x (displayItems arguments) : Bytes))
    | SemanticType.SProduct left right => (cat (cat (lit "Prod (") (typeDisplay left)) (cat (cat (lit ") (") (typeDisplay right)) (lit ")")))
    | SemanticType.SFunction parameters outcome => (cat (cat (lit "(") (prim list_fold (fun (acc : Bytes) (shown : Bytes) => (cat acc (cat (lit "(") (cat shown (lit ") -> "))))) 0x (displayItems parameters) : Bytes)) (cat (lit "(") (cat (typeDisplay outcome) (lit "))"))))
    | SemanticType.SMap key entry => (cat (cat (lit "Map (") (typeDisplay key)) (cat (cat (lit ") (") (typeDisplay entry)) (lit ")")))
    | SemanticType.SSet inner => (cat (lit "Set (") (cat (typeDisplay inner) (lit ")")))
    | SemanticType.SNil => 0x
    | SemanticType.SItem head tail => 0x),
  executable: true,
  mutual: Display,
  recursive_argument: semantic
}
