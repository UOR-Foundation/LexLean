definition{
  name: rtToken,
  parameters: [maxDepth : Nat, st : Rt, tok : Tok],
  result: Rt,
  body: (match tok
    | Tok.TOpen => (if ((prim length (project st stack) : Nat) <? maxDepth) then (record Rt {pos := ((project st pos) + 1), stack := (cons (nil Node) (project st stack)), root := (project st root), error := (ctor Option.none [Failure]), done := false}) else (rtFail st (record Failure {kind := 1, message := (cat (lit "max_scope_depth exceeded: configured ") (LexLeanStd.Numeral::decimal maxDepth))})))
    | Tok.TClose => (match (project st stack)
      | List.nil => (rtFail st (failed (lit "unbalanced `)`")))
      | List.cons top rest => (rtPlace st (ctor Node.NList (chainOf top)) rest))
    | Tok.TAtom text => (rtPlace st (ctor Node.NAtom text) (project st stack))),
  executable: true

  name: rtStep,
  parameters: [toks : (List Tok), maxDepth : Nat, st : Rt],
  result: (Option Rt),
  body: (match (project st error)
    | Option.some stopped => (ctor Option.none [Rt])
    | Option.none => (if (project st done) then (ctor Option.none [Rt]) else (match (prim index toks (project st pos) : (Option Tok))
      | Option.none => (match (project st stack)
        | List.nil => (match (project st root)
          | Option.some found => (ctor Option.some [Rt] (record Rt {pos := (project st pos), stack := (nil (List Node)), root := (project st root), error := (ctor Option.none [Failure]), done := true}))
          | Option.none => (ctor Option.some [Rt] (rtFail st (failed (lit "unexpected end of expression")))))
        | List.cons top rest => (ctor Option.some [Rt] (rtFail st (failed (lit "unexpected end of expression")))))
      | Option.some tok => (match (project st root)
        | Option.some found => (ctor Option.some [Rt] (rtFail st (failed (lit "trailing tokens after the expression"))))
        | Option.none => (ctor Option.some [Rt] (rtToken maxDepth st tok)))))),
  executable: true
}

definition{
  name: uvVisit,
  parameters: [fuel : Nat, st : Uv, rest : (List UWork), node : Node],
  result: Uv,
  body: (match node
    | Node.NAtom text => (if (allDigits text) then (match (decimalFault text)
      | Option.some failure => (uvFail st failure)
      | Option.none => (record Uv {work := rest, values := (cons (ctor Universe.UNum (decimalValue text)) (project st values)), error := (ctor Option.none [Failure])})) else (if (isLseIdentifier text) then (record Uv {work := rest, values := (cons (ctor Universe.UVar text) (project st values)), error := (ctor Option.none [Failure])}) else (uvFail st (failed (cat (quoted text) (lit " is not a universe"))))))
    | Node.NList chain => (let items : (List Node) := (itemsOf chain fuel); (match items
      | List.nil => (uvFail st (failed (lit "expected a universe form")))
      | List.cons headNode arguments => (match headNode
        | Node.NAtom head => (let argc : Nat := (prim length arguments : Nat); (let known : Bool := (if (prim equal head (lit "succ") : Bool) then true else (if (prim equal head (lit "max") : Bool) then true else (prim equal head (lit "imax") : Bool))); (let fits : Bool := (if (prim equal head (lit "succ") : Bool) then (argc =? 1) else (if (prim equal head (lit "max") : Bool) then (2 <=? argc) else (if (prim equal head (lit "imax") : Bool) then (argc =? 2) else false))); (if fits then (record Uv {work := (pushVisitsU arguments (cons (ctor UWork.UBuild head argc) rest)), values := (project st values), error := (ctor Option.none [Failure])}) else (if known then (uvFail st (failed (cat (cat (lit "`(") head) (lit " ...)` has the wrong arity")))) else (uvFail st (failed (cat (quoted head) (lit " is not a universe form")))))))))
        | Node.NList inner => (uvFail st (failed (lit "expected a universe form")))
        | Node.NNil => (uvFail st (failed (lit "expected a universe form")))
        | Node.NItem h t => (uvFail st (failed (lit "expected a universe form"))))))
    | Node.NNil => (uvFail st (failed (lit "expected a universe form")))
    | Node.NItem head tail => (uvFail st (failed (lit "expected a universe form")))),
  axioms: [Quot.sound, propext],
  executable: true
}

        | Node.NItem h t => (cvFail st (failed (lit "expected a form, found a list"))))))
    | Node.NAtom text => (cvFail st (failed (lit "expected a list")))
    | Node.NNil => (cvFail st (failed (lit "expected a list")))
