definition{
  name: zonk,
  parameters: [fuel : Fuel, metas : (List (Bytes * LexiconLse::Lse)), expr : LexiconLse::Lse],
  result: (Option LexiconLse::Lse),
  body: (match fuel
    | Fuel.FZ => (ctor Option.none [LexiconLse::Lse])
    | Fuel.FS f => (match expr
      | LexiconLse::Lse.LLocal name => (if (isMeta name) then (match (solvedOf metas name)
        | Option.some solution => (zonk f metas solution)
        | Option.none => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LLocal name))) else (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LLocal name)))
      | LexiconLse::Lse.LApp g args => (match (both (zonk f metas g) (zonk f metas args))
        | Option.some zs => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LApp (first zs) (second zs)))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LArg h t => (match (both (zonk f metas h) (zonk f metas t))
        | Option.some zs => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LArg (first zs) (second zs)))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LPi bs body => (match (both (zonk f metas bs) (zonk f metas body))
        | Option.some zs => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LPi (first zs) (second zs)))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LLam bs body => (match (both (zonk f metas bs) (zonk f metas body))
        | Option.some zs => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LLam (first zs) (second zs)))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LBinder m n t r => (match (both (zonk f metas t) (zonk f metas r))
        | Option.some zs => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LBinder m n (first zs) (second zs)))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LLet n t v body => (match (both (zonk f metas t) (zonk f metas v))
        | Option.some zs => (match (zonk f metas body)
          | Option.some zbody => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LLet n (first zs) (second zs) zbody))
          | Option.none => (ctor Option.none [LexiconLse::Lse]))
        | Option.none => (ctor Option.none [LexiconLse::Lse]))
      | LexiconLse::Lse.LSortProp => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LSortProp))
      | LexiconLse::Lse.LSortType u => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LSortType u))
      | LexiconLse::Lse.LConst p e us => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LConst p e us))
      | LexiconLse::Lse.LNat digits => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LNat digits))
      | LexiconLse::Lse.LNil => (ctor Option.some [LexiconLse::Lse] (ctor LexiconLse::Lse.LNil)))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: freeLocals,
  parameters: [fuel : Fuel, expr : LexiconLse::Lse, bound : (List Bytes)],
  result: (Option (List Bytes)),
  body: (match fuel
    | Fuel.FZ => (ctor Option.none [(List Bytes)])
    | Fuel.FS f => (match expr
      | LexiconLse::Lse.LLocal name => (if (boundIn bound name) then (ctor Option.some [(List Bytes)] (nil Bytes)) else (ctor Option.some [(List Bytes)] (cons name (nil Bytes))))
      | LexiconLse::Lse.LApp g args => (unionOf (freeLocals f g bound) (freeLocals f args bound))
      | LexiconLse::Lse.LArg h t => (unionOf (freeLocals f h bound) (freeLocals f t bound))
      | LexiconLse::Lse.LPi bs body => (freeBinders f bs bound body)
      | LexiconLse::Lse.LLam bs body => (freeBinders f bs bound body)
      | LexiconLse::Lse.LLet n t v body => (unionOf (unionOf (freeLocals f t bound) (freeLocals f v bound)) (freeLocals f body (cons n bound)))
      | LexiconLse::Lse.LSortProp => (ctor Option.some [(List Bytes)] (nil Bytes))
      | LexiconLse::Lse.LSortType u => (ctor Option.some [(List Bytes)] (nil Bytes))
      | LexiconLse::Lse.LConst p e us => (ctor Option.some [(List Bytes)] (nil Bytes))
      | LexiconLse::Lse.LNat digits => (ctor Option.some [(List Bytes)] (nil Bytes))
      | LexiconLse::Lse.LNil => (ctor Option.some [(List Bytes)] (nil Bytes))
      | LexiconLse::Lse.LBinder m n t r => (ctor Option.some [(List Bytes)] (nil Bytes)))),
  executable: true,
  mutual: Free,
  recursive_argument: fuel
}

definition{
  name: freeBinders,
  parameters: [fuel : Fuel, chain : LexiconLse::Lse, bound : (List Bytes), body : LexiconLse::Lse],
  result: (Option (List Bytes)),
  body: (match fuel
    | Fuel.FZ => (ctor Option.none [(List Bytes)])
    | Fuel.FS f => (match chain
      | LexiconLse::Lse.LBinder m n t r => (unionOf (freeLocals f t bound) (freeBinders f r (cons n bound) body))
      | LexiconLse::Lse.LNil => (freeLocals f body bound)
      | LexiconLse::Lse.LSortProp => (freeLocals f body bound)
      | LexiconLse::Lse.LSortType u => (freeLocals f body bound)
      | LexiconLse::Lse.LConst p e us => (freeLocals f body bound)
      | LexiconLse::Lse.LLocal name => (freeLocals f body bound)
      | LexiconLse::Lse.LApp g args => (freeLocals f body bound)
      | LexiconLse::Lse.LPi bs inner => (freeLocals f body bound)
      | LexiconLse::Lse.LLam bs inner => (freeLocals f body bound)
      | LexiconLse::Lse.LLet n t v inner => (freeLocals f body bound)
      | LexiconLse::Lse.LNat digits => (freeLocals f body bound)
      | LexiconLse::Lse.LArg h t => (freeLocals f body bound))),
  executable: true,
  mutual: Free,
  recursive_argument: fuel
}

definition{
  name: substIn,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    name : Bytes,
    replacement : LexiconLse::Lse,
    avoid : (List Bytes)
  ],
  result: Sr,
  body: (match fuel
    | Fuel.FZ => (record Sr {out := expr, ck := (ckExhaust st)})
    | Fuel.FS f => (match expr
      | LexiconLse::Lse.LLocal lname => (if (prim equal lname name : Bool) then (record Sr {out := replacement, ck := st}) else (record Sr {out := (ctor LexiconLse::Lse.LLocal lname), ck := st}))
      | LexiconLse::Lse.LApp g args => (let sg : Sr := (substIn f st g name replacement avoid); (let sa : Sr := (substIn f (project sg ck) args name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LApp (project sg out) (project sa out)), ck := (project sa ck)})))
      | LexiconLse::Lse.LArg h t => (let sh : Sr := (substIn f st h name replacement avoid); (let stl : Sr := (substIn f (project sh ck) t name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LArg (project sh out) (project stl out)), ck := (project stl ck)})))
      | LexiconLse::Lse.LPi bs body => (let sb : Sb := (substChain f st bs name replacement avoid (nil (Bytes * Bytes)) false); (let sbody : Sr := (substBody f sb body name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LPi (project sb chain) (project sbody out)), ck := (project sbody ck)})))
      | LexiconLse::Lse.LLam bs body => (let sb : Sb := (substChain f st bs name replacement avoid (nil (Bytes * Bytes)) false); (let sbody : Sr := (substBody f sb body name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LLam (project sb chain) (project sbody out)), ck := (project sbody ck)})))
      | LexiconLse::Lse.LLet bound t v body => (let stt : Sr := (substIn f st t name replacement avoid); (let sv : Sr := (substIn f (project stt ck) v name replacement avoid); (if (prim equal bound name : Bool) then (record Sr {out := (ctor LexiconLse::Lse.LLet bound (project stt out) (project sv out) body), ck := (project sv ck)}) else (if (boundIn avoid bound) then (let named : (Bytes * Ck) := (freshLocal (project sv ck) bound); (let renamed : Sr := (substIn f (second named) body bound (ctor LexiconLse::Lse.LLocal (first named)) (nil Bytes)); (let sbody : Sr := (substIn f (project renamed ck) (project renamed out) name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LLet (first named) (project stt out) (project sv out) (project sbody out)), ck := (project sbody ck)})))) else (let sbody : Sr := (substIn f (project sv ck) body name replacement avoid); (record Sr {out := (ctor LexiconLse::Lse.LLet bound (project stt out) (project sv out) (project sbody out)), ck := (project sbody ck)}))))))
      | LexiconLse::Lse.LBinder m n t r => (record Sr {out := (ctor LexiconLse::Lse.LBinder m n t r), ck := st})
      | LexiconLse::Lse.LSortProp => (record Sr {out := (ctor LexiconLse::Lse.LSortProp), ck := st})
      | LexiconLse::Lse.LSortType u => (record Sr {out := (ctor LexiconLse::Lse.LSortType u), ck := st})

definition{

      | LexiconLse::Lse.LSortProp => (urOk st)

definition{
  name: requireSort,
  parameters: [fuel : Fuel, st : Ck, expr : LexiconLse::Lse, ty : LexiconLse::Lse],
  result: Ir,
  body: (let zonked : Sr := (zonkSt fuel st ty); (let z : LexiconLse::Lse := (project zonked out); (let st2 : Ck := (project zonked ck); (let ke : Nat := (kindOf expr); (if (if (ke =? 8) then true else (ke =? 6)) then (let txtExpr : Bytes := (shown expr); (irFail st2 (cat txtExpr (lit " is a term, not a type")))) else (if ((kindOf z) <? 2) then (irOk z st2) else (match (metaOf z)
    | Option.some m => (irOk z st2)
    | Option.none => (if (mayUnfold fuel st2 z) then (irOk z st2) else (let txtExpr : Bytes := (shown expr); (let txtZ : Bytes := (shown z); (irFail st2 (cat (cat (cat txtExpr (lit " is not a type: its type is ")) txtZ) 0x)))))))))))),
  axioms: [Quot.sound, propext],

definition{
  name: infer,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match expr
      | LexiconLse::Lse.LSortProp => (irOk (ctor LexiconLse::Lse.LSortType (ctor LexiconLse::Universe.UNum 1)) st)
      | LexiconLse::Lse.LSortType u => (irOk (ctor LexiconLse::Lse.LSortType (ctor LexiconLse::Universe.USucc u)) st)
      | LexiconLse::Lse.LConst p e us => (let found : (Nat * Bool) := (lookupRow st p e); (let k : Nat := (first found); (if (k =? 0) then (irFail st (cat (cat (cat (cat (lit "`") p) (lit "::")) e) (lit "` does not resolve"))) else (if (k =? 1) then (irMeta st) else (if (k =? 2) then (irFail st (cat (cat (cat (cat (lit "`") p) (lit "::")) e) (lit "` has no signature and cannot appear in an expression"))) else (irOk (signatureOf st p e) st))))))
      | LexiconLse::Lse.LLocal name => (if (isMeta name) then (irMeta st) else (match (scopeFind scope name)
        | Option.some ty => (irOk ty st)
        | Option.none => (irFail st (cat (cat (lit "local `") name) (lit "` is not in scope")))))
      | LexiconLse::Lse.LNat digits => (let made : Sr := (literalMeta st); (irOk (project made out) (project made ck)))
      | LexiconLse::Lse.LApp g args => (let kg : Nat := (kindOf g); (if (if (kg <? 2) then true else (kg =? 8)) then (let txtExpr : Bytes := (shown expr); (let txtG : Bytes := (shown g); (irFail st (cat (cat (cat txtExpr (lit " applies ")) txtG) (lit ", which is not a function"))))) else (let fnTy : Ir := (infer f st g scope); (match (project fnTy error)
        | Option.some message => fnTy
        | Option.none => (let items : (List LexiconLse::Lse) := (chainItems f args); (inferApp f (project fnTy ck) expr g items (prim length items : Nat) (project fnTy ty) 0 scope))))))
      | LexiconLse::Lse.LPi bs body => (inferBinding f st bs body scope false bs)
      | LexiconLse::Lse.LLam bs body => (inferBinding f st bs body scope true bs)
      | LexiconLse::Lse.LLet n t v body => (let declared : Ir := (expectType f st t scope); (match (project declared error)
        | Option.some message => declared
        | Option.none => (let bound : Ir := (infer f (project declared ck) v scope); (match (project bound error)
          | Option.some message => bound
          | Option.none => (let agreed : Ur := (unify f (project bound ck) t (project bound ty)); (match (project agreed error)
            | Option.some message => (irFail (project agreed ck) message)
            | Option.none => (infer f (project agreed ck) body (cons (n, t) scope))))))))
      | LexiconLse::Lse.LNil => (irOk (ctor LexiconLse::Lse.LSortProp) st)
      | LexiconLse::Lse.LArg h t => (irOk (ctor LexiconLse::Lse.LSortProp) st)
      | LexiconLse::Lse.LBinder m n t r => (irOk (ctor LexiconLse::Lse.LSortProp) st))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferBinding,
  parameters: [
    fuel : Fuel,
    st : Ck,
    chain : LexiconLse::Lse,
    body : LexiconLse::Lse,
    scope : (List (Bytes * LexiconLse::Lse)),
    lam : Bool,
    binders : LexiconLse::Lse
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match chain
      | LexiconLse::Lse.LBinder m n t r => (let typedBinder : Ir := (expectType f st t scope); (match (project typedBinder error)
        | Option.some message => typedBinder
        | Option.none => (let next : Ir := (inferBinding f (project typedBinder ck) r body (cons (n, t) scope) lam binders); (record Ir {ty := (project next ty), error := (project next error), ck := (project next ck)}))))
      | LexiconLse::Lse.LNil => (if lam then (let inner : Ir := (infer f st body scope); (match (project inner error)
        | Option.some message => inner
        | Option.none => (irOk (ctor LexiconLse::Lse.LPi binders (project inner ty)) (project inner ck)))) else (expectType f st body scope))
      | LexiconLse::Lse.LSortProp => (expectType f st body scope)
      | LexiconLse::Lse.LSortType u => (expectType f st body scope)
      | LexiconLse::Lse.LConst p e us => (expectType f st body scope)
      | LexiconLse::Lse.LLocal name => (expectType f st body scope)
      | LexiconLse::Lse.LApp g args => (expectType f st body scope)
      | LexiconLse::Lse.LPi bs inner => (expectType f st body scope)
      | LexiconLse::Lse.LLam bs inner => (expectType f st body scope)
      | LexiconLse::Lse.LLet n t v inner => (expectType f st body scope)
      | LexiconLse::Lse.LNat digits => (expectType f st body scope)
      | LexiconLse::Lse.LArg h t => (expectType f st body scope))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: expectType,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (let inferred : Ir := (infer f st expr scope); (match (project inferred error)
      | Option.some message => inferred
      | Option.none => (requireSort f (project inferred ck) expr (project inferred ty))))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferRest,
  parameters: [
    fuel : Fuel,
    st : Ck,
    remaining : (List LexiconLse::Lse),
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match remaining
      | List.nil => (irMeta st)
      | List.cons argument more => (let seen : Ir := (infer f st argument scope); (match (project seen error)
        | Option.some message => seen
        | Option.none => (let next : Ir := (inferRest f (project seen ck) more scope); (record Ir {ty := (project next ty), error := (project next error), ck := (project next ck)})))))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferApp,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    g : LexiconLse::Lse,
    remaining : (List LexiconLse::Lse),
    total : Nat,
    fty : LexiconLse::Lse,
    supplied : Nat,
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match remaining
      | List.nil => (irOk fty st)
      | List.cons headArg rest => (let zonked : Sr := (zonkSt f st fty); (let z : LexiconLse::Lse := (project zonked out); (let st2 : Ck := (project zonked ck); (match z
        | LexiconLse::Lse.LPi bs body => (inferBinders f st2 expr g (binderItems f bs) body remaining total supplied scope)
        | LexiconLse::Lse.LSortProp => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LSortType u => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LConst p e us => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LLocal name => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LApp h args => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LLam bs body => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LLet n t v body => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LNat digits => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LNil => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LArg h t => (inferStuck f st2 expr g remaining total supplied z scope)
        | LexiconLse::Lse.LBinder m n t r => (inferStuck f st2 expr g remaining total supplied z scope))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferStuck,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    g : LexiconLse::Lse,
    remaining : (List LexiconLse::Lse),
    total : Nat,
    supplied : Nat,
    z : LexiconLse::Lse,
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match (metaOf z)
      | Option.some m => (inferRest f st remaining scope)
      | Option.none => (if (mayUnfold f st z) then (inferRest f st remaining scope) else (let txtExpr : Bytes := (shown expr); (let txtG : Bytes := (shown g); (let txtZ : Bytes := (shown z); (irFail st (cat (cat (cat (cat (cat (cat (cat (cat (cat txtExpr (lit " supplies ")) (LexLeanStd.Numeral::decimal total)) (lit " explicit argument")) (plural total)) (lit " but ")) txtG) (lit " accepts ")) (LexLeanStd.Numeral::decimal supplied)) (cat (cat (lit ": after them its type is ") txtZ) (lit ", not a function")))))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferBinders,
  parameters: [
    fuel : Fuel,
    st : Ck,
    expr : LexiconLse::Lse,
    g : LexiconLse::Lse,
    binders : (List Bd),
    body : LexiconLse::Lse,
    remaining : (List LexiconLse::Lse),
    total : Nat,
    supplied : Nat,
    scope : (List (Bytes * LexiconLse::Lse))
  ],
  result: Ir,
  body: (match fuel
    | Fuel.FZ => (irFail (ckExhaust st) 0x)
    | Fuel.FS f => (match binders
      | List.nil => (inferApp f st expr g remaining total body supplied scope)
      | List.cons binder rest => (if ((project binder mode) =? 0) then (match remaining
        | List.nil => (irOk (ctor LexiconLse::Lse.LPi (chainOf binders) body) st)
        | List.cons argument more => (let actual : Ir := (infer f st argument scope); (match (project actual error)
          | Option.some message => actual
          | Option.none => (let agreed : Ur := (unify f (project actual ck) (project binder ty) (project actual ty)); (match (project agreed error)
            | Option.some message => (irFail (project agreed ck) message)
            | Option.none => (inferInstantiate f (project agreed ck) expr g binder argument rest body more total (supplied + 1) scope)))))) else (let made : Sr := (freshMeta st); (inferInstantiate f (project made ck) expr g binder (project made out) rest body remaining total supplied scope))))),
  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: inferInstantiate,

    fuel : Fuel,
    st : Ck,

  result: Ir,
  body: (match fuel

  axioms: [Quot.sound, propext],
  executable: true,
  mutual: Infer,
  recursive_argument: fuel
}

definition{
  name: literalTypes,

  result: ((List (Option LexiconLse::Lse)) * Ck),
  body: (match fuel
    | Fuel.FZ => ((nil (Option LexiconLse::Lse)), (ckExhaust st))
    | Fuel.FS f => (match metas
      | List.nil => ((nil (Option LexiconLse::Lse)), st)
      | List.cons metaName more => (let solved : Sr := (zonkSt f st (ctor LexiconLse::Lse.LLocal metaName)); (let pending : (Bool * Ck) := (hasUnsolved f (project solved ck) (project solved out)); (let others : ((List (Option LexiconLse::Lse)) * Ck) := (literalTypes f (second pending) more); ((cons (if (flagOf pending) then (ctor Option.none [LexiconLse::Lse]) else (ctor Option.some [LexiconLse::Lse] (project solved out))) (first others)), (second others))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: checkValue,
  parameters: [
    value : LexiconLse::Lse,
    signature : (Option LexiconLse::Lse),
    rows : (List CRow),
    fallback : Nat,
    depth : Nat
  ],

  body: (let fuel : Fuel := (mkFuel depth); (let inferred : Ir := (infer fuel (start rows fallback) value (nil (Bytes * LexiconLse::Lse))); (match signature
    | Option.none => (outcome fuel inferred)
    | Option.some sig => (if (if (project (project inferred ck) exhausted) then true else (match (project inferred error)
      | Option.some message => true
      | Option.none => false)) then (outcome fuel inferred) else (let agreed : Ur := (unify fuel (project inferred ck) sig (project inferred ty)); (match (project agreed error)
      | Option.some message => (let txtValue : Bytes := (shown value); (let txtSig : Bytes := (shown sig); (outcome fuel (irFail (project agreed ck) (cat (cat (cat (cat (cat (lit "defined value ") txtValue) (lit " does not match the signature ")) txtSig) (lit ": ")) message)))))
      | Option.none => (outcome fuel (irOk (project inferred ty) (project agreed ck))))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}
