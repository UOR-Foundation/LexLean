\begin{lexlean}{Budget}
\useglossary{lexlean.std.nat@1.2.0}
\useglossary{reasoning.titles@1.0.0}
\title{Spending budgets}
\begin{semanticmodule}
definition{
  name: Spent,
  type_parameters: [T],
  parameters: [s : ('T * Nat), t : ('T * Nat)],
  result: Prop,
  body: (((first t) = (first s)) /\ ((second t) < (second s)))
}

definition{
  name: Remaining,
  type_parameters: [T],
  parameters: [s : ('T * Nat)],
  result: Nat,
  body: (second s)
}

logic{
  name: Countdown,
  type_parameters: [T],
  ranking: Remaining,
  relation: Spent,
  state: {name: s, type: ('T * Nat), next: t}
}

theorem{
  name: tick_sound,
  type_parameters: [T],
  parameters: [s : ('T * Nat)],
  statement: (((2 <? (second s)) = true) -> (Spent ['T] s ((first s), (prim subtract (second s) 1 : Nat)))),
  proof: linear_arithmetic{definitions: [Spent]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: tick_progress,
  type_parameters: [T],
  parameters: [s : ('T * Nat)],
  statement: (((2 <? (second s)) = true) -> ((Remaining ['T] ((first s), (prim subtract (second s) 1 : Nat))) < (Remaining ['T] s))),
  proof: linear_arithmetic{definitions: [Remaining]},
  axioms: [Quot.sound, propext]
}

inference_rule{
  name: Tick,
  type_parameters: [T],
  axioms: [Quot.sound, propext],
  conclusion: ((first s), (prim subtract (second s) 1 : Nat)),
  executable: true,
  guard: (2 <? (second s)),
  logic: {member: Countdown, type_arguments: ['T]},
  progress: tick_progress,
  soundness: tick_sound
}

theorem{
  name: burn_sound,
  type_parameters: [T],
  parameters: [s : ('T * Nat), k : Nat],
  statement: ((((k <=? (second s)) && (0 <? k)) = true) -> (Spent ['T] s ((first s), (prim subtract (second s) k : Nat)))),
  proof: linear_arithmetic{definitions: [Spent]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: burn_progress,
  type_parameters: [T],
  parameters: [s : ('T * Nat), k : Nat],
  statement: ((((k <=? (second s)) && (0 <? k)) = true) -> ((Remaining ['T] ((first s), (prim subtract (second s) k : Nat))) < (Remaining ['T] s))),
  proof: linear_arithmetic{definitions: [Remaining]},
  axioms: [Quot.sound, propext]
}

inference_rule{
  name: Burn,
  type_parameters: [T],
  axioms: [Quot.sound, propext],
  binding: {name: k, type: Nat, candidates: (cons 2 (cons 1 (nil Nat)))},
  conclusion: ((first s), (prim subtract (second s) k : Nat)),
  executable: true,
  guard: ((k <=? (second s)) && (0 <? k)),
  logic: {member: Countdown, type_arguments: ['T]},
  progress: burn_progress,
  soundness: burn_sound
}

definition{
  name: Empty,
  type_parameters: [T],
  parameters: [x : 'T, r : Nat],
  result: Prop,
  body: (r = 0)
}

definition{
  name: emptyCheck,
  type_parameters: [T],
  parameters: [x : 'T, r : Nat],
  result: Bool,
  body: (r =? 0),
  executable: true
}

theorem{
  name: empty_sound,
  type_parameters: [T],
  parameters: [x : 'T, r : Nat],
  statement: (((emptyCheck ['T] x r) = true) -> (Empty ['T] x r)),
  proof: linear_arithmetic{definitions: [Empty, emptyCheck]},
  axioms: [propext]
}

verifier{
  name: Drained,
  type_parameters: [T],
  axioms: [propext],
  candidate: {name: r, type: Nat},
  check: emptyCheck,
  sound: empty_sound,
  specification: Empty,
  subject: {name: x, type: 'T}
}

theorem{
  name: allowance_bound,
  type_parameters: [T],
  parameters: [x : ('T * Nat)],
  statement: ((Remaining ['T] ((first x), ((second x) + 3))) < ((second x) + 4)),
  proof: linear_arithmetic{definitions: [Remaining]},
  axioms: [Quot.sound, propext]
}

reasoner{
  name: Spend,
  type_parameters: [T],
  axioms: [Quot.sound, propext],
  answer: {name: account, type: Nat, value: (ctor Option.some [Nat] (second account))},
  claims: [terminates{theorem: allowance_bound}],
  executable: true,
  logic: {member: Countdown, type_arguments: ['T]},
  observation: {name: x, type: ('T * Nat)},
  observe: ((first x), ((second x) + 3)),
  rules: [{member: Tick, type_arguments: ['T]}, {member: Burn, type_arguments: ['T]}],
  strategy: forward{fuel: ((second x) + 4)},
  verifier: {member: Drained, type_arguments: [('T * Nat)]}
}

reasoner{
  name: Refund,
  axioms: [Quot.sound, propext],
  answer: {
    name: account,
    type: Nat,
    value: (if ((first account) <? 5) then (ctor Option.some [Nat] (second account)) else (ctor Option.none [Nat]))
  },
  executable: true,
  logic: {member: Countdown, type_arguments: [Nat]},
  observation: {name: x, type: Nat},
  observe: (x, (x + 3)),
  rules: [{member: Tick, type_arguments: [Nat]}, {member: Burn, type_arguments: [Nat]}],
  strategy: forward{fuel: (x + 4)},
  verifier: {member: Drained, type_arguments: [Nat]}
}
\end{semanticmodule}
\end{lexlean}
