definition{
  name: takeN,
  parameters: [cells : RustMemory::Cells, count : Nat],
  result: RustMemory::Cells,
  body: (match count
    | Nat.zero => (ctor RustMemory::Cells.CNil)
    | Nat.succ less => (match cells
      | RustMemory::Cells.CNil => (ctor RustMemory::Cells.CNil)
      | RustMemory::Cells.CCons head tail => (ctor RustMemory::Cells.CCons head (takeN tail less)))),
  recursive_argument: count
}

definition{
  name: snoc,
  parameters: [cells : RustMemory::Cells, item : TargetSyntax::Value],
  result: RustMemory::Cells,
  body: (match cells
    | RustMemory::Cells.CNil => (ctor RustMemory::Cells.CCons item (ctor RustMemory::Cells.CNil))
    | RustMemory::Cells.CCons head tail => (ctor RustMemory::Cells.CCons head (snoc tail item))),
  recursive_argument: cells
}

definition{
  name: absRun,
  parameters: [cells : RustMemory::Cells, length : Nat, onto : (List TargetSyntax::Value)],
  result: (List TargetSyntax::Value),
  body: (match length
    | Nat.zero => onto
    | Nat.succ less => (match cells
      | RustMemory::Cells.CNil => onto
      | RustMemory::Cells.CCons head tail => (absRun tail less (cons head onto)))),
  recursive_argument: length
}

definition{
  name: unconsAbs,
  parameters: [
    found : (Option TargetSyntax::Value),
    rest : (List TargetSyntax::Value),
    otherwise : (List TargetSyntax::Value)
  ],
  result: (List TargetSyntax::Value),
  body: (match found | Option.none => otherwise | Option.some head => (cons head rest))
}

theorem{
  name: run_cons_abs,
  parameters: [
    cells : RustMemory::Cells,
    item : TargetSyntax::Value,
    length : Nat,
    onto : (List TargetSyntax::Value)
  ],
  statement: ((absRun (snoc (takeN cells length) item) (length + 1) onto) = (cons item (absRun cells length onto))),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [absRun, snoc, takeN]}, binders: [], constructor: zero},
      {
        proof: cases{
          branches: [
            {proof: simplify{definitions: [absRun, snoc, takeN]}, binders: [], constructor: CNil},
            {
              proof: simplify{definitions: [absRun, hypothesis, snoc, takeN]},
              binders: [head, tail],
              constructor: CCons
            }
          ],
          scrutinee: cells
        },
        binders: [less, hypothesis],
        constructor: succ
      }
    ],
    generalizing: [cells, onto],
    scrutinee: length
  },
  axioms: [propext]
}

definition{
  name: cellsAt,
  parameters: [store : Store, address : Nat],
  result: RustMemory::Cells,
  body: (match store
    | Store.SEmpty => (ctor RustMemory::Cells.CNil)
    | Store.SCons c rest => (match address
      | Nat.zero => c
      | Nat.succ lower => (cellsAt rest lower))),
  recursive_argument: store
}

definition{
  name: runWalk,
  parameters: [store : Store, address : Nat, length : Nat, onto : (List TargetSyntax::Value)],
  result: (List TargetSyntax::Value),
  body: (match store
    | Store.SEmpty => (absRun (ctor RustMemory::Cells.CNil) length onto)
    | Store.SCons c rest => (match address
      | Nat.zero => (absRun c length onto)
      | Nat.succ lower => (runWalk rest lower length onto))),
  recursive_argument: store
}

definition{
  name: headAt,
  parameters: [store : Store, address : Nat, slot : Nat],
  result: (Option TargetSyntax::Value),
  body: (match store
    | Store.SEmpty => (ctor Option.none [TargetSyntax::Value])
    | Store.SCons c rest => (match address
      | Nat.zero => (RustMemory::get c slot)
      | Nat.succ lower => (headAt rest lower slot))),
  recursive_argument: store
}

definition{
  name: consRun,
  parameters: [store : Store, address : Nat, length : Nat, item : TargetSyntax::Value],
  result: Store,
  body: (match address
    | Nat.zero => (match store
      | Store.SEmpty => (ctor Store.SCons (snoc (takeN (ctor RustMemory::Cells.CNil) length) item) (ctor Store.SEmpty))
      | Store.SCons c rest => (ctor Store.SCons (snoc (takeN c length) item) rest))
    | Nat.succ lower => (match store
      | Store.SEmpty => (ctor Store.SCons (ctor RustMemory::Cells.CNil) (consRun (ctor Store.SEmpty) lower length item))
      | Store.SCons c rest => (ctor Store.SCons c (consRun rest lower length item)))),
  recursive_argument: address
}

definition{
  name: gapOf,
  parameters: [left : Nat, right : Nat],
  result: Gap,
  body: (match left
    | Nat.zero => (match right
      | Nat.zero => (ctor Gap.same)
      | Nat.succ more => (ctor Gap.over more))
    | Nat.succ less => (match right
      | Nat.zero => (ctor Gap.under less)
      | Nat.succ more => (gapOf less more))),
  recursive_argument: left
}

definition{
  name: gapPick,
  parameters: [
    gap : Gap,
    whenSame : (List TargetSyntax::Value),
    otherwise : (List TargetSyntax::Value)
  ],
  result: (List TargetSyntax::Value),
  body: (match gap
    | Gap.same => whenSame
    | Gap.under distance => otherwise
    | Gap.over distance => otherwise)
}

theorem{
  name: walk_cons,
  parameters: [
    store : Store,
    address : Nat,
    length : Nat,
    item : TargetSyntax::Value,
    onto : (List TargetSyntax::Value)
  ],
  statement: ((runWalk (consRun store address length item) address (length + 1) onto) = (cons item (runWalk store address length onto))),
  proof: induction{
    branches: [
      {
        proof: induction{
          branches: [
            {
              proof: apply{
                arguments: [(ctor RustMemory::Cells.CNil), item, length, onto],
                theorem: run_cons_abs
              },
              binders: [],
              constructor: zero
            },
            {
              proof: simplify{definitions: [consRun, hypothesis, runWalk]},
              binders: [lower, hypothesis],
              constructor: succ
            }
          ],
          scrutinee: address
        },
        binders: [],
        constructor: SEmpty
      },
      {
        proof: cases{
          branches: [
            {
              proof: apply{arguments: [c, item, length, onto], theorem: run_cons_abs},
              binders: [],
              constructor: zero
            },
            {
              proof: simplify{definitions: [consRun, hypothesis, runWalk]},
              binders: [lower],
              constructor: succ
            }
          ],
          scrutinee: address
        },
        binders: [c, rest, hypothesis],
        constructor: SCons
      }
    ],
    generalizing: [address],
    scrutinee: store
  },
  axioms: [propext]
}

definition{
  name: absTerm,
  parameters: [store : Store, base : Terminal],
  result: (List TargetSyntax::Value),
  body: (match base
    | Terminal.TNil => (nil TargetSyntax::Value)
    | Terminal.TRun address more => (runWalk store address (more + 1) (nil TargetSyntax::Value)))
}

definition{
  name: absHandle,
  parameters: [store : Store, handle : Handle],
  result: (List TargetSyntax::Value),
  body: (listAppend (project handle spine) (absTerm store (project handle base)))
}

definition{
  name: consWorld,
  parameters: [
    store : Store,
    fresh : Nat,
    unique : RustMemory::Flag,
    item : TargetSyntax::Value,
    handle : Handle
  ],
  result: World,
  body: (match (project handle spine)
    | List.nil => (match (project handle base)
      | Terminal.TNil => (record World {store := (consRun store fresh 0 item), handle := (record Handle {spine := (nil TargetSyntax::Value), base := (ctor Terminal.TRun fresh 0)})})
      | Terminal.TRun address more => (match unique
        | RustMemory::Flag.yes => (record World {store := (consRun store address (more + 1) item), handle := (record Handle {spine := (nil TargetSyntax::Value), base := (ctor Terminal.TRun address (more + 1))})})
        | RustMemory::Flag.no => (record World {store := store, handle := (record Handle {spine := (cons item (nil TargetSyntax::Value)), base := (ctor Terminal.TRun address more)})})))
    | List.cons front rest => (record World {store := store, handle := (record Handle {spine := (cons item (cons front rest)), base := (project handle base)})}))
}

definition{
  name: unconsHandle,
  parameters: [store : Store, handle : Handle],
  result: (Option Taken),
  body: (match (project handle spine)
    | List.nil => (match (project handle base)
      | Terminal.TNil => (ctor Option.none [Taken])
      | Terminal.TRun address more => (match (headAt store address more)
        | Option.none => (ctor Option.none [Taken])
        | Option.some found => (ctor Option.some [Taken] (record Taken {head := found, rest := (record Handle {spine := (nil TargetSyntax::Value), base := (match more
          | Nat.zero => (ctor Terminal.TNil)
          | Nat.succ less => (ctor Terminal.TRun address less))})}))))
    | List.cons front rest => (ctor Option.some [Taken] (record Taken {head := front, rest := (record Handle {spine := rest, base := (project handle base)})})))
}

definition{
  name: takenFrom,
  parameters: [found : (Option TargetSyntax::Value), rest : Handle],
  result: (Option Taken),
  body: (match found
    | Option.none => (ctor Option.none [Taken])
    | Option.some got => (ctor Option.some [Taken] (record Taken {head := got, rest := rest})))
}

  name: releaseLoop,
  parameters: [fuel : Nat, cells : (List RustMemory::Flag)],
  result: Out,
  body: (match cells
    | List.nil => (ctor Out.done)
    | List.cons lead rest => (match fuel
      | Nat.zero => (ctor Out.spent)
      | Nat.succ less => (match lead
        | RustMemory::Flag.yes => (releaseLoop less rest)
        | RustMemory::Flag.no => (ctor Out.stopped)))),
  recursive_argument: cells
}
