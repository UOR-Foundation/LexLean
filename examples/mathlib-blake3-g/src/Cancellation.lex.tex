\begin{section}{addition}
\heading{Cancellation}
  \begin{theorem}{add-sub-cancel}
  \exactaxioms{propext}
  For every \(bitVec(32)\) \(a\) and \(bitVec(32)\) \(b\), \(hSubHSub(hAddHAdd(a, b), b) = a\).
  \begin{proof}
  Close the goal with \(lexLeanSpecBitVec32BitVecAddSubCancel(a, b)\).
  \end{proof}
  \end{theorem}

  \begin{theorem}{add-sub-cancel-group}
  \exactaxioms{Classical.choice;Quot.sound;propext}
  For every \(bitVec(32)\) \(a\) and \(bitVec(32)\) \(b\), \(hSubHSub(hAddHAdd(a, b), b) = a\).
  \begin{proof}
  Close the goal with \(lexLeanSpecBitVec32AddSubCancelRight(a, b)\).
  \end{proof}
  \end{theorem}

  \begin{theorem}{add-sub-cancel-u32}
  \exactaxioms{propext}
  For every \(uInt32\) \(a\) and \(uInt32\) \(b\), \(hSubHSub(hAddHAdd(a, b), b) = a\).
  \begin{proof}
  Close the goal with \(lexLeanSpecU32UInt32AddSubCancel(a, b)\).
  \end{proof}
  \end{theorem}

  \begin{lemma}{add-sub-sub-cancel}
  \exactaxioms{Classical.choice;Quot.sound;propext}
  For every \(bitVec(32)\) \(p\) and \(bitVec(32)\) \(q\) and \(bitVec(32)\) \(r\), \(hSubHSub(hSubHSub(hAddHAdd(hAddHAdd(p, q), r), q), r) = p\).
  \begin{proof}
  \begin{rewrite}{goal}
  \forward{lexLeanSpecBitVec32AddRightComm(p, q, r)}
  \forward{lexLeanSpecBitVec32AddSubCancelRight(hAddHAdd(p, r), q)}
  \forward{lexLeanSpecBitVec32AddSubCancelRight(p, r)}
  \end{rewrite}
  \end{proof}
  \end{lemma}
