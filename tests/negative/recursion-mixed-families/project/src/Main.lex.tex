\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
inductive{
  name: Tree,
  type_parameters: [],
  parameters: [],
  constructors: [{name: leaf, fields: []}, {name: node, fields: [Tree, Nat, Tree]}]
}

definition{
  name: treeDepth,
  parameters: [tree : Tree],
  result: Nat,
  body: (match tree | Tree.leaf => 0 | Tree.node left value right => (countDown value)),
  mutual: Mixed,
  recursive_argument: tree
}

definition{
  name: countDown,
  parameters: [number : Nat],
  result: Nat,
  body: (match number | Nat.zero => 0 | Nat.succ previous => (countDown previous)),
  mutual: Mixed,
  recursive_argument: number
}
\end{semanticmodule}
\end{lexlean}
