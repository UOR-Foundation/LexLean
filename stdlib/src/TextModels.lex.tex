definition{
  name: withIndex,
  parameters: [vocabulary : (List String), start : Nat],
  result: (List (Nat * String)),
  body: (match vocabulary
    | List.nil => (nil (Nat * String))
    | List.cons head tail => (cons (start, head) (withIndex tail (start + 1)))),
  executable: true,
  recursive_argument: vocabulary
}

definition{
  name: takeInts,
  parameters: [count : Nat, values : (List Int)],
  result: (List Int),
  body: (match count
    | Nat.zero => (nil Int)
    | Nat.succ previous => (match values
      | List.nil => (nil Int)
      | List.cons head tail => (cons head (takeInts previous tail)))),
  executable: true,
  recursive_argument: count
}

definition{
  name: zerosInts,
  parameters: [count : Nat],
  result: (List Int),
  body: (match count
    | Nat.zero => (nil Int)
    | Nat.succ previous => (cons (0 : Int) (zerosInts previous))),
  executable: true,
  recursive_argument: count
}

definition{
  name: fitWidth,
  parameters: [width : Nat, values : (List Int)],
  result: (List Int),
  body: (prim append (takeInts width values) (zerosInts (prim subtract width (prim length values : Nat) : Nat)) : (List Int)),
  executable: true
}

definition{
  name: fits,
  parameters: [capacity : Nat, context : (List UInt32)],
  result: Bool,
  body: ((prim length context : Nat) <=? capacity),
  executable: true
}

definition{
  name: dropTokens,
  parameters: [count : Nat, context : (List UInt32)],
  result: (List UInt32),
  body: (match count
    | Nat.zero => context
    | Nat.succ previous => (match context
      | List.nil => (nil UInt32)
      | List.cons head tail => (dropTokens previous tail))),
  executable: true,
  recursive_argument: count
}

definition{

}

definition{
  name: genGo,
  parameters: [score : (Fn ((List UInt32)) (List Int)), choose : (Fn ((List Int) UInt32) ((Option Nat) * UInt32)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, context : (List UInt32), output : (List UInt32), seed : UInt32],
  result: (((List UInt32) * Nat) * UInt32),
  body: (match fuel
    | Nat.zero => ((output, 1), seed)
    | Nat.succ previous => (let step : ((Option Nat) * UInt32) := (apply choose (apply score context) seed); (match (first step)
      | Option.none => ((output, 2), (second step))
      | Option.some index => (if (index <? size) then (match (tokenId index)
        | Option.none => ((output, 2), (second step))
        | Option.some token => (if (prim equal token stop : Bool) then ((output, 0), (second step)) else (match (pushWith keep capacity context token)
          | Option.none => ((output, 3), (second step))
          | Option.some grown => (genGo score choose size capacity keep stop previous grown (prim append output (cons token (nil UInt32)) : (List UInt32)) (second step))))) else ((output, 2), (second step)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: fuel
}

definition{
  name: generateWith,
  parameters: [score : (Fn ((List UInt32)) (List Int)), choose : (Fn ((List Int) UInt32) ((Option Nat) * UInt32)), size : Nat, capacity : Nat, keep : Bool, stop : UInt32, fuel : Nat, prompt : (List UInt32), seed : UInt32],
  result: (((List UInt32) * Nat) * UInt32),
  body: (match (admitWith keep capacity prompt)
    | Option.none => (((nil UInt32), 3), seed)
    | Option.some context => (genGo score choose size capacity keep stop fuel context (nil UInt32) seed)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: xorshift32,
  parameters: [value : UInt32],
  result: UInt32,
  body: (let stepOne : UInt32 := (match (prim shift_left value (13 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor value shifted : UInt32)
    | Option.none => value); (let stepTwo : UInt32 := (match (prim shift_right stepOne (17 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor stepOne shifted : UInt32)
    | Option.none => stepOne); (match (prim shift_left stepTwo (5 : UInt32) : (Option UInt32))
    | Option.some shifted => (prim bit_xor stepTwo shifted : UInt32)
    | Option.none => stepTwo))),
  axioms: [Quot.sound, propext],
  executable: true
}

  executable: true,

definition{
  name: pieceOf,
  parameters: [vocabulary : (List String), token : UInt32],
  result: (Option (List UInt8)),
  body: (if ((natOfToken token) <? (byteVocab)) then (match (prim checked_convert token : (Option UInt8))
    | Option.some byte => (ctor Option.some [(List UInt8)] (cons byte (nil UInt8)))
    | Option.none => (ctor Option.none [(List UInt8)])) else (match (prim index vocabulary (prim subtract (natOfToken token) (byteVocab) : Nat) : (Option String))
    | Option.some piece => (ctor Option.some [(List UInt8)] (pieceBytes piece))
    | Option.none => (ctor Option.none [(List UInt8)]))),
  axioms: [Quot.sound, propext],
  executable: true

definition{

  recursive_argument: prefix
