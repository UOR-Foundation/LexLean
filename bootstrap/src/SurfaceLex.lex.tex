definition{
  name: hexOf,
  parameters: [byte : UInt8],
  result: (Option Nat),
  body: (let value : Nat := (prim to_nat byte : Nat); (if (value <? 48) then (ctor Option.none [Nat]) else (if (value <=? 57) then (ctor Option.some [Nat] (prim subtract value 48 : Nat)) else (if (value <? 65) then (ctor Option.none [Nat]) else (if (value <=? 70) then (ctor Option.some [Nat] (prim subtract value 55 : Nat)) else (if (value <? 97) then (ctor Option.none [Nat]) else (if (value <=? 102) then (ctor Option.some [Nat] (prim subtract value 87 : Nat)) else (ctor Option.none [Nat])))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: hexAt,
  parameters: [subject : Bytes, position : Nat],
  result: (Option Nat),
  body: (match (LexLeanStd.Text::byteAt subject position)

    | Option.none => (ctor Option.none [Nat])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: fourHex,
  parameters: [subject : Bytes, position : Nat],
  result: (Option Nat),
  body: (match (hexAt subject position)
    | Option.some a => (match (hexAt subject (position + 1))
      | Option.some b => (match (hexAt subject (position + 2))
        | Option.some c => (match (hexAt subject (position + 3))
          | Option.some d => (ctor Option.some [Nat] ((prim multiply ((prim multiply ((prim multiply a 16 : Nat) + b) 16 : Nat) + c) 16 : Nat) + d))
          | Option.none => (ctor Option.none [Nat]))
        | Option.none => (ctor Option.none [Nat]))
      | Option.none => (ctor Option.none [Nat]))
    | Option.none => (ctor Option.none [Nat])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: utf8Of,
  parameters: [code : Nat],
  result: Bytes,
  body: (if (code <? 128) then (byteOf code) else (if (code <? 2048) then (cat (byteOf (192 + (prim quotient code 64 0 : Nat))) (byteOf (128 + (prim remainder code 64 0 : Nat)))) else (cat (byteOf (224 + (prim quotient code 4096 0 : Nat))) (cat (byteOf (128 + (prim remainder (prim quotient code 64 0 : Nat) 64 0 : Nat))) (byteOf (128 + (prim remainder code 64 0 : Nat))))))),
  executable: true
}

    | Option.some code => (if (55296 <=? code) then (if (code <=? 57343) then (halted state position (cat (cat (lit "`\\u") (hex4 code)) (lit "` is not a Unicode scalar value"))) else (taken state (position + 6) (utf8Of code))) else (taken state (position + 6) (utf8Of code)))

definition{
  name: escapeAt,
  parameters: [subject : Bytes, n : Nat, state : Scanned],
  result: Scanned,
  body: (let position : Nat := (project state pos); (match (LexLeanStd.Text::byteAt subject (position + 1))
    | Option.none => (record Scanned {pos := n, pieces := (project state pieces), finished := false, error := (ctor Option.none [Failure])})
    | Option.some escape => (if (prim equal escape (34 : UInt8) : Bool) then (taken state (position + 2) (lit "\"")) else (if (prim equal escape (92 : UInt8) : Bool) then (taken state (position + 2) (lit "\\")) else (if (prim equal escape (47 : UInt8) : Bool) then (taken state (position + 2) (lit "/")) else (if (prim equal escape (98 : UInt8) : Bool) then (taken state (position + 2) (byteOf 8)) else (if (prim equal escape (102 : UInt8) : Bool) then (taken state (position + 2) (byteOf 12)) else (if (prim equal escape (110 : UInt8) : Bool) then (taken state (position + 2) (byteOf 10)) else (if (prim equal escape (114 : UInt8) : Bool) then (taken state (position + 2) (byteOf 13)) else (if (prim equal escape (116 : UInt8) : Bool) then (taken state (position + 2) (byteOf 9)) else (if (prim equal escape (117 : UInt8) : Bool) then (unicodeEscape subject state) else (halted state position (cat (cat (lit "`\\") (LexLeanStd.Text::sliceOr subject (position + 1) (widthAt subject (position + 1)))) (lit "` is not an escape")))))))))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: scanStep,
  parameters: [subject : Bytes, n : Nat, state : Scanned],
  result: (Option Scanned),
  body: (match (project state error)
    | Option.some stopped => (ctor Option.none [Scanned])
    | Option.none => (if (project state finished) then (ctor Option.none [Scanned]) else (let position : Nat := (project state pos); (if (position <? n) then (if (byteIs subject position (34 : UInt8)) then (ctor Option.some [Scanned] (record Scanned {pos := (position + 1), pieces := (project state pieces), finished := true, error := (ctor Option.none [Failure])})) else (if (byteIs subject position (92 : UInt8)) then (ctor Option.some [Scanned] (escapeAt subject n state)) else (if ((byteNum subject position) <? 32) then (ctor Option.some [Scanned] (halted state position (lit "a control character must be escaped in a string"))) else (let stop : Nat := (rawEnd subject n position); (ctor Option.some [Scanned] (taken state stop (LexLeanStd.Text::sliceOr subject position (prim subtract stop position : Nat)))))))) else (ctor Option.none [Scanned]))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: quoted,
  parameters: [subject : Bytes, n : Nat, state : Lx],
  result: (Option Lx),
  body: (let start : Nat := (project state pos); (let scanned : Scanned := (readString subject n start); (match (project scanned error)
    | Option.some failure => (ctor Option.some [Lx] (record Lx {pos := start, tokens := (project state tokens), error := (ctor Option.some [Failure] failure)}))
    | Option.none => (if (project scanned finished) then (pushed state (ctor Token.StrToken (joined (project scanned pieces))) (project scanned pos)) else (ctor Option.some [Lx] (record Lx {pos := start, tokens := (project state tokens), error := (ctor Option.some [Failure] (failed start (lit "unterminated string")))})))))),
  axioms: [Quot.sound, propext],
  executable: true

  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: lexStep,
  parameters: [subject : Bytes, n : Nat, state : Lx],
  result: (Option Lx),
  body: (match (project state error)
    | Option.some stopped => (ctor Option.none [Lx])
    | Option.none => (let position : Nat := (project state pos); (if (position <? n) then (let white : Nat := (AxiomOutput::whiteLength subject position); (if (0 <? white) then (ctor Option.some [Lx] (record Lx {pos := (position + white), tokens := (project state tokens), error := (ctor Option.none [Failure])})) else (if (byteIs subject position (123 : UInt8)) then (pushed state (ctor Token.LBrace) (position + 1)) else (if (byteIs subject position (125 : UInt8)) then (pushed state (ctor Token.RBrace) (position + 1)) else (if (byteIs subject position (91 : UInt8)) then (pushed state (ctor Token.LBrack) (position + 1)) else (if (byteIs subject position (93 : UInt8)) then (pushed state (ctor Token.RBrack) (position + 1)) else (if (byteIs subject position (40 : UInt8)) then (pushed state (ctor Token.LParen) (position + 1)) else (if (byteIs subject position (41 : UInt8)) then (pushed state (ctor Token.RParen) (position + 1)) else (if (byteIs subject position (44 : UInt8)) then (pushed state (ctor Token.Comma) (position + 1)) else (if (byteIs subject position (59 : UInt8)) then (pushed state (ctor Token.Semi) (position + 1)) else (if (byteIs subject position (58 : UInt8)) then (colonAt subject state) else (if (byteIs subject position (34 : UInt8)) then (quoted subject n state) else (wordAt subject n state))))))))))))) else (ctor Option.none [Lx])))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: lexText,
  parameters: [text : Bytes],
  result: (Result (List Spanned) Failure),
  body: (let n : Nat := (prim length text : Nat); (let final : Lx := (first (prim iterate_until (fun [n, text] (state : Lx) => (lexStep text n state)) (n + 1) (record Lx {pos := 0, tokens := (nil Spanned), error := (ctor Option.none [Failure])}) : (Lx * Bool))); (match (project final error)
    | Option.some failure => (ctor Result.error [(List Spanned), Failure] failure)
    | Option.none => (ctor Result.ok [(List Spanned), Failure] (reverseSpans (project final tokens)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}

definition{
  name: escapePiece,
  parameters: [value : Nat],
  result: Bytes,
  body: (if (value =? 34) then (lit "\\\"") else (if (value =? 92) then (lit "\\\\") else (if (value =? 8) then (lit "\\b") else (if (value =? 12) then (lit "\\f") else (if (value =? 10) then (lit "\\n") else (if (value =? 13) then (lit "\\r") else (if (value =? 9) then (lit "\\t") else (cat (lit "\\u00") (hex2 value))))))))),
  executable: true
}

definition{
  name: quoteStep,
  parameters: [subject : Bytes, n : Nat, state : Quoting],
  result: (Option Quoting),
  body: (let position : Nat := (project state pos); (if (position <? n) then (let value : Nat := (byteNum subject position); (if (if (value <? 32) then true else (if (value =? 34) then true else (value =? 92))) then (ctor Option.some [Quoting] (record Quoting {pos := (position + 1), pieces := (cons (escapePiece value) (project state pieces))})) else (let stop : Nat := (rawEnd subject n position); (ctor Option.some [Quoting] (record Quoting {pos := stop, pieces := (cons (LexLeanStd.Text::sliceOr subject position (prim subtract stop position : Nat)) (project state pieces))}))))) else (ctor Option.none [Quoting]))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: quoteText,
  parameters: [text : Bytes],
  result: Bytes,
  body: (let n : Nat := (prim length text : Nat); (let final : Quoting := (first (prim iterate_until (fun [n, text] (state : Quoting) => (quoteStep text n state)) (n + 1) (record Quoting {pos := 0, pieces := (nil Bytes)}) : (Quoting * Bool))); (cat (cat (lit "\"") (joined (project final pieces))) (lit "\"")))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}
