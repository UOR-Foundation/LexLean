definition{
  name: checkDeclared,
  parameters: [machine : Machine, action : Action],
  result: (Option Rejection),
  body: (let kind : ActionKind := (project action kind); (if (performed kind) then (ctor Option.none [Rejection]) else (if (preparation kind) then (match (project action charge)
    | Charge.ChConstant cost => (if (0 <? cost) then (ctor Option.none [Rejection]) else (ctor Option.some [Rejection] (ctor Rejection.RjHiddenCost kind)))
    | Charge.ChFree => (match (project machine boundary)
      | Boundary.BdComplete => (ctor Option.some [Rejection] (ctor Rejection.RjUnboundPreparation kind))
      | Boundary.BdPreparedState => (if (hasKind (project machine prepared) kind) then (ctor Option.none [Rejection]) else (ctor Option.some [Rejection] (ctor Rejection.RjUnboundPreparation kind)))
      | Boundary.BdPreparedPlan => (if (hasKind (project machine prepared) kind) then (ctor Option.none [Rejection]) else (ctor Option.some [Rejection] (ctor Rejection.RjUnboundPreparation kind))))
    | Charge.ChSteps => (ctor Option.some [Rejection] (ctor Rejection.RjHiddenCost kind))
    | Charge.ChUndeclared => (ctor Option.some [Rejection] (ctor Rejection.RjHiddenCost kind))) else (ctor Option.some [Rejection] (ctor Rejection.RjUnrealizableAction kind))))),
  axioms: [Quot.sound, propext],
  executable: true
