\begin{lexlean}{Types}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
inductive{
  name: Shape,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: circle, fields: [Nat]},
    {name: rect, fields: [Nat, Nat]},
    {name: empty, fields: []}
  ]
}

inductive{
  name: Tree,
  type_parameters: [],
  parameters: [],
  constructors: [{name: leaf, fields: []}, {name: node, fields: [Tree, Nat, Tree]}]
}

structure{
  name: Point,
  type_parameters: [],
  parameters: [],
  fields: [{name: x, type: Nat}, {name: y, type: Nat}]
}

structure{
  name: Box,
  type_parameters: [Item],
  parameters: [],
  fields: [{name: content, type: 'Item}, {name: count, type: Nat}]
}

class{
  name: Weights,
  type_parameters: [],
  parameters: [],
  fields: [{name: base, type: Nat}, {name: scale, type: Nat}]
}

instance{
  name: defaultWeights,
  fields: [{field: base, value: 2}, {field: scale, value: 3}],
  arguments: [],
  class: Weights,
  priority: 1000
}

definition{
  name: area,
  parameters: [shape : Shape],
  result: Nat,
  body: (match shape
    | Shape.circle r => (prim multiply (prim multiply r r : Nat) 3 : Nat)
    | Shape.rect w h => (prim multiply w h : Nat)
    | Shape.empty => 0),
  executable: true
}

definition{
  name: treeSum,
  parameters: [tree : Tree],
  result: Nat,
  body: (match tree | Tree.leaf => 0 | Tree.node l v r => (((treeSum l) + v) + (treeSum r))),
  executable: true,
  recursive_argument: tree
}

definition{
  name: unbox,
  type_parameters: [Item],
  parameters: [box : (Box 'Item)],
  result: 'Item,
  body: (project box content),
  executable: true
}

definition{
  name: twice,
  parameters: [step : (Fn (Nat) Nat), value : Nat],
  result: Nat,
  body: (apply step (apply step value)),
  executable: true
}

definition{
  name: mapList,
  type_parameters: [A, B],
  parameters: [transform : (Fn ('A) 'B), values : (List 'A)],
  result: (List 'B),
  body: (match values
    | List.nil => (nil 'B)
    | List.cons head tail => (cons (apply transform head) (mapList ['A, 'B] transform tail))),
  executable: true,
  recursive_argument: values
}

definition{
  name: isEven,
  parameters: [n : Nat],
  result: Bool,
  body: (match n | Nat.zero => true | Nat.succ k => (isOdd k)),
  executable: true,
  mutual: Parity,
  recursive_argument: n
}

definition{
  name: isOdd,
  parameters: [n : Nat],
  result: Bool,
  body: (match n | Nat.zero => false | Nat.succ k => (isEven k)),
  executable: true,
  mutual: Parity,
  recursive_argument: n
}
\end{semanticmodule}
\end{lexlean}
