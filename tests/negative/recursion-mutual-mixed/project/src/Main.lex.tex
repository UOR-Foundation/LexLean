\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
theorem{
  name: down_decreases,
  parameters: [number : Nat],
  statement: (((number =? 0) = false) -> ((prim subtract number 1 : Nat) < number)),
  proof: linear_arithmetic{}
}

definition{
  name: upward,
  parameters: [number : Nat],
  result: Nat,
  body: (match number | Nat.zero => 0 | Nat.succ previous => (downward previous)),
  mutual: Mixed,
  recursive_argument: number
}

definition{
  name: downward,
  parameters: [number : Nat],
  result: Nat,
  body: (if (number =? 0) then 0 else (upward (prim subtract number 1 : Nat))),
  mutual: Mixed,
  termination: {evidence: [down_decreases], measure: number}
}
\end{semanticmodule}
\end{lexlean}
