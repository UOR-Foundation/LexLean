\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: treeSize,
  parameters: [tree : (Types::Tree Nat)],
  result: Nat,
  body: (match tree
    | Types::Tree.leaf => 0
    | Types::Tree.node left value right => (((treeSize left) + 1) + (treeSize right))),
  recursive_argument: tree
}

definition{
  name: mirror,
  parameters: [tree : (Types::Tree Nat)],
  result: (Types::Tree Nat),
  body: (match tree
    | Types::Tree.leaf => (ctor Types::Tree.leaf [Nat])
    | Types::Tree.node left value right => (ctor Types::Tree.node [Nat] (mirror right) value (mirror left))),
  recursive_argument: tree
}

definition{
  name: sampleTree,
  parameters: [],
  result: (Types::Tree Nat),
  body: (ctor Types::Tree.node [Nat] (ctor Types::Tree.node [Nat] (ctor Types::Tree.leaf [Nat]) 1 (ctor Types::Tree.leaf [Nat])) 2 (ctor Types::Tree.leaf [Nat]))
}

theorem{
  name: sample_tree_size,
  parameters: [],
  statement: ((treeSize (sampleTree)) = 2),
  proof: decide{}
}

theorem{
  name: mirror_involutive,
  parameters: [tree : (Types::Tree Nat)],
  statement: ((mirror (mirror tree)) = tree),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [mirror]}, binders: [], constructor: leaf},
      {
        proof: simplify{definitions: [leftHypothesis, mirror, rightHypothesis]},
        binders: [left, value, right, leftHypothesis, rightHypothesis],
        constructor: node
      }
    ],
    scrutinee: tree
  },
  axioms: [propext]
}

definition{
  name: roseLabel,
  parameters: [rose : (Types::Rose Nat)],
  result: Nat,
  body: (match rose | Types::Rose.node label children => label)
}

theorem{
  name: rose_label,
  parameters: [],
  statement: ((roseLabel (ctor Types::Rose.node [Nat] 7 (nil (Types::Rose Nat)))) = 7),
  proof: reflexivity{}
}

definition{
  name: isLiteral,
  parameters: [expression : Types::Expr],
  result: Bool,
  body: (match expression
    | Types::Expr.literal value => true
    | Types::Expr.plus left right => false
    | Types::Expr.block statements result => false)
}

theorem{
  name: block_is_not_literal,
  parameters: [],
  statement: ((isLiteral (ctor Types::Expr.block (cons (ctor Types::Stmt.assign "x" (ctor Types::Expr.literal 1)) (nil Types::Stmt)) (ctor Types::Expr.literal 0))) = false),
  proof: reflexivity{}
}

definition{
  name: outcomeOrZero,
  parameters: [outcome : (Types::Outcome String Nat)],
  result: Nat,
  body: (match outcome | Types::Outcome.failure reason => 0 | Types::Outcome.success value => value)
}

theorem{
  name: success_value,
  parameters: [],
  statement: ((outcomeOrZero (ctor Types::Outcome.success [String, Nat] 5)) = 5),
  proof: reflexivity{}
}

definition{
  name: swap,
  parameters: [pair : (Nat * Bool)],
  result: (Bool * Nat),
  body: ((second pair), (first pair))
}

definition{
  name: sumPair,
  parameters: [pair : (Nat * Nat)],
  result: Nat,
  body: (match pair | Prod.mk left right => (left + right))
}

theorem{
  name: swap_twice_first,
  parameters: [],
  statement: ((first (swap (3, true))) = true),
  proof: reflexivity{}
}

theorem{name: sum_pair, parameters: [], statement: ((sumPair (3, 4)) = 7), proof: reflexivity{}}

theorem{
  name: located_row,
  parameters: [],
  statement: ((first (project (record Types::Located [Bool] {value := true, position := (1, 2)}) position)) = 1),
  proof: reflexivity{}
}
