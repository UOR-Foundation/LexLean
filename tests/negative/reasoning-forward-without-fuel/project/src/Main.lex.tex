\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
structure{
  name: Memo,
  type_parameters: [],
  parameters: [],
  fields: [{name: temp, type: Nat}, {name: flags, type: Nat}, {name: alert, type: Nat}]
}

definition{
  name: Justified,
  parameters: [s : Memo, t : Memo],
  result: Prop,
  body: (((project t flags) = ((project s flags) + 1)) /\ ((project t flags) <= 2))
}

definition{name: Sound, parameters: [s : Memo], result: Prop, body: ((project s flags) <= 2)}

definition{
  name: Pending,
  parameters: [s : Memo],
  result: Nat,
  body: (prim subtract 2 (project s flags) : Nat),
  executable: true
}

theorem{
  name: sound_preserved,
  parameters: [s : Memo, t : Memo],
  statement: ((Sound s) -> ((Justified s t) -> (Sound t))),
  proof: linear_arithmetic{definitions: [Justified, Sound]},
  axioms: [Quot.sound, propext]
}

logic{
  name: Triage,
  axioms: [Quot.sound, propext],
  invariant: {predicate: Sound, preserves: sound_preserved},
  ranking: Pending,
  relation: Justified,
  state: {name: s, type: Memo, next: t}
}

theorem{
  name: fever_sound,
  parameters: [s : Memo],
  statement: ((((38 <? (project s temp)) && ((project s flags) <? 2)) = true) -> (Justified s (record Memo {temp := 37, flags := ((project s flags) + 1), alert := (project s alert)}))),
  proof: linear_arithmetic{definitions: [Justified]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: fever_progress,
  parameters: [s : Memo],
  statement: ((Sound s) -> ((((38 <? (project s temp)) && ((project s flags) <? 2)) = true) -> ((Pending (record Memo {temp := 37, flags := ((project s flags) + 1), alert := (project s alert)})) < (Pending s)))),
  proof: linear_arithmetic{definitions: [Pending, Sound]},
  axioms: [Quot.sound, propext]
}

inference_rule{
  name: Fever,
  axioms: [Quot.sound, propext],
  conclusion: (record Memo {temp := 37, flags := ((project s flags) + 1), alert := (project s alert)}),
  executable: true,
  guard: ((38 <? (project s temp)) && ((project s flags) <? 2)),
  logic: {member: Triage},
  progress: fever_progress,
  soundness: fever_sound
}

theorem{
  name: pick_sound,
  parameters: [s : Memo, b : Nat],
  statement: ((((100 <? b) && ((project s flags) <? 2)) = true) -> (Justified s (record Memo {temp := (project s temp), flags := ((project s flags) + 1), alert := b}))),
  proof: linear_arithmetic{definitions: [Justified]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: pick_progress,
  parameters: [s : Memo, b : Nat],
  statement: ((Sound s) -> ((((100 <? b) && ((project s flags) <? 2)) = true) -> ((Pending (record Memo {temp := (project s temp), flags := ((project s flags) + 1), alert := b})) < (Pending s)))),
  proof: linear_arithmetic{definitions: [Pending, Sound]},
  axioms: [Quot.sound, propext]
}

inference_rule{
  name: Pick,
  axioms: [Quot.sound, propext],
  binding: {
    name: b,
    type: Nat,
    candidates: (cons (project s temp) (cons ((project s temp) + 1) (nil Nat)))
  },
  conclusion: (record Memo {temp := (project s temp), flags := ((project s flags) + 1), alert := b}),
  executable: true,
  guard: ((100 <? b) && ((project s flags) <? 2)),
  logic: {member: Triage},
  progress: pick_progress,
  soundness: pick_sound
}

definition{name: Bounded, parameters: [x : Nat, r : Nat], result: Prop, body: (r <= 2)}

definition{
  name: boundedCheck,
  parameters: [x : Nat, r : Nat],
  result: Bool,
  body: (r <=? 2),
  executable: true
}

theorem{
  name: bounded_sound,
  parameters: [x : Nat, r : Nat],
  statement: (((boundedCheck x r) = true) -> (Bounded x r)),
  proof: linear_arithmetic{definitions: [Bounded, boundedCheck]},
  axioms: [propext]
}

theorem{
  name: bounded_complete,
  parameters: [x : Nat, r : Nat],
  statement: ((Bounded x r) -> ((boundedCheck x r) = true)),
  proof: linear_arithmetic{definitions: [Bounded, boundedCheck]},
  axioms: [propext]
}

verifier{
  name: Bound,
  axioms: [propext],
  candidate: {name: r, type: Nat},
  check: boundedCheck,
  complete: bounded_complete,
  sound: bounded_sound,
  specification: Bounded,
  subject: {name: x, type: Nat}
}

theorem{
  name: clinic_initial,
  parameters: [x : Nat],
  statement: (Sound (record Memo {temp := x, flags := 0, alert := 0})),
  proof: linear_arithmetic{definitions: [Sound]},
  axioms: [Quot.sound, propext]
}

theorem{
  name: clinic_fuel,
  parameters: [x : Nat],
  statement: ((Pending (record Memo {temp := x, flags := 0, alert := 0})) < (x + 3)),
  proof: linear_arithmetic{definitions: [Pending]},
  axioms: [Quot.sound, propext]
}

reasoner{
  name: Clinic,
  axioms: [Quot.sound, propext],
  answer: {name: s, type: Nat, value: (ctor Option.some [Nat] (project s flags))},
  claims: [initial_invariant{theorem: clinic_initial}, terminates{theorem: clinic_fuel}],
  executable: true,
  logic: {member: Triage},
  observation: {name: x, type: Nat},
  observe: (record Memo {temp := x, flags := 0, alert := 0}),
  rules: [{member: Fever}, {member: Pick}],
  strategy: forward{},
  verifier: {member: Bound}
}
\end{semanticmodule}
\end{lexlean}
