definition{
  name: quoteStep,
  parameters: [subject : Bytes, state : Quoting],
  result: (Option Quoting),
  body: (match (LexLeanStd.Text::byteAt subject (project state index))
    | Option.some byte => (let next : Nat := ((project state index) + 1); (let inside : Bool := (project state inString); (if (project state escaped) then (ctor Option.some [Quoting] (record Quoting {index := next, inString := inside, escaped := false})) else (if (if inside then (prim equal byte (92 : UInt8) : Bool) else false) then (ctor Option.some [Quoting] (record Quoting {index := next, inString := inside, escaped := true})) else (if (prim equal byte (34 : UInt8) : Bool) then (ctor Option.some [Quoting] (record Quoting {index := next, inString := (not inside), escaped := false})) else (ctor Option.some [Quoting] (record Quoting {index := next, inString := inside, escaped := false})))))))
    | Option.none => (ctor Option.none [Quoting])),
  executable: true

  name: argNext,
  parameters: [
    keep : Bool,

    depth : Nat,

  result: (Option ArgScan),
  body: (let quoting : Quoting := (if keep then (quoteRun (project atom text) (project scan inString) (project scan escaped)) else (record Quoting {index := 0, inString := (project scan inString), escaped := (project scan escaped)})); (ctor Option.some [ArgScan] (record ArgScan {pos := ((project scan pos) + 1), depth := depth, pieces := pieces, inString := (project quoting inString), escaped := (project quoting escaped), closing := (ctor Option.none [PrimitiveScan::Atom])}))),
  executable: true

definition{
  name: argStep,
  parameters: [ctx : Ctx, keep : Bool, scan : ArgScan],
  result: (Option ArgScan),
  body: (match (project scan closing)
    | Option.some closed => (ctor Option.none [ArgScan])
    | Option.none => (match (atomAt ctx (project scan pos))
      | Option.none => (ctor Option.none [ArgScan])
      | Option.some atom => (let quoted : Bool := (project scan inString); (let depth : Nat := (project scan depth); (let pieces : (List Bytes) := (project scan pieces); (if (if quoted then false else (isAtom atom 3 (lit "{"))) then (argNext keep scan atom (depth + 1) (cons (lit "{") pieces)) else (if (if quoted then false else (isAtom atom 3 (lit "}"))) then (if (depth =? 0) then (ctor Option.some [ArgScan] (record ArgScan {pos := (project scan pos), depth := depth, pieces := pieces, inString := quoted, escaped := (project scan escaped), closing := (ctor Option.some [PrimitiveScan::Atom] atom)})) else (argNext keep scan atom (prim subtract depth 1 : Nat) (cons (lit "}") pieces))) else (if (if quoted then false else ((PrimitiveScan::classCode atom) =? 6)) then (argNext keep scan atom depth pieces) else (argNext keep scan atom depth (cons (project atom text) pieces)))))))))),
  executable: true
}

  executable: true
