definition{
  name: itemJson,
  parameters: [item : PhraseItem, state : IrTerm::Renumber],
  result: (LexLeanStd.Json::Json * IrTerm::Renumber),
  body: (match item
    | PhraseItem.WWord entry form => ((jobj (IrTerm::member "k" (IrTerm::jtext "word") (IrTerm::member "entry" (jbytes entry) (IrTerm::member "form" (jbytes form) (jnil))))), state)
    | PhraseItem.WMath term => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson term state); ((jobj (IrTerm::member "k" (IrTerm::jtext "math") (IrTerm::member "t" (first written) (jnil)))), (second written)))
    | PhraseItem.WPunct entry => ((jobj (IrTerm::member "k" (IrTerm::jtext "punct") (IrTerm::member "entry" (jbytes entry) (jnil)))), state)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: blockJson,
  parameters: [block : Block, state : IrTerm::Renumber],
  result: (LexLeanStd.Json::Json * IrTerm::Renumber),
  body: (match block
    | Block.BSection component heading params nested => (let headed : (LexLeanStd.Json::Json * IrTerm::Renumber) := (phraseJson heading state); (let bound : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrTerm::termJson params (second headed)); (let inner : (LexLeanStd.Json::Json * IrTerm::Renumber) := (blockJson nested (second bound)); ((jobj (IrTerm::member "k" (IrTerm::jtext "section") (IrTerm::member "component" (jbytes component) (IrTerm::member "heading" (first headed) (IrTerm::member "params" (jarr (first bound)) (IrTerm::member "blocks" (jarr (first inner)) (jnil))))))), (second inner)))))
    | Block.BDecl declaration => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (IrProof::declarationJson declaration state); ((jobj (IrTerm::member "k" (IrTerm::jtext "declaration") (IrTerm::member "d" (first written) (jnil)))), (second written)))
    | Block.BNil => ((jnil), state)
    | Block.BItem head tail => (let written : (LexLeanStd.Json::Json * IrTerm::Renumber) := (blockJson head state); (let rest : (LexLeanStd.Json::Json * IrTerm::Renumber) := (blockJson tail (second written)); ((ctor LexLeanStd.Json::Json.JItem (first written) (first rest)), (second rest))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: block
}

  executable: true

definition{
  name: optionalTail,
  parameters: [linked : DocumentModule],
  result: LexLeanStd.Json::Json,
  body: (let withTooling : LexLeanStd.Json::Json := (match (project linked tooling)
    | Option.some value => (IrTerm::member "tooling" value (jnil))
    | Option.none => (jnil)); (let withSemantic : LexLeanStd.Json::Json := (match (project linked semantic)
    | Option.some value => (IrTerm::member "semantic" value withTooling)
    | Option.none => withTooling); (let withCore : LexLeanStd.Json::Json := (match (project linked core)
    | Option.some text => (IrTerm::member "core" (jbytes text) withSemantic)
    | Option.none => withSemantic); (let foreign : LexLeanStd.Json::Json := (foreignImports linked); (let withForeign : LexLeanStd.Json::Json := (match foreign
    | LexLeanStd.Json::Json.JNil => withCore
    | LexLeanStd.Json::Json.JBool flag => withCore
    | LexLeanStd.Json::Json.JInt negative magnitude => withCore
    | LexLeanStd.Json::Json.JStr text => withCore
    | LexLeanStd.Json::Json.JArr items => withCore
    | LexLeanStd.Json::Json.JObj members => withCore
    | LexLeanStd.Json::Json.JItem element rest => withCore
    | LexLeanStd.Json::Json.JMember key inner rest => (IrTerm::member "import_lean" (jobj foreign) withCore)); (match (project linked origin)
    | Option.some named => (IrTerm::member "origin" (jbytes named) withForeign)
    | Option.none => withForeign)))))),
  axioms: [Quot.sound, propext],
  executable: true
}
