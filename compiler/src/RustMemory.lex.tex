inductive{
  name: Flag,
  type_parameters: [],
  parameters: [],
  constructors: [{name: yes, fields: []}, {name: no, fields: []}]
}

inductive{
  name: Cells,
  type_parameters: [],
  parameters: [],
  constructors: [{name: CNil, fields: []}, {name: CCons, fields: [TargetSyntax::Value, Cells]}]

structure{
  name: Buf,
  type_parameters: [],
  parameters: [],
  fields: [{name: slots, type: Cells}, {name: len, type: Nat}]

definition{
  name: plus,
  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (match left | Nat.zero => right | Nat.succ more => ((plus more right) + 1)),
  recursive_argument: left
}

definition{
  name: lessThan,
  parameters: [index : Nat, bound : Nat],
  result: Flag,
  body: (match bound
    | Nat.zero => (ctor Flag.no)
    | Nat.succ more => (match index
      | Nat.zero => (ctor Flag.yes)
      | Nat.succ lower => (lessThan lower more))),
  recursive_argument: bound
}

definition{
  name: sameNat,
  parameters: [left : Nat, right : Nat],
  result: Flag,
  body: (match right
    | Nat.zero => (match left | Nat.zero => (ctor Flag.yes) | Nat.succ leftLess => (ctor Flag.no))
    | Nat.succ rightLess => (match left
      | Nat.zero => (ctor Flag.no)
      | Nat.succ leftLess => (sameNat leftLess rightLess))),
  recursive_argument: right
}

definition{
  name: minNat,
  parameters: [left : Nat, right : Nat],
  result: Nat,
  body: (match left
    | Nat.zero => 0
    | Nat.succ more => (match right | Nat.zero => 0 | Nat.succ less => ((minNat more less) + 1))),
  recursive_argument: left
}

definition{
  name: guardGet,
  parameters: [flag : Flag, found : (Option TargetSyntax::Value)],
  result: (Option TargetSyntax::Value),
  body: (match flag | Flag.yes => found | Flag.no => (ctor Option.none [TargetSyntax::Value]))
}

definition{
  name: pick,
  parameters: [
    flag : Flag,
    whenYes : (Option TargetSyntax::Value),
    whenNo : (Option TargetSyntax::Value)
  ],
  result: (Option TargetSyntax::Value),
  body: (match flag | Flag.yes => whenYes | Flag.no => whenNo)
}

definition{
  name: size,
  parameters: [cells : Cells],
  result: Nat,
  body: (match cells | Cells.CNil => 0 | Cells.CCons head tail => ((size tail) + 1)),
  recursive_argument: cells
}

definition{
  name: get,
  parameters: [cells : Cells, index : Nat],
  result: (Option TargetSyntax::Value),
  body: (match cells
    | Cells.CNil => (ctor Option.none [TargetSyntax::Value])
    | Cells.CCons head tail => (match index
      | Nat.zero => (ctor Option.some [TargetSyntax::Value] head)
      | Nat.succ lower => (get tail lower))),
  recursive_argument: cells
}

definition{
  name: put,
  parameters: [cells : Cells, index : Nat, item : TargetSyntax::Value],
  result: Cells,
  body: (match cells
    | Cells.CNil => (ctor Cells.CNil)
    | Cells.CCons head tail => (match index
      | Nat.zero => (ctor Cells.CCons item tail)
      | Nat.succ lower => (ctor Cells.CCons head (put tail lower item)))),
  recursive_argument: cells
}

definition{
  name: blank,
  parameters: [count : Nat, fill : TargetSyntax::Value],
  result: Cells,
  body: (match count
    | Nat.zero => (ctor Cells.CNil)
    | Nat.succ less => (ctor Cells.CCons fill (blank less fill))),
  recursive_argument: count
}

definition{
  name: append,
  parameters: [front : Cells, back : Cells],
  result: Cells,
  body: (match front
    | Cells.CNil => back
    | Cells.CCons head tail => (ctor Cells.CCons head (append tail back))),
  recursive_argument: front
}

definition{
  name: both,
  parameters: [left : Flag, right : Flag],
  result: Flag,
  body: (match left | Flag.yes => right | Flag.no => (ctor Flag.no))
}

definition{
  name: bufGet,
  parameters: [buf : Buf, index : Nat],
  result: (Option TargetSyntax::Value),
  body: (guardGet (lessThan index (project buf len)) (get (project buf slots) index))
}

definition{
  name: bufPush,
  parameters: [buf : Buf, item : TargetSyntax::Value],
  result: (Option Buf),
  body: (match (lessThan (project buf len) (size (project buf slots)))
    | Flag.yes => (ctor Option.some [Buf] (record Buf {slots := (put (project buf slots) (project buf len) item), len := ((project buf len) + 1)}))
    | Flag.no => (ctor Option.none [Buf]))
}

definition{
  name: bufTruncate,
  parameters: [buf : Buf, keep : Nat],
  result: Buf,
  body: (record Buf {slots := (project buf slots), len := (minNat keep (project buf len))})
}

theorem{
  name: size_put,
  parameters: [cells : Cells, index : Nat, item : TargetSyntax::Value],
  statement: ((size (put cells index item)) = (size cells)),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [put, size]}, binders: [], constructor: CNil},
      {
        proof: cases{
          branches: [
            {proof: simplify{definitions: [put, size]}, binders: [], constructor: zero},
            {
              proof: simplify{definitions: [hypothesis, put, size]},
              binders: [lower],
              constructor: succ
            }
          ],
          scrutinee: index
        },
        binders: [head, tail, hypothesis],
        constructor: CCons
      }
    ],
    generalizing: [index],
    scrutinee: cells
  },
  axioms: [propext]
}
