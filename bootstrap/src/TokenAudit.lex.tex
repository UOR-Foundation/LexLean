      | LexemeKind.Ident segs => false

definition{

definition{
  name: auditIdent,
  parameters: [
    segments : (List Segment),
    tail : (List Lexeme),
    attributeDepth : Nat,
    pending : Bool,
    imports : (List Bytes)
  ],
  result: AuditState,
  body: (match (firstForbiddenConstant segments)
    | Option.some badConst => (let full : Bytes := (dottedSegments segments); (let msg : Bytes := (prim append (prim append (prim append (prim append (prim utf8_encode "the forbidden constant `" : Bytes) badConst : Bytes) (prim utf8_encode "` (in `" : Bytes) : Bytes) full : Bytes) (prim utf8_encode "`)" : Bytes) : Bytes); (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.some [Bytes] msg)})))
    | Option.none => (match segments
      | List.nil => (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])})
      | List.cons onlySeg restSegs => (match restSegs
        | List.nil => (let badTok : Bool := (isForbiddenToken (project onlySeg text)); (let activeTok : Bool := (if (not (project onlySeg quoted)) then true else (0 <? attributeDepth)); (if (if badTok then activeTok else false) then (let msg : Bytes := (prim append (prim append (prim utf8_encode "the forbidden token `" : Bytes) (project onlySeg text) : Bytes) (prim utf8_encode "`" : Bytes) : Bytes); (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.some [Bytes] msg)})) else (if (project onlySeg quoted) then (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])}) else (if (if (prim equal (project onlySeg text) (prim utf8_encode "attribute" : Bytes) : Bool) then (nextOpensBracket tail) else false) then (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := true, error := (ctor Option.none [Bytes])}) else (if (prim equal (project onlySeg text) (prim utf8_encode "set_option" : Bytes) : Bool) then (let optName : Bytes := (dottedNext tail); (let okOpt : Bool := (isAllowedOption optName); (if okOpt then (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])}) else (let msg : Bytes := (prim append (prim append (prim utf8_encode "the option `" : Bytes) optName : Bytes) (prim utf8_encode "` is not allowed" : Bytes) : Bytes); (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.some [Bytes] msg)}))))) else (if (prim equal (project onlySeg text) (prim utf8_encode "import" : Bytes) : Bool) then (let modName : Bytes := (dottedNext tail); (let okMod : Bool := (bytesInList modName imports); (if okMod then (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])}) else (let msg : Bytes := (prim append (prim append (prim utf8_encode "the import of `" : Bytes) modName : Bytes) (prim utf8_encode "` is not allowed" : Bytes) : Bytes); (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.some [Bytes] msg)}))))) else (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])}))))))))
        | List.cons secondSeg moreSegs => (record AuditState {remaining := tail, attributeDepth := attributeDepth, pending := pending, error := (ctor Option.none [Bytes])})))),
  executable: true
}

definition{
  name: stepAudit,
  parameters: [imports : (List Bytes), state : AuditState],
  result: (Option AuditState),
  body: (let dep : Nat := (project state attributeDepth); (let pend : Bool := (project state pending); (match (project state error)
    | Option.some msg => (ctor Option.none [AuditState])
    | Option.none => (match (project state remaining)
      | List.nil => (ctor Option.none [AuditState])
      | List.cons current tail => (match (project current kind)
        | LexemeKind.Ident segs => (let nextSt : AuditState := (auditIdent segs tail dep pend imports); (ctor Option.some [AuditState] nextSt))
        | LexemeKind.Number => (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := pend, error := (ctor Option.none [Bytes])}))
        | LexemeKind.Literal => (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := pend, error := (ctor Option.none [Bytes])}))
        | LexemeKind.Command cmd => (let msg : Bytes := (prim append (prim append (prim utf8_encode "the command `" : Bytes) cmd : Bytes) (prim utf8_encode "`" : Bytes) : Bytes); (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := pend, error := (ctor Option.some [Bytes] msg)})))
        | LexemeKind.Symbol sym => (if (if (prim equal sym (prim utf8_encode "@" : Bytes) : Bool) then (nextOpensBracket tail) else false) then (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := true, error := (ctor Option.none [Bytes])})) else (if (prim equal sym (prim utf8_encode "[" : Bytes) : Bool) then (if pend then (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := 1, pending := false, error := (ctor Option.none [Bytes])})) else (if (0 <? dep) then (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := (dep + 1), pending := false, error := (ctor Option.none [Bytes])})) else (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := 0, pending := false, error := (ctor Option.none [Bytes])})))) else (if (prim equal sym (prim utf8_encode "]" : Bytes) : Bool) then (if (0 <? dep) then (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := (prim subtract dep 1 : Nat), pending := pend, error := (ctor Option.none [Bytes])})) else (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := pend, error := (ctor Option.none [Bytes])}))) else (ctor Option.some [AuditState] (record AuditState {remaining := tail, attributeDepth := dep, pending := pend, error := (ctor Option.none [Bytes])})))))))))),
  executable: true
}

definition{
  name: auditTokens,
  parameters: [text : Bytes, imports : (List Bytes)],
  result: (Result Unit Bytes),
  body: (match (lex text)
    | Result.error msg => (ctor Result.error [Unit, Bytes] msg)
    | Result.ok lexemes => (let count : Nat := (listLength [Lexeme] lexemes); (let finalState : AuditState := (first (prim iterate_until (fun [imports] (s : AuditState) => (stepAudit imports s)) (count + 1) (record AuditState {remaining := lexemes, attributeDepth := 0, pending := false, error := (ctor Option.none [Bytes])}) : (AuditState * Bool))); (match (project finalState error)
      | Option.some msg => (ctor Result.error [Unit, Bytes] msg)
      | Option.none => (ctor Result.ok [Unit, Bytes] ()))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]}
}
