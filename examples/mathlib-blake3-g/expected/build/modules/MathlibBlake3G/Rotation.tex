\begin{lemma}
\label{ll:rotation:unmix}
For every \(\operatorname{bitVec}(32)\) \(p\) and \(\operatorname{bitVec}(32)\) \(q\) and \(\operatorname{nat}\) \(n\), \(\operatorname{hXorHXor}(\operatorname{bitVecRotateLeft}(\operatorname{bitVecRotateRight}(\operatorname{hXorHXor}(p, q), n), n), q) = p\).
\end{lemma}
\begin{proof}
Rewrite the goal using \(\rightarrow \operatorname{blake3GHostRotateLeftRotateRight}(\operatorname{hXorHXor}(p, q), n)\).
The goal follows from \(\texttt{Cancellation::xor-cancel}(p, q)\).
\end{proof}
