\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
theorem{
  name: countdown_decreases,
  parameters: [number : Nat],
  statement: (((number <? 2) = false) -> ((prim subtract number 2 : Nat) < number)),
  proof: linear_arithmetic{},
  axioms: [Quot.sound, propext]
}

definition{
  name: countdown,
  parameters: [number : Nat],
  result: Nat,
  body: (match number
    | Nat.zero => 0
    | Nat.succ previous => (countdown (prim subtract number 2 : Nat))),
  axioms: [Quot.sound, propext],
  termination: {evidence: [countdown_decreases], measure: number}
}
\end{semanticmodule}
\end{lexlean}
