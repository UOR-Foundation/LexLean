\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: mapList,
  type_parameters: [Input, Output],
  parameters: [transform : (Fn ('Input) 'Output), values : (List 'Input)],
  result: (List 'Output),
  body: (match values
    | List.nil => (nil 'Output)
    | List.cons head tail => (cons (apply transform head) (mapList ['Input, 'Output] transform tail))),
  recursive_argument: values
}

definition{
  name: bad,
  parameters: [values : (List Nat)],
  result: (List Nat),
  body: (mapList (fun (value : Nat) => value) values)
}
\end{semanticmodule}
\end{lexlean}
