\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: encodeNats,
  parameters: [items : (List Nat)],
  result: (List TargetSyntax::Value),
  body: (match items
    | List.nil => (nil TargetSyntax::Value)
    | List.cons item rest => (cons (ctor TargetSyntax::Value.nat item) (encodeNats rest))),
  recursive_argument: items
}
