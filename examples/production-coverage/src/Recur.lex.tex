definition{
  name: roseSize,

  body: (match rose | Syntax::Rose.node label children => (1 + (forestSize ['Item] children))),
  executable: true,
  mutual: RoseSize,
  recursive_argument: rose
}

definition{
  name: forestSize,
  type_parameters: [Item],
  parameters: [forest : (List (Syntax::Rose 'Item))],
  result: Nat,
  body: (match forest
    | List.nil => 0
    | List.cons head tail => ((roseSize ['Item] head) + (forestSize ['Item] tail))),
  executable: true,
  mutual: RoseSize,
  recursive_argument: forest
}

  name: exprSize,
  parameters: [expression : Syntax::Expr],
  result: Nat,
  body: (match expression
    | Syntax::Expr.literal value => 1
    | Syntax::Expr.plus left right => (((exprSize left) + (exprSize right)) + 1)
    | Syntax::Expr.block statements result => ((statementsSize statements) + (exprSize result))),
  executable: true,
  mutual: SyntaxSize,
  recursive_argument: expression
}

  name: reassociate_plus,

    rest : Syntax::Term
  ],
  statement: ((term = (ctor Syntax::Term.plus left right)) -> ((left = (ctor Syntax::Term.plus inner rest)) -> ((weight (ctor Syntax::Term.plus inner (ctor Syntax::Term.plus rest right))) < (weight term)))),

  axioms: [Quot.sound, propext]

definition{
  name: reassociate,
  parameters: [term : Syntax::Term],
  result: Syntax::Term,
  body: (match term
    | Syntax::Term.literal value => (ctor Syntax::Term.literal value)
    | Syntax::Term.plus left right => (match left
      | Syntax::Term.literal value => (ctor Syntax::Term.plus (ctor Syntax::Term.literal value) (reassociate right))
      | Syntax::Term.plus inner rest => (reassociate (ctor Syntax::Term.plus inner (ctor Syntax::Term.plus rest right))))),
  axioms: [Quot.sound, propext],
  executable: true,
  termination: {evidence: [reassociate_literal, reassociate_plus], measure: (weight term)}

definition{
  name: foldExpr,
  parameters: [expression : Syntax::Expr],
  result: Syntax::Expr,
  body: (match expression
    | Syntax::Expr.literal value => (ctor Syntax::Expr.literal value)
    | Syntax::Expr.plus left right => (let foldedLeft : Syntax::Expr := (foldExpr left); (let foldedRight : Syntax::Expr := (foldExpr right); (match foldedLeft
      | Syntax::Expr.literal leftValue => (match foldedRight
        | Syntax::Expr.literal rightValue => (ctor Syntax::Expr.literal (leftValue + rightValue))
        | Syntax::Expr.plus rightFirst rightSecond => (ctor Syntax::Expr.plus foldedLeft foldedRight)
        | Syntax::Expr.block rightStatements rightResult => (ctor Syntax::Expr.plus foldedLeft foldedRight))
      | Syntax::Expr.plus leftFirst leftSecond => (ctor Syntax::Expr.plus foldedLeft foldedRight)
      | Syntax::Expr.block leftStatements leftResult => (ctor Syntax::Expr.plus foldedLeft foldedRight))))
    | Syntax::Expr.block statements result => (ctor Syntax::Expr.block (foldStatements statements) (foldExpr result))),
  executable: true,
  mutual: Folding,
  recursive_argument: expression
}

definition{
  name: foldStatements,
  parameters: [statements : (List Syntax::Stmt)],
  result: (List Syntax::Stmt),
  body: (match statements
    | List.nil => (nil Syntax::Stmt)
    | List.cons head tail => (cons (foldStatement head) (foldStatements tail))),
  executable: true,
  mutual: Folding,
  recursive_argument: statements
}

definition{
  name: foldStatement,

        constructor: Syntax::Stmt.assign

      {
        body: (ctor Syntax::Stmt.sequence (foldStatement var{
          name: first
        }) (foldStatement var{name: second})),
        binders: [first, second],
        constructor: Syntax::Stmt.sequence
      }
    ],
    scrutinee: statement
  },
  executable: true,
  mutual: Folding,
  recursive_argument: statement
}
