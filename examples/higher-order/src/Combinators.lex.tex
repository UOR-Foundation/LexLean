\begin{lexlean}{Combinators}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: mapList,
  type_parameters: [Input, Output],
  parameters: [transform : (Fn ('Input) 'Output), values : (List 'Input)],
  result: (List 'Output),
  body: (match values
    | List.nil => (nil 'Output)
    | List.cons head tail => (cons (apply transform head) (mapList ['Input, 'Output] transform tail))),
  executable: true,
  recursive_argument: values
}

definition{
  name: foldList,
  type_parameters: [Input, Output],
  parameters: [step : (Fn ('Output 'Input) 'Output), initial : 'Output, values : (List 'Input)],
  result: 'Output,
  body: (match values
    | List.nil => initial
    | List.cons head tail => (foldList ['Input, 'Output] step (apply step initial head) tail)),
  executable: true,
  recursive_argument: values
}

definition{
  name: compose,
  type_parameters: [Input, Output, Final],
  parameters: [outer : (Fn ('Output) 'Final), inner : (Fn ('Input) 'Output)],
  result: (Fn ('Input) 'Final),
  body: (fun [inner, outer] (value : 'Input) => (apply outer (apply inner value)))
}

definition{
  name: increment,
  parameters: [value : Nat],
  result: Nat,
  body: (value + 1),
  executable: true
}

inductive{
  name: Arith,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: literal, fields: [Nat]},
    {name: plus, fields: [Arith, Arith]},
    {name: twice, fields: [Arith]}
  ]
}

structure{
  name: Visitor,
  type_parameters: [],
  parameters: [],
  fields: [
    {name: onLiteral, type: (Fn (Nat) Nat)},
    {name: onPlus, type: (Fn (Nat Nat) Nat)},
    {name: onTwice, type: (Fn (Nat) Nat)}
  ]
}

definition{
  name: visit,
  parameters: [visitor : Visitor, expression : Arith],
  result: Nat,
  body: (match expression
    | Arith.literal value => (apply (project visitor onLiteral) value)
    | Arith.plus left right => (apply (project visitor onPlus) (visit visitor left) (visit visitor right))
    | Arith.twice inner => (apply (project visitor onTwice) (visit visitor inner))),
  recursive_argument: expression
}

definition{
  name: evaluate,
  parameters: [expression : Arith],
  result: Nat,
  body: (match expression
    | Arith.literal value => value
    | Arith.plus left right => ((evaluate left) + (evaluate right))
    | Arith.twice inner => (let half : Nat := (evaluate inner); (half + half))),
  executable: true,
  recursive_argument: expression
}
\end{semanticmodule}
\end{lexlean}
