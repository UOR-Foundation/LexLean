definition{
  name: termJson,
  parameters: [term : Term, state : Renumber],
  result: (LexLeanStd.Json::Json * Renumber),
  body: (match term
    | Term.TSort level => ((ctor LexLeanStd.Json::Json.JObj (member "k" (jtext "sort") (member "u" (universeJson level) (ctor LexLeanStd.Json::Json.JNil)))), state)
    | Term.TLocal localId => (let resolved : (Resolved * Renumber) := (resolve state localId); ((localJson (first resolved)), (second resolved)))
    | Term.TGlobal global levels => ((globalTermJson global levels), state)
    | Term.TApp function arguments implicits => (let head : (LexLeanStd.Json::Json * Renumber) := (termJson function state); (let args : (LexLeanStd.Json::Json * Renumber) := (termJson arguments (second head)); ((appJson (first head) (first args) implicits), (second args))))
    | Term.TPi binders body => (let bound : (LexLeanStd.Json::Json * Renumber) := (termJson binders state); (let inner : (LexLeanStd.Json::Json * Renumber) := (termJson body (second bound)); ((ctor LexLeanStd.Json::Json.JObj (member "k" (jtext "pi") (member "b" (ctor LexLeanStd.Json::Json.JArr (first bound)) (member "v" (first inner) (ctor LexLeanStd.Json::Json.JNil))))), (second inner))))
    | Term.TLam binders body => (let bound : (LexLeanStd.Json::Json * Renumber) := (termJson binders state); (let inner : (LexLeanStd.Json::Json * Renumber) := (termJson body (second bound)); ((ctor LexLeanStd.Json::Json.JObj (member "k" (jtext "lam") (member "b" (ctor LexLeanStd.Json::Json.JArr (first bound)) (member "v" (first inner) (ctor LexLeanStd.Json::Json.JNil))))), (second inner))))
    | Term.TLet localId mode spelling typed bound body => (let typeJson : (LexLeanStd.Json::Json * Renumber) := (termJson typed state); (let valueJson : (LexLeanStd.Json::Json * Renumber) := (termJson bound (second typeJson)); (let index : (Nat * Renumber) := (bind (second valueJson) localId); (let inner : (LexLeanStd.Json::Json * Renumber) := (termJson body (second index)); ((ctor LexLeanStd.Json::Json.JObj (member "k" (jtext "let") (member "b" (binderJson (first index) mode spelling (first typeJson)) (member "x" (first valueJson) (member "v" (first inner) (ctor LexLeanStd.Json::Json.JNil)))))), (second inner))))))
    | Term.TNat decimal expected => (let inner : (LexLeanStd.Json::Json * Renumber) := (termJson expected state); ((ctor LexLeanStd.Json::Json.JObj (member "k" (jtext "nat") (member "d" (ctor LexLeanStd.Json::Json.JStr decimal) (member "t" (first inner) (ctor LexLeanStd.Json::Json.JNil))))), (second inner)))
    | Term.TNil => ((ctor LexLeanStd.Json::Json.JNil), state)
    | Term.TItem head tail => (let first_ : (LexLeanStd.Json::Json * Renumber) := (termJson head state); (let rest : (LexLeanStd.Json::Json * Renumber) := (termJson tail (second first_)); ((ctor LexLeanStd.Json::Json.JItem (first first_) (first rest)), (second rest))))
    | Term.TBind localId mode spelling typed tail => (let typeJson : (LexLeanStd.Json::Json * Renumber) := (termJson typed state); (let index : (Nat * Renumber) := (bind (second typeJson) localId); (let rest : (LexLeanStd.Json::Json * Renumber) := (termJson tail (second index)); ((ctor LexLeanStd.Json::Json.JItem (binderJson (first index) mode spelling (first typeJson)) (first rest)), (second rest)))))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: term
}

definition{
  name: nodeCount,
  parameters: [term : Term],
  result: Nat,
  body: (match term
    | Term.TSort level => 1
    | Term.TLocal localId => 1
    | Term.TGlobal global levels => 1
    | Term.TApp function arguments implicits => ((1 + (nodeCount function)) + (nodeCount arguments))
    | Term.TPi binders body => ((1 + (nodeCount binders)) + (nodeCount body))
    | Term.TLam binders body => ((1 + (nodeCount binders)) + (nodeCount body))
    | Term.TLet localId mode spelling typed bound body => (((1 + (nodeCount typed)) + (nodeCount bound)) + (nodeCount body))
    | Term.TNat decimal expected => (1 + (nodeCount expected))
    | Term.TNil => 0
    | Term.TItem head tail => ((nodeCount head) + (nodeCount tail))
    | Term.TBind localId mode spelling typed tail => ((nodeCount typed) + (nodeCount tail))),
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]},
  recursive_argument: term
}

definition{
  name: termDepth,
  parameters: [term : Term],
  result: Nat,
  body: (match term
    | Term.TSort level => 1
    | Term.TLocal localId => 1
    | Term.TGlobal global levels => 1
    | Term.TApp function arguments implicits => (1 + (larger (termDepth function) (termDepth arguments)))
    | Term.TPi binders body => (1 + (larger (termDepth binders) (termDepth body)))
    | Term.TLam binders body => (1 + (larger (termDepth binders) (termDepth body)))
    | Term.TLet localId mode spelling typed bound body => (1 + (larger (larger (termDepth typed) (termDepth bound)) (termDepth body)))
    | Term.TNat decimal expected => (1 + (termDepth expected))
    | Term.TNil => 0
    | Term.TItem head tail => (larger (termDepth head) (termDepth tail))
    | Term.TBind localId mode spelling typed tail => (larger (termDepth typed) (termDepth tail))),
  executable: true,
  production: {effects: [allocation, overflow, recursion], targets: [rust-std]},
  recursive_argument: term
}
