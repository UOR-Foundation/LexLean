definition{

  parameters: [left : Ty, right : Ty],
  result: Bool,
  body: (match left
    | Ty.TUnit => ((tyTag right) =? 0)
    | Ty.TBool => ((tyTag right) =? 1)

    | Ty.TString => ((tyTag right) =? 5)
    | Ty.TBytes => ((tyTag right) =? 6)
    | Ty.TOrdering => ((tyTag right) =? 7)
    | Ty.TNil => ((tyTag right) =? 14)
    | Ty.TFixed width => (if ((tyTag right) =? 4) then ((kindIndex width) =? (kindIndex (tyWidth right))) else false)
    | Ty.TOption value => (let same : Bool := (tyEq value (tyFirst right)); (if ((tyTag right) =? 8) then same else false))
    | Ty.TResult ok error => (let okSame : Bool := (tyEq ok (tyFirst right)); (let errorSame : Bool := (tyEq error (tySecond right)); (if ((tyTag right) =? 9) then (if okSame then errorSame else false) else false)))
    | Ty.TList element => (let same : Bool := (tyEq element (tyFirst right)); (if ((tyTag right) =? 10) then same else false))

    | Ty.TAdt index => (if ((tyTag right) =? 12) then (index =? (tyAdtIndex right)) else false)
    | Ty.TFn parameters result => (let parametersSame : Bool := (tyEq parameters (tyFirst right)); (let resultSame : Bool := (tyEq result (tySecond right)); (if ((tyTag right) =? 13) then (if parametersSame then resultSame else false) else false)))
    | Ty.TCons head tail => (let headSame : Bool := (tyEq head (tyFirst right)); (let tailSame : Bool := (tyEq tail (tySecond right)); (if ((tyTag right) =? 15) then (if headSame then tailSame else false) else false)))),

  recursive_argument: left
