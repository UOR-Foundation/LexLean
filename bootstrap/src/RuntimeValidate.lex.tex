definition{
  name: exprOk,
  parameters: [sc : Sc, fr : Fr, e : Expr],
  result: (Option Bytes),
  body: (match e
    | Expr.EVar name => (match (findVar (project fr vars) name)
      | Option.some held => (ctor Option.none [Bytes])
      | Option.none => (if (inList (project sc statics) name) then (ctor Option.none [Bytes]) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: variable-scope: ") (rcat (rtick name) (rtext " is not in scope"))))))
    | Expr.ELit => (ctor Option.none [Bytes])
    | Expr.EPath path => (pathOk sc path)
    | Expr.ECall path args => (match (callCheck sc path (exprLen args))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprsOk sc fr args))
    | Expr.EMethod recv name args => (match (exprOk sc fr recv)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (methodOk sc fr recv name (exprLen args))
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (exprsOk sc fr args)))
    | Expr.EField inner at => (exprOk sc fr inner)
    | Expr.EFieldNamed inner name => (exprOk sc fr inner)
    | Expr.EBorrow inner => (exprOk sc fr inner)
    | Expr.EEquals a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.EMatch scrutinee arms => (match (exprOk sc fr scrutinee)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (if (isEmptyChain arms) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: match-arms: ") (rtext "a match with no arm"))) else (armsOk sc fr arms)))
    | Expr.ELet pat ty value => (ctor Option.some [Bytes] (rcat (rtext "runtime support: let-position: ") (rtext "a `let` that is not a statement of a block")))
    | Expr.EAssign place value => (match (assignable fr place)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (exprOk sc fr place)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (exprOk sc fr value)))
    | Expr.EIf cond thenBlock elseBlock => (match (exprOk sc fr cond)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (blockOk sc fr thenBlock)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (blockOk sc fr elseBlock)))
    | Expr.EWhile cond body => (match (exprOk sc fr cond)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (blockOk sc (withBound fr (nil Var) true) body))
    | Expr.EWhileLet pat value body => (match (exprOk sc fr value)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (patOk sc pat)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match (bindsOf (nil Var) pat)
          | Result.error problem => (ctor Option.some [Bytes] problem)
          | Result.ok bound => (blockOk sc (withBound fr bound true) body))))
    | Expr.ELoop body => (blockOk sc (withBound fr (nil Var) true) body)
    | Expr.EFor pat value body => (match (exprOk sc fr value)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (patOk sc pat)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match (bindsOf (nil Var) pat)
          | Result.error problem => (ctor Option.some [Bytes] problem)
          | Result.ok bound => (blockOk sc (withBound fr bound true) body))))
    | Expr.EBreak => (if (project fr inLoop) then (ctor Option.none [Bytes]) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: break-position: ") (rtext "a `break` outside a loop"))))
    | Expr.EReturn value => (if (isNone value) then (ctor Option.none [Bytes]) else (exprOk sc fr value))
    | Expr.ETuple items => (if ((exprLen items) <? 2) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: tuple-arity: ") (rcat (rtext "a tuple of ") (rdec (exprLen items))))) else (exprsOk sc fr items))
    | Expr.EStructLit path names values => (match (pathOk sc path)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (recordFields sc path)
        | Option.none => (ctor Option.some [Bytes] (rcat (rtext "runtime support: struct-fields: ") (rcat (rtick (joinWith path "::")) (rtext " is no record struct or record variant"))))
        | Option.some held => (if (sameFields names held (exprLen values)) then (exprsOk sc fr values) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: struct-fields: ") (rcat (rtick (joinWith path "::")) (rcat (rtext " is given ") (rcat (joinWith names ", ") (rcat (rtext ", not ") (joinWith held ", "))))))))))
    | Expr.EBinary a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.ENot inner => (exprOk sc fr inner)
    | Expr.ECast inner ty => (match (tyOk sc ty (prim length (project fr generics) : Nat) (not (isAbsent (project fr selfTy))))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr inner))
    | Expr.EIndex a b => (match (exprOk sc fr a)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc fr b))
    | Expr.ERange inner lower upper => (match (exprOk sc fr inner)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (if (isNone lower) then (ctor Option.none [Bytes]) else (exprOk sc fr lower))
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (if (isNone upper) then (ctor Option.none [Bytes]) else (exprOk sc fr upper))))
    | Expr.EDeref inner => (exprOk sc fr inner)
    | Expr.EBorrowMut inner => (exprOk sc fr inner)
    | Expr.EMethodOf recv name tys args => (match (exprOk sc fr recv)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (turbofishTypes name)
        | Option.none => (ctor Option.some [Bytes] (rcat (rtext "runtime support: method-name: ") (rcat (rtick name) (rtext " takes no type arguments in the closed table"))))
        | Option.some count => (if (if (count =? (tyLen tys)) then ((turbofishArgs name) =? (exprLen args)) else false) then (if (if (project sc core) then (bEq name (rtext "collect")) else false) then (ctor Option.some [Bytes] (rcat (rtext "runtime support: hidden-allocation: ") (rcat (rtext "rust-core renders ") (rcat (rtick name) (rtext ", which requires heap allocation"))))) else (match (tyOk sc tys (prim length (project fr generics) : Nat) (not (isAbsent (project fr selfTy))))
          | Option.some stop => (ctor Option.some [Bytes] stop)
          | Option.none => (exprsOk sc fr args))) else (ctor Option.some [Bytes] (rcat (rtext "runtime support: method-arity: ") (rcat (rtick name) (rcat (rtext " takes ") (rcat (rdec count) (rcat (rtext " type arguments and ") (rcat (rdec (turbofishArgs name)) (rtext " arguments")))))))))))
    | Expr.EBlock stmts tail => (match (stmtsOk sc fr stmts)
      | Result.error problem => (ctor Option.some [Bytes] problem)
      | Result.ok inner => (exprOk sc inner tail))
    | Expr.ENone => (ctor Option.none [Bytes])
    | Expr.ENil => (ctor Option.none [Bytes])
    | Expr.ECons d0 d1 => (ctor Option.none [Bytes])
    | Expr.ELetS d0 d1 d2 d3 => (ctor Option.none [Bytes])
    | Expr.EArm d0 d1 d2 d3 => (ctor Option.none [Bytes])),
  axioms: [],
  executable: true,
  mutual: CheckRt,
  recursive_argument: e
}

definition{

    | Item.IUnit name => (ctor Option.none [Bytes])
    | Item.IAlias name params ty => (tyOk sc ty (prim length params : Nat) false)
    | Item.IStatic name ty init => (match (tyOk sc ty 0 false)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (exprOk sc (emptyFr ()) init))
    | Item.ITrait name sigs => (match (match (firstDuplicateSig sigs)
        | Option.some twice => (ctor Option.some [Bytes] (rcat (rtext "runtime support: trait-method: ") (rcat (rtick name) (rcat (rtext " declares ") (rcat (rtick twice) (rtext " twice"))))))
        | Option.none => (ctor Option.none [Bytes]))
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (prim list_fold (fun [sc] (found : (Option Bytes)) (sig : Sig) => (match found
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (sigOk sc sig 0 true))) (ctor Option.none [Bytes]) sigs : (Option Bytes)))
    | Item.IImpl generics tr ty fns => (match (genericsOk sc generics)
      | Option.some stop => (ctor Option.some [Bytes] stop)
      | Option.none => (match (tyOk sc ty (prim length generics : Nat) false)
        | Option.some stop => (ctor Option.some [Bytes] stop)
        | Option.none => (match tr
          | List.cons named more => (completes sc named generics ty fns)
          | List.nil => (prim list_fold (fun [generics, sc, ty] (found : (Option Bytes)) (f : FnDef) => (match found
            | Option.some stop => (ctor Option.some [Bytes] stop)
            | Option.none => (fnOk sc generics ty f))) (ctor Option.none [Bytes]) fns : (Option Bytes)))))
    | Item.IFn f => (freeFnOk sc f)
    | Item.ITuple name generics tys => (tyOk sc tys (prim length generics : Nat) false)
    | Item.IRecord name generics fields tys => (tyOk sc tys (prim length generics : Nat) false)
    | Item.IEnum name generics variants => (prim list_fold (fun [generics, sc] (found : (Option Bytes)) (variant : Variant) => (match found
      | Option.some stop => (ctor Option.some [Bytes] stop)
