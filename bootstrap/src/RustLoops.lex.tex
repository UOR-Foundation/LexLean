definition{
  name: occurs,
  parameters: [subject : RustAst::Expr, me : Nat],
  result: Bool,
  body: (match subject
    | RustAst::Expr.ELit u2x0 u2x1 => false
    | RustAst::Expr.ERead u2x0 u2x1 u2x2 => false
    | RustAst::Expr.EWrap kind inner at => (occurs inner me)
    | RustAst::Expr.ECall callee args prop at => (match callee
      | RustAst::Callee.CalFn called => (if (called =? me) then true else (occurs args me))
      | RustAst::Callee.CalRt u1x0 => (occurs args me)
      | RustAst::Callee.CalMachine u1x0 u1x1 => (occurs args me))
    | RustAst::Expr.EApply holder args prop at => (occurs args me)
    | RustAst::Expr.EConstruct con args at => (occurs args me)
    | RustAst::Expr.EPair left right at => (if (occurs left me) then true else (occurs right me))
    | RustAst::Expr.EIf c t e at => (if (occurs c me) then true else (if (occurs t me) then true else (occurs e me)))
    | RustAst::Expr.EMatch s arms at => (if (occurs s me) then true else (occurs arms me))
