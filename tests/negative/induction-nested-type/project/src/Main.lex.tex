\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
inductive{
  name: Json,
  type_parameters: [],
  parameters: [],
  constructors: [
    {name: null, fields: []},
    {name: text, fields: [String]},
    {name: array, fields: [(List Json)]}
  ]
}

definition{
  name: size,
  parameters: [value : Json],
  result: Nat,
  body: (match value
    | Json.null => 1
    | Json.text content => 1
    | Json.array items => (1 + (sizeList items))),
  mutual: Size,
  recursive_argument: value
}

definition{
  name: sizeList,
  parameters: [items : (List Json)],
  result: Nat,
  body: (match items | List.nil => 0 | List.cons head tail => ((size head) + (sizeList tail))),
  mutual: Size,
  recursive_argument: items
}

definition{
  name: leaves,
  parameters: [value : Json],
  result: Nat,
  body: (match value
    | Json.null => 1
    | Json.text content => 1
    | Json.array items => (leavesList items)),
  mutual: Leaves,
  recursive_argument: value
}

definition{
  name: leavesList,
  parameters: [items : (List Json)],
  result: Nat,
  body: (match items | List.nil => 0 | List.cons head tail => ((leaves head) + (leavesList tail))),
  mutual: Leaves,
  recursive_argument: items
}

theorem{
  name: leaves_le_size,
  parameters: [value : Json],
  statement: ((leaves value) <= (size value)),
  proof: induction{
    branches: [
      {proof: linear_arithmetic{definitions: [leaves, size]}, binders: [], constructor: null},
      {
        proof: linear_arithmetic{definitions: [leaves, size]},
        binders: [content],
        constructor: text
      },
      {proof: linear_arithmetic{definitions: [leaves, size]}, binders: [items, itemsHyp], constructor: array}
    ],
    scrutinee: value
  },
  axioms: [Quot.sound, propext]
}
\end{semanticmodule}
\end{lexlean}
