definition{
  name: matchesTy,
  parameters: [fuel : Nat, shape : RuntimeSyntax::RtType, value : TargetSyntax::Value],
  result: Bool,
  body: (match fuel
    | Nat.zero => false
    | Nat.succ remaining => (match shape
      | RuntimeSyntax::RtType.unit => ((tagOf value) =? 0)
      | RuntimeSyntax::RtType.bool => ((tagOf value) =? 1)
      | RuntimeSyntax::RtType.nat => ((tagOf value) =? 2)
      | RuntimeSyntax::RtType.fixed kind => (((tagOf value) =? (kindIndex kind)) || ((tagOf value) =? (aliasTag kind)))
      | RuntimeSyntax::RtType.ordering => ((tagOf value) =? 14)
      | RuntimeSyntax::RtType.selfType => true
      | RuntimeSyntax::RtType.param ignoredIndex => true
      | RuntimeSyntax::RtType.ref inner => (matchesTy remaining inner value)
      | RuntimeSyntax::RtType.pair leftShape rightShape => (match (pairParts value)
        | Option.none => false
        | Option.some parts => ((matchesTy remaining leftShape (first parts)) && (matchesTy remaining rightShape (second parts))))
      | RuntimeSyntax::RtType.named ignoredName ignoredArguments => false)),
  recursive_argument: fuel
}

    | RuntimeSyntax::RtItem.implItem generics trait selfType functions => (findFree rest wanted)

    | RuntimeSyntax::RtItem.alias ignoredAlias2 ignoredParameters ignoredTarget => (findMethod fuel rest wanted value)
    | RuntimeSyntax::RtItem.staticItem ignoredStatic ignoredType ignoredInit => (findMethod fuel rest wanted value)
    | RuntimeSyntax::RtItem.traitItem ignoredTrait ignoredSigs => (findMethod fuel rest wanted value)

        | Option.none => (findMethod fuel rest wanted value)

definition{
  name: orderOfPath,
  parameters: [path : (List String)],
  result: (Option TargetSyntax::Order),
  body: (match path
    | List.nil => (ctor Option.none [TargetSyntax::Order])
    | List.cons head tail => (match tail
      | List.nil => (ctor Option.none [TargetSyntax::Order])
      | List.cons last more => (match more
        | List.cons ignoredMore ignoredRest => (ctor Option.none [TargetSyntax::Order])
        | List.nil => (if (prim equal head "Ordering" : Bool) then (if (prim equal last "Less" : Bool) then (ctor Option.some [TargetSyntax::Order] (ctor TargetSyntax::Order.less)) else (if (prim equal last "Equal" : Bool) then (ctor Option.some [TargetSyntax::Order] (ctor TargetSyntax::Order.same)) else (if (prim equal last "Greater" : Bool) then (ctor Option.some [TargetSyntax::Order] (ctor TargetSyntax::Order.more)) else (ctor Option.none [TargetSyntax::Order])))) else (ctor Option.none [TargetSyntax::Order]))))),
  recursive_argument: path
}

definition{
  name: rtEval,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, environment : (List (String * TargetSyntax::Value)), expression : RuntimeSyntax::RtExpr],
  result: RtOutcome,
  body: (match fuel
    | Nat.zero => (ctor RtOutcome.exhausted)
    | Nat.succ remaining => (match expression
      | RuntimeSyntax::RtExpr.var name => (match (lookupVar environment name)
        | Option.none => (ctor RtOutcome.stuck)
        | Option.some found => (ctor RtOutcome.value found))
      | RuntimeSyntax::RtExpr.lit written => (match written
        | RuntimeSyntax::RtLit.unit => (ctor RtOutcome.value (ctor TargetSyntax::Value.unit))
        | RuntimeSyntax::RtLit.bool flag => (ctor RtOutcome.value (ctor TargetSyntax::Value.bool flag))
        | RuntimeSyntax::RtLit.nat number => (ctor RtOutcome.value (ctor TargetSyntax::Value.nat number)))
      | RuntimeSyntax::RtExpr.path segments => (match (orderOfPath segments)
        | Option.none => (ctor RtOutcome.stuck)
        | Option.some order => (ctor RtOutcome.value (ctor TargetSyntax::Value.ordering order)))
      | RuntimeSyntax::RtExpr.call segments arguments => (match segments
        | List.nil => (ctor RtOutcome.stuck)
        | List.cons callee rest => (match rest
          | List.cons ignoredSegment ignoredSegments => (ctor RtOutcome.stuck)
          | List.nil => (match (findFree (project crate items) callee)
            | Option.none => (ctor RtOutcome.stuck)
            | Option.some function => (match (rtEvalList remaining crate environment arguments)
              | RtOutcomes.values given => (rtApply remaining crate function given)
              | RtOutcomes.stuck => (ctor RtOutcome.stuck)
              | RtOutcomes.exhausted => (ctor RtOutcome.exhausted)))))
      | RuntimeSyntax::RtExpr.method receiver name arguments => (match (rtEval remaining crate environment receiver)
        | RtOutcome.value held => (match (rtEvalList remaining crate environment arguments)
          | RtOutcomes.values given => (if (prim equal name "cmp" : Bool) then (fromPrimitive (TargetSemantics::primitive (ctor TargetSyntax::Prim.compare) (cons held given))) else (match (findMethod remaining (project crate items) name held)
            | Option.none => (ctor RtOutcome.stuck)
            | Option.some function => (rtApply remaining crate function (cons held given))))
          | RtOutcomes.stuck => (ctor RtOutcome.stuck)
          | RtOutcomes.exhausted => (ctor RtOutcome.exhausted))
        | RtOutcome.stuck => (ctor RtOutcome.stuck)
        | RtOutcome.exhausted => (ctor RtOutcome.exhausted))
      | RuntimeSyntax::RtExpr.field inner position => (match (rtEval remaining crate environment inner)
        | RtOutcome.value held => (match (pairParts held)
          | Option.none => (ctor RtOutcome.stuck)
          | Option.some parts => (if (position =? 0) then (ctor RtOutcome.value (first parts)) else (if (position =? 1) then (ctor RtOutcome.value (second parts)) else (ctor RtOutcome.stuck))))
        | RtOutcome.stuck => (ctor RtOutcome.stuck)
        | RtOutcome.exhausted => (ctor RtOutcome.exhausted))
      | RuntimeSyntax::RtExpr.borrow inner => (rtEval remaining crate environment inner)
      | RuntimeSyntax::RtExpr.equals left right => (match (rtEval remaining crate environment left)
        | RtOutcome.value leftHeld => (match (rtEval remaining crate environment right)
          | RtOutcome.value rightHeld => (fromPrimitive (TargetSemantics::primitive (ctor TargetSyntax::Prim.equal) (cons leftHeld (cons rightHeld (nil TargetSyntax::Value)))))
          | RtOutcome.stuck => (ctor RtOutcome.stuck)
          | RtOutcome.exhausted => (ctor RtOutcome.exhausted))
        | RtOutcome.stuck => (ctor RtOutcome.stuck)
        | RtOutcome.exhausted => (ctor RtOutcome.exhausted))
      | RuntimeSyntax::RtExpr.matchOn scrutinee arms => (match (rtEval remaining crate environment scrutinee)
        | RtOutcome.value held => (rtEvalArms remaining crate environment held arms)
        | RtOutcome.stuck => (ctor RtOutcome.stuck)
        | RtOutcome.exhausted => (ctor RtOutcome.exhausted)))),
  axioms: [Classical.choice, Quot.sound, propext],
  mutual: RtMachine,
  recursive_argument: fuel
}

definition{
  name: rtEvalList,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, environment : (List (String * TargetSyntax::Value)), expressions : (List RuntimeSyntax::RtExpr)],
  result: RtOutcomes,
  body: (match fuel
    | Nat.zero => (ctor RtOutcomes.exhausted)
    | Nat.succ remaining => (match expressions
      | List.nil => (ctor RtOutcomes.values (nil TargetSyntax::Value))
      | List.cons head rest => (match (rtEval remaining crate environment head)
        | RtOutcome.value headValue => (match (rtEvalList remaining crate environment rest)
          | RtOutcomes.values restValues => (ctor RtOutcomes.values (cons headValue restValues))
          | RtOutcomes.stuck => (ctor RtOutcomes.stuck)
          | RtOutcomes.exhausted => (ctor RtOutcomes.exhausted))
        | RtOutcome.stuck => (ctor RtOutcomes.stuck)
        | RtOutcome.exhausted => (ctor RtOutcomes.exhausted)))),
  axioms: [Classical.choice, Quot.sound, propext],
  mutual: RtMachine,
  recursive_argument: fuel
}

definition{
  name: rtEvalArms,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, environment : (List (String * TargetSyntax::Value)), scrutinee : TargetSyntax::Value, arms : (List RuntimeSyntax::RtArm)],
  result: RtOutcome,
  body: (match fuel
    | Nat.zero => (ctor RtOutcome.exhausted)
    | Nat.succ remaining => (match arms
      | List.nil => (ctor RtOutcome.stuck)
      | List.cons arm rest => (match arm
        | RuntimeSyntax::RtArm.mk pattern body => (match pattern
          | RuntimeSyntax::RtPat.wild => (rtEval remaining crate environment body)
          | RuntimeSyntax::RtPat.bind name => (rtEval remaining crate (cons (name, scrutinee) environment) body)
          | RuntimeSyntax::RtPat.path segments => (match (orderOfPath segments)
            | Option.none => (rtEvalArms remaining crate environment scrutinee rest)
            | Option.some wanted => (match scrutinee
              | TargetSyntax::Value.ordering held => (if (TargetSemantics::sameOrder wanted held) then (rtEval remaining crate environment body) else (rtEvalArms remaining crate environment scrutinee rest))
              | TargetSyntax::Value.unit => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.bool ignoredBool => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.nat ignoredNat => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.int ignoredInt => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.u8 ignoredU8 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.u16 ignoredU16 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.u32 ignoredU32 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.u64 ignoredU64 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.i8 ignoredI8 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.i16 ignoredI16 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.i32 ignoredI32 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.i64 ignoredI64 => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.string ignoredString => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.bytes ignoredBytes => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.none => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.some ignoredSome => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.ok ignoredOk => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.error ignoredError => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.list ignoredList => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.pair ignoredFirst ignoredSecond => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.adt ignoredAdt ignoredFields => (rtEvalArms remaining crate environment scrutinee rest)
              | TargetSyntax::Value.closure ignoredClosure ignoredCaptures => (rtEvalArms remaining crate environment scrutinee rest))))))),
  axioms: [Classical.choice, Quot.sound, propext],
  mutual: RtMachine,
  recursive_argument: fuel
}

definition{
  name: rtApply,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, function : RuntimeSyntax::RtFn, values : (List TargetSyntax::Value)],
  result: RtOutcome,
  body: (match fuel
    | Nat.zero => (ctor RtOutcome.exhausted)
    | Nat.succ remaining => (match (if (project (project function signature) receiver) then (match values
      | List.nil => (ctor Option.none [(List (String * TargetSyntax::Value))])
      | List.cons held others => (bindParams (project (project function signature) parameters) others (cons ("self", held) (nil (String * TargetSyntax::Value))))) else (bindParams (project (project function signature) parameters) values (nil (String * TargetSyntax::Value))))
      | Option.none => (ctor RtOutcome.stuck)
      | Option.some bound => (rtEvalBlock remaining crate bound (project function body)))),
  axioms: [Classical.choice, Quot.sound, propext],
  mutual: RtMachine,
  recursive_argument: fuel
}

definition{
  name: rtEvalBlock,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, environment : (List (String * TargetSyntax::Value)), block : RuntimeSyntax::RtBlock],
  result: RtOutcome,
  body: (match fuel
    | Nat.zero => (ctor RtOutcome.exhausted)
    | Nat.succ remaining => (match block
      | RuntimeSyntax::RtBlock.mk statements tail => (match (rtEvalList remaining crate environment statements)
        | RtOutcomes.values ignoredValues => (rtEval remaining crate environment tail)
        | RtOutcomes.stuck => (ctor RtOutcome.stuck)
        | RtOutcomes.exhausted => (ctor RtOutcome.exhausted)))),
  axioms: [Classical.choice, Quot.sound, propext],
  mutual: RtMachine,
  recursive_argument: fuel
}

definition{
  name: rtRun,
  parameters: [fuel : Nat, crate : RuntimeSyntax::RtCrate, wanted : String, values : (List TargetSyntax::Value)],
  result: RtOutcome,
  body: (match (findFree (project crate items) wanted)
    | Option.none => (ctor RtOutcome.stuck)
    | Option.some function => (rtApply fuel crate function values)),
  axioms: [Classical.choice, Quot.sound, propext]
}
