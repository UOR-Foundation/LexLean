  recursive_argument: actions

definition{

    | List.cons action rest => (if (1 <? (countKind all (project action kind))) then (ctor Option.some [Rejection] (ctor Rejection.duplicateAction (project action kind))) else (checkDistinct rest all))),

definition{
  name: checkPerformed,
  parameters: [machine : Machine, wanted : ActionKind],
  result: (Option Rejection),
  body: (match (findAction (project machine actions) wanted)
    | Option.none => (ctor Option.some [Rejection] (ctor Rejection.unaccountedAction wanted))
    | Option.some charge => (match charge
      | Charge.steps => (ctor Option.none [Rejection])
      | Charge.constant ignored1 => (ctor Option.some [Rejection] (ctor Rejection.hiddenCost wanted))
      | Charge.free => (ctor Option.some [Rejection] (ctor Rejection.hiddenCost wanted))
      | Charge.undeclared => (ctor Option.some [Rejection] (ctor Rejection.hiddenCost wanted))))
}

definition{
  name: checkDeclared,
  parameters: [machine : Machine, action : Action],
  result: (Option Rejection),
  body: (let kind : ActionKind := (project action kind); (if (performed kind) then (ctor Option.none [Rejection]) else (if (preparation kind) then (match (project action charge)
    | Charge.steps => (ctor Option.some [Rejection] (ctor Rejection.hiddenCost kind))
    | Charge.constant amount => (if (amount =? 0) then (ctor Option.some [Rejection] (ctor Rejection.hiddenCost kind)) else (ctor Option.none [Rejection]))
    | Charge.free => (match (project machine boundary)
      | Boundary.complete => (ctor Option.some [Rejection] (ctor Rejection.unboundPreparation kind))
      | Boundary.preparedState => (if (preparedFor (project machine prepared) kind) then (ctor Option.none [Rejection]) else (ctor Option.some [Rejection] (ctor Rejection.unboundPreparation kind)))
      | Boundary.preparedPlan => (if (preparedFor (project machine prepared) kind) then (ctor Option.none [Rejection]) else (ctor Option.some [Rejection] (ctor Rejection.unboundPreparation kind))))
    | Charge.undeclared => (ctor Option.some [Rejection] (ctor Rejection.hiddenCost kind))) else (ctor Option.some [Rejection] (ctor Rejection.unrealizableAction kind)))))
}

definition{
  name: checkAllDeclared,

}

definition{
  name: selectorEq,
  parameters: [left : Selector, right : Selector],
  result: Bool,
  body: (match left
    | Selector.fixed plan => (match right
      | Selector.fixed other => (plan =? other)
      | Selector.dispatch ignoredThreshold ignoredSmall ignoredLarge => false)
    | Selector.dispatch threshold small large => (match right
      | Selector.fixed ignoredPlan => false
      | Selector.dispatch otherThreshold otherSmall otherLarge => ((threshold =? otherThreshold) && ((small =? otherSmall) && (large =? otherLarge)))))
}
