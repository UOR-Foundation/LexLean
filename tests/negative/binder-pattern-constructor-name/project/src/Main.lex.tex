\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: f,
  parameters: [o : (Option (Option Nat))],
  result: Nat,
  body: (match o
    | Option.none => 0
    | Option.some none => (match none | Option.none => 1 | Option.some n => n)),
  executable: true
}
\end{semanticmodule}
\end{lexlean}
