definition{
  name: itemsOf,
  parameters: [cells : RustMemory::Cells],
  result: (List TargetSyntax::Value),
  body: (match cells
    | RustMemory::Cells.CNil => (nil TargetSyntax::Value)
    | RustMemory::Cells.CCons head tail => (cons head (itemsOf tail))),
  recursive_argument: cells
}

structure{
  name: Replaced,
  type_parameters: [],
  parameters: [],
  fields: [{name: old, type: RustLists::Handle}, {name: now, type: RustLists::Handle}]
}

definition{
  name: firstOf,
  parameters: [cells : RustMemory::Cells, count : Nat],
  result: (List TargetSyntax::Value),
  body: (match count
    | Nat.zero => (nil TargetSyntax::Value)
    | Nat.succ less => (match cells
      | RustMemory::Cells.CNil => (nil TargetSyntax::Value)
      | RustMemory::Cells.CCons head tail => (cons head (firstOf tail less)))),
  recursive_argument: count
}

definition{
  name: takeItems,
  parameters: [items : (List TargetSyntax::Value), count : Nat],
  result: (List TargetSyntax::Value),
  body: (match count
    | Nat.zero => (nil TargetSyntax::Value)
    | Nat.succ less => (match items
      | List.nil => (nil TargetSyntax::Value)
      | List.cons head tail => (cons head (takeItems tail less)))),
  recursive_argument: count
}

definition{
  name: erasedPush,
  parameters: [items : (List TargetSyntax::Value), item : TargetSyntax::Value],
  result: (List TargetSyntax::Value),
  body: (prim append items (cons item (nil TargetSyntax::Value)) : (List TargetSyntax::Value))
}

definition{
  name: erasedTruncate,
  parameters: [items : (List TargetSyntax::Value), count : Nat],
  result: (List TargetSyntax::Value),
  body: (takeItems items count)
}

definition{
  name: erasedLength,
  parameters: [items : (List TargetSyntax::Value)],
  result: Nat,
  body: (prim length items : Nat)
}

definition{
  name: readBuf,
  parameters: [buf : RustMemory::Buf],
  result: (List TargetSyntax::Value),
  body: (firstOf (project buf slots) (project buf len))
}

definition{
  name: pushGrow,
  parameters: [buf : RustMemory::Buf, item : TargetSyntax::Value],
  result: RustMemory::Buf,
  body: (match (RustMemory::bufPush buf item)
    | Option.some got => got
    | Option.none => (RustLists::keepOr (RustMemory::bufPush (RustLists::growBuf buf ((project buf len) + 1) item) item) buf))
}

definition{
  name: bufNew,
  parameters: [capacity : Nat, fill : TargetSyntax::Value],
  result: RustMemory::Buf,
  body: (record RustMemory::Buf {slots := (RustMemory::blank capacity fill), len := 0})
}

definition{
  name: cloneHandle,
  parameters: [handle : RustLists::Handle],
  result: RustLists::Handle,
  body: handle
}

definition{
  name: replaceHandle,
  parameters: [place : RustLists::Handle, value : RustLists::Handle],
  result: Replaced,
  body: (record Replaced {old := place, now := value})
}

  proof: induction{
    branches: [
      {
        proof: simplify{definitions: [firstOf, itemsOf, RustMemory::append, RustMemory::size]},
        binders: [],
        constructor: CNil

        proof: simplify{
          definitions: [firstOf, hypothesis, itemsOf, RustMemory::append, RustMemory::size]

        constructor: CCons

    scrutinee: front
  },
  axioms: [propext]
}
