definition{
  name: hexChar,
  parameters: [n : Nat],
  result: Bytes,
  body: (Json::byteOf (if (n <? 10) then (48 + n) else (87 + n))),
  executable: true
}

definition{
  name: digestOn,
  parameters: [operand : Bytes, start : Nat, count : Nat],
  result: Bytes,
  body: (LexLeanStd.Text::sliceOr operand start count),
  executable: true
}

definition{
  name: mirrorStep,
  parameters: [count : Nat, digest : Bytes, state : (Nat * Bytes)],
  result: (Option (Nat * Bytes)),
  body: (if ((first state) <? count) then (ctor Option.some [(Nat * Bytes)] (((first state) + 1), (prim append (second state) (hexChar (prim subtract 15 (Json::hexValue (Label::byteNat digest (first state))) : Nat)) : Bytes))) else (ctor Option.none [(Nat * Bytes)])),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: mirror,
  parameters: [operand : Bytes, start : Nat, count : Nat],
  result: Bytes,
  body: (let digest : Bytes := (digestOn operand start count); (second (first (prim iterate_until (fun [count, digest] (state : (Nat * Bytes)) => (mirrorStep count digest state)) (count + 1) (0, 0x) : ((Nat * Bytes) * Bool))))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: lower,
  parameters: [left : Bytes, right : Bytes],
  result: Bytes,
  body: (if (Json::lessKey right left) then right else left),
  axioms: [propext],
  executable: true
}

definition{
  name: upper,
  parameters: [left : Bytes, right : Bytes],
  result: Bytes,
  body: (if (Json::lessKey right left) then left else right),
  axioms: [propext],
  executable: true
}

definition{
  name: sortedQuarters,
  parameters: [operand : Bytes, start : Nat, width : Nat],
  result: Bytes,
  body: (let a0 : Bytes := (LexLeanStd.Text::sliceOr operand start width); (let b0 : Bytes := (LexLeanStd.Text::sliceOr operand (start + width) width); (let c0 : Bytes := (LexLeanStd.Text::sliceOr operand (start + (prim multiply 2 width : Nat)) width); (let d0 : Bytes := (LexLeanStd.Text::sliceOr operand (start + (prim multiply 3 width : Nat)) width); (let a1 : Bytes := (lower a0 b0); (let b1 : Bytes := (upper a0 b0); (let c1 : Bytes := (lower c0 d0); (let d1 : Bytes := (upper c0 d0); (let a2 : Bytes := (lower a1 c1); (let c2 : Bytes := (upper a1 c1); (let b2 : Bytes := (lower b1 d1); (let d2 : Bytes := (upper b1 d1); (let b3 : Bytes := (lower b2 c2); (let c3 : Bytes := (upper b2 c2); (prim append a2 (prim append b3 (prim append c3 d2 : Bytes) : Bytes) : Bytes))))))))))))))),
  axioms: [propext],
  executable: true
}

definition{
  name: degreeTag,
  parameters: [operand : Bytes, start : Nat],
  result: Bytes,
  body: (let high : Nat := (Json::hexValue (Label::byteNat operand start)); (let low : Nat := (Json::hexValue (Label::byteNat operand (start + 1))); (if ((prim remainder ((prim multiply high 16 : Nat) + low) 9 0 : Nat) =? 8) then 0x06 else 0x05))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: composeG2On,

  axioms: [Quot.sound, propext],
  executable: true,
