definition{
  name: nodeTerm,
  parameters: [node : CoreJson::Node, terms : (List Term)],
  result: Term,
  body: (match node
    | CoreJson::Node.NBVar position => (ctor Term.Atom (ctor Tdoc.TVar position))
    | CoreJson::Node.NSort level => (ctor Term.Prefix (tt (cat (lit "Sort ") (levelText level))))
    | CoreJson::Node.NConst label levels => (ctor Term.Atom (ctor Tdoc.TConst label (levelsText levels)))
    | CoreJson::Node.NApp function applied => (let head : Head := (headOf (termAt terms 0)); (ctor Term.App (project head head) (cons (asArgument (termAt terms 1)) (project head args))))
    | CoreJson::Node.NLambda label info bound body => (ctor Term.Low (ctor Tdoc.TGroup (tseq (cons (tt (lit "fun ")) (cons (binderDoc info label (termAt terms 0)) (cons (tt (lit " =>")) (cons (ctor Tdoc.TLine) (cons (ctor Tdoc.TBind label (whole (termAt terms 1))) (nil Tdoc)))))))))
    | CoreJson::Node.NForall label info bound body => (ctor Term.Low (ctor Tdoc.TGroup (tseq (cons (if (both ((prim length label : Nat) =? 0) (isExplicit info)) then (asDomain (termAt terms 0)) else (binderDoc info label (termAt terms 0))) (cons (tt (lit " ->")) (cons (ctor Tdoc.TLine) (cons (ctor Tdoc.TBind label (whole (termAt terms 1))) (nil Tdoc))))))))
    | CoreJson::Node.NLet label bound value rest dependent => (ctor Term.Low (ctor Tdoc.TGroup (tseq (cons (tt (cat (cat (if dependent then (lit "have ") else (lit "let ")) (binderName label)) (lit " : "))) (cons (whole (termAt terms 0)) (cons (tt (lit " := ")) (cons (whole (termAt terms 1)) (cons (tt (lit ";")) (cons (ctor Tdoc.TLine) (cons (ctor Tdoc.TBind label (whole (termAt terms 2))) (nil Tdoc)))))))))))
    | CoreJson::Node.NNat text => (ctor Term.Atom (tt (if (canonicalNatural text) then text else (cat (lit "nat ") (SurfaceLex::quoteText text)))))
    | CoreJson::Node.NString scalars => (ctor Term.Atom (tt (SurfaceLex::quoteText (prim list_fold (fun (out : Bytes) (scalar : Nat) => (cat out (utf8 scalar))) (lit "") scalars : Bytes))))
    | CoreJson::Node.NMdata entries inner => (ctor Term.Prefix (ctor Tdoc.TGroup (tseq (cons (tt (cat (cat (lit "annot [") (joined (LexLeanStd.Order::reverseList [Bytes] (prim list_fold (fun (out : (List Bytes)) (entry : CoreJson::Entry) => (cons (metaText entry) out)) (nil Bytes) entries : (List Bytes))) (lit ", "))) (lit "]"))) (cons (ctor Tdoc.TLine) (cons (asArgument (termAt terms 0)) (nil Tdoc)))))))
    | CoreJson::Node.NProj label structure position inner => (ctor Term.Prefix (ctor Tdoc.TGroup (tseq (cons (tt (cat (cat (cat (cat (cat (lit "proj ") (spell structure)) (lit " ")) (spell label)) (lit " ")) (num position))) (cons (ctor Tdoc.TLine) (cons (asArgument (termAt terms 0)) (nil Tdoc)))))))),
  axioms: [Quot.sound, propext],
  executable: true
}

    | CoreJson::Node.NForall label info bound body => (cons bound (cons body (nil Nat)))
    | CoreJson::Node.NLet label bound value rest dependent => (cons bound (cons value (cons rest (nil Nat))))
    | CoreJson::Node.NMdata entries inner => (cons inner (nil Nat))
    | CoreJson::Node.NProj label structure position inner => (cons inner (nil Nat))
    | CoreJson::Node.NBVar position => (nil Nat)
    | CoreJson::Node.NSort level => (nil Nat)
    | CoreJson::Node.NConst label levels => (nil Nat)
    | CoreJson::Node.NNat text => (nil Nat)
    | CoreJson::Node.NString scalars => (nil Nat)),
  executable: true

      | LexLeanStd.Json::Json.JItem element rest => (ctor Option.none [Nat])

definition{
  name: lookStep,
  parameters: [key : Bytes, state : (LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))],
  result: (Option (LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))),
  body: (match (second state)
    | Option.some hit => (ctor Option.none [(LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))])
    | Option.none => (match (first state)
      | LexLeanStd.Json::Json.JMember name value rest => (ctor Option.some [(LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))] (if (same name key) then ((ctor LexLeanStd.Json::Json.JNil), (ctor Option.some [LexLeanStd.Json::Json] value)) else (rest, (ctor Option.none [LexLeanStd.Json::Json]))))
      | LexLeanStd.Json::Json.JBool flag => (ctor Option.none [(LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))])
      | LexLeanStd.Json::Json.JInt negative magnitude => (ctor Option.none [(LexLeanStd.Json::Json * (Option LexLeanStd.Json::Json))])
