\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: depth,
  parameters: [values : (List Nat)],
  result: Nat,
  body: (match values
    | List.nil => 0
    | List.cons head tail => (apply (fun (rest : (List Nat)) => (depth rest)) tail)),
  recursive_argument: values
}
\end{semanticmodule}
\end{lexlean}
