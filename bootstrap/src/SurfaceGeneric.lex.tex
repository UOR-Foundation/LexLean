definition{
  name: closeTop,
  parameters: [st : Rd],
  result: Rd,
  body: (match (project st stack)
    | List.nil => (refuse st 0 (lit "nothing to close"))
    | List.cons top rest => (match top
      | Frame.FrameArr items => (record Rd {pos := ((project st pos) + 1), depth := (prim subtract (project st depth) 1 : Nat), mode := 3, stack := rest, value := (ctor LexLeanStd.Json::Json.JArr (itemChain items)), error := (ctor Option.none [SurfaceLex::Failure])})
      | Frame.FrameObj pending offset members => (record Rd {pos := ((project st pos) + 1), depth := (prim subtract (project st depth) 1 : Nat), mode := 3, stack := rest, value := (ctor LexLeanStd.Json::Json.JObj (memberChain (LexLeanStd.Order::sortByKey [LexLeanStd.Json::Json] members))), error := (ctor Option.none [SurfaceLex::Failure])}))),
  axioms: [propext],
  executable: true
}

  executable: true

definition{
  name: completeValue,
  parameters: [toks : (List SurfaceLex::Spanned), st : Rd],
  result: Rd,
  body: (match (project st stack)
    | List.nil => (record Rd {pos := (project st pos), depth := (project st depth), mode := 9, stack := (nil Frame), value := (project st value), error := (ctor Option.none [SurfaceLex::Failure])})
    | List.cons top rest => (match top
      | Frame.FrameArr items => (atComma toks st (cons (ctor Frame.FrameArr (cons (project st value) items)) rest) 1 4)
      | Frame.FrameObj key offset members => (if (hasKey members key) then (refuse st offset (cat (cat (lit "the member `") key) (lit "` is written twice"))) else (atComma toks st (cons (ctor Frame.FrameObj 0x 0 (cons (key, (project st value)) members)) rest) 2 5)))),
  executable: true
}

  result: Rd,
  body: (let pos : Nat := (project st pos); (let word : Bytes := (textAt toks pos); (if (prim equal word 0x74727565 : Bool) then (valueDone st (pos + 1) (ctor LexLeanStd.Json::Json.JBool true)) else (if (prim equal word 0x66616c7365 : Bool) then (valueDone st (pos + 1) (ctor LexLeanStd.Json::Json.JBool false)) else (if (isInteger word) then (match (intOf word)
    | Option.some number => (valueDone st (pos + 1) number)
    | Option.none => (refuse st start (cat (cat (lit "`") word) (lit "` does not fit a 64-bit integer")))) else (if (isBare word) then (if ((codeAt toks (pos + 1)) =? 0) then (valuePush st (pos + 2) (ctor Frame.FrameObj 0x 0 (cons ((lit "kind"), (ctor LexLeanStd.Json::Json.JStr word)) (nil (Bytes * LexLeanStd.Json::Json)))) 2) else (valueDone st (pos + 1) (ctor LexLeanStd.Json::Json.JStr word))) else (refuse st start (cat (cat (lit "`") word) (lit "` is not a name; quote it to write a string"))))))))),
