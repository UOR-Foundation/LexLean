  name: pairUp,
  parameters: [pieces : (List Bytes)],
  result: (Option (List Bytes)),
  body: (match pieces
    | List.nil => (ctor Option.none [(List Bytes)])
    | List.cons head rest => (match rest
      | List.nil => (ctor Option.none [(List Bytes)])
      | List.cons next later => (let folded : ((Option Bytes) * (List Bytes)) := (prim list_fold (fun (state : ((Option Bytes) * (List Bytes))) (piece : Bytes) => (pairStep state piece)) ((ctor Option.none [Bytes]), (nil Bytes)) pieces : ((Option Bytes) * (List Bytes))); (match (first folded)

        | Option.none => (ctor Option.some [(List Bytes)] (Text::reverse (second folded))))))),
  executable: true
}

  name: joinAll,

  executable: true
