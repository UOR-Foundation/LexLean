\begin{lexlean}{Word}

\importmodule{LexLeanStd.Sha256}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: shiftRight,
  parameters: [value : UInt64, amount : UInt32],
  result: UInt64,
  body: (match (prim shift_right value amount : (Option UInt64))
    | Option.some shifted => shifted
    | Option.none => (0 : UInt64)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: shiftLeft,
  parameters: [value : UInt64, amount : UInt32],
  result: UInt64,
  body: (match (prim shift_left value amount : (Option UInt64))
    | Option.some shifted => shifted
    | Option.none => (0 : UInt64)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: rotateRight,
  parameters: [value : UInt64, amount : UInt32, complement : UInt32],
  result: UInt64,
  body: (prim bit_or (shiftRight value amount) (shiftLeft value complement) : UInt64),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: rotateLeft,
  parameters: [value : UInt64, amount : UInt32, complement : UInt32],
  result: UInt64,
  body: (prim bit_or (shiftLeft value amount) (shiftRight value complement) : UInt64),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: sum,
  parameters: [left : UInt64, right : UInt64],
  result: UInt64,
  body: (match (prim checked_add left right : (Option UInt64))
    | Option.some total => total
    | Option.none => (0 : UInt64)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: add,
  parameters: [left : UInt64, right : UInt64],
  result: UInt64,
  body: (let mask : UInt64 := (4294967295 : UInt64); (let low : UInt64 := (sum (prim bit_and left mask : UInt64) (prim bit_and right mask : UInt64)); (let high : UInt64 := (sum (sum (shiftRight left (32 : UInt32)) (shiftRight right (32 : UInt32))) (shiftRight low (32 : UInt32))); (prim bit_or (shiftLeft (prim bit_and high mask : UInt64) (32 : UInt32)) (prim bit_and low mask : UInt64) : UInt64)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: nibble,
  parameters: [value : UInt64, shift : UInt32],
  result: Bytes,
  body: (LexLeanStd.Sha256::hexDigit (LexLeanStd.Sha256::narrow (shiftRight value shift))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: byteWord,
  parameters: [message : Bytes, position : Nat],
  result: UInt64,
  body: (match (prim index message position : (Option UInt8))
    | Option.some byte => (match (prim checked_convert byte : (Option UInt64))
      | Option.some wide => wide
      | Option.none => (0 : UInt64))
    | Option.none => (0 : UInt64)),
  axioms: [Quot.sound, propext],
  executable: true

  axioms: [Quot.sound, propext],
  executable: true
