definition{
  name: collectTerm,
  parameters: [program : CalculusCheck::Program, state : Collect, expr : CalculusCheck::Expr, scope : (List (Nat * CalculusCheck::Ty)), subject : CalculusCheck::Ty],
  result: Collect,
  body: (match expr
    | CalculusCheck::Expr.EValue ty value => (record Collect [] {known := (register (project state known) ty), closures := (project state closures)})
    | CalculusCheck::Expr.EVar name => state
    | CalculusCheck::Expr.ELet name ty bound body => (let registered : Collect := (record Collect [] {known := (register (project state known) ty), closures := (project state closures)}); (let inner : Collect := (collectTerm program registered bound scope (ctor CalculusCheck::Ty.TNil)); (collectTerm program inner body (CalculusCheck::extendScope name ty scope) (ctor CalculusCheck::Ty.TNil))))
    | CalculusCheck::Expr.ECond condition thenBranch elseBranch => (let one : Collect := (collectTerm program state condition scope (ctor CalculusCheck::Ty.TNil)); (let two : Collect := (collectTerm program one thenBranch scope (ctor CalculusCheck::Ty.TNil)); (collectTerm program two elseBranch scope (ctor CalculusCheck::Ty.TNil))))
    | CalculusCheck::Expr.EMatch ty scrutinee arms => (let registered : Collect := (record Collect [] {known := (register (project state known) ty), closures := (project state closures)}); (let inner : Collect := (collectTerm program registered scrutinee scope (ctor CalculusCheck::Ty.TNil)); (collectTerm program inner arms scope (inferTy program scrutinee scope))))
    | CalculusCheck::Expr.EBuild shape ty operands => (let registered : Collect := (record Collect [] {known := (register (project state known) ty), closures := (project state closures)}); (collectTerm program registered operands scope (ctor CalculusCheck::Ty.TNil)))
    | CalculusCheck::Expr.ECall function operands => (collectTerm program state operands scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.EClosure function captures => (let inner : Collect := (collectTerm program state captures scope (ctor CalculusCheck::Ty.TNil)); (let declared : CalculusCheck::Ty := (CalculusCheck::functionTypesOr program function); (let count : Nat := (CalculusCheck::exprChainLength captures); (let closureType : CalculusCheck::Ty := (ctor CalculusCheck::Ty.TFn (tyDrop declared count) (CalculusCheck::functionResult program function)); (let registered : Collect := (record Collect [] {known := (register (project inner known) closureType), closures := (project inner closures)}); (record Collect [] {known := (project registered known), closures := (closAdd (project registered closures) closureType function (tyTake declared count))}))))))
    | CalculusCheck::Expr.EApply target operands => (let inner : Collect := (collectTerm program state target scope (ctor CalculusCheck::Ty.TNil)); (collectTerm program inner operands scope (ctor CalculusCheck::Ty.TNil)))
    | CalculusCheck::Expr.EPrim operation operands => (collectTerm program state operands scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.EFirst value => (collectTerm program state value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.ESecond value => (collectTerm program state value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.EField value index => (collectTerm program state value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.ENil => state
    | CalculusCheck::Expr.ECons head tail => (let inner : Collect := (collectTerm program state head scope subject); (collectTerm program inner tail scope subject))
    | CalculusCheck::Expr.EArm shape binders body => (collectTerm program state body (CalculusCheck::bindAll binders (armFields program shape subject) scope) (ctor CalculusCheck::Ty.TNil))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: expr
}

definition{
  name: collectFunctions,
  parameters: [program : CalculusCheck::Program, state : Collect, bodies : CalculusCheck::Expr, index : Nat, dead : (List Bool)],
  result: Collect,
  body: (match bodies
    | CalculusCheck::Expr.EValue d0 d1 => state
    | CalculusCheck::Expr.EVar d0 => state
    | CalculusCheck::Expr.ELet d0 d1 d2 d3 => state
    | CalculusCheck::Expr.ECond d0 d1 d2 => state
    | CalculusCheck::Expr.EMatch d0 d1 d2 => state
    | CalculusCheck::Expr.EBuild d0 d1 d2 => state
    | CalculusCheck::Expr.ECall d0 d1 => state
    | CalculusCheck::Expr.EClosure d0 d1 => state
    | CalculusCheck::Expr.EApply d0 d1 => state
    | CalculusCheck::Expr.EPrim d0 d1 => state
    | CalculusCheck::Expr.EFirst d0 => state
    | CalculusCheck::Expr.ESecond d0 => state
    | CalculusCheck::Expr.EField d0 d1 => state
    | CalculusCheck::Expr.ENil => state
    | CalculusCheck::Expr.ECons body rest => (let live : Bool := (match (CalculusCheck::listAt [Bool] dead index)
      | Option.some flag => (not flag)
      | Option.none => true); (let inner : Collect := (if live then (collectTerm program state body (CalculusCheck::bindAll (CalculusCheck::functionNames program index) (CalculusCheck::functionTypesOr program index) (nil (Nat * CalculusCheck::Ty))) (ctor CalculusCheck::Ty.TNil)) else state); (collectFunctions program inner rest (index + 1) dead)))
    | CalculusCheck::Expr.EArm d0 d1 d2 => state),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: bodies
}

definition{
  name: reachTerm,
  parameters: [program : CalculusCheck::Program, found : Reach, expr : CalculusCheck::Expr, scope : (List (Nat * CalculusCheck::Ty)), subject : CalculusCheck::Ty],
  result: Reach,
  body: (match expr
    | CalculusCheck::Expr.EValue ty value => found
    | CalculusCheck::Expr.EVar name => found
    | CalculusCheck::Expr.ELet name ty bound body => (let inner : Reach := (reachTerm program found bound scope (ctor CalculusCheck::Ty.TNil)); (reachTerm program inner body (CalculusCheck::extendScope name ty scope) (ctor CalculusCheck::Ty.TNil)))
    | CalculusCheck::Expr.ECond condition thenBranch elseBranch => (let one : Reach := (reachTerm program found condition scope (ctor CalculusCheck::Ty.TNil)); (let two : Reach := (reachTerm program one thenBranch scope (ctor CalculusCheck::Ty.TNil)); (reachTerm program two elseBranch scope (ctor CalculusCheck::Ty.TNil))))
    | CalculusCheck::Expr.EMatch ty scrutinee arms => (let inner : Reach := (reachTerm program found scrutinee scope (ctor CalculusCheck::Ty.TNil)); (reachTerm program inner arms scope (inferTy program scrutinee scope)))
    | CalculusCheck::Expr.EBuild shape ty operands => (let inner : Reach := (reachTerm program found operands scope (ctor CalculusCheck::Ty.TNil)); (match shape
      | CalculusCheck::Shape.SSucc => (record Reach [] {own := true, calls := (project inner calls), applies := (project inner applies)})
      | CalculusCheck::Shape.SNone => inner
      | CalculusCheck::Shape.SSome => inner
      | CalculusCheck::Shape.SOk => inner
      | CalculusCheck::Shape.SError => inner
      | CalculusCheck::Shape.SNil => inner
      | CalculusCheck::Shape.SCons => inner
      | CalculusCheck::Shape.SZero => inner
      | CalculusCheck::Shape.SPair => inner
      | CalculusCheck::Shape.STrue => inner
      | CalculusCheck::Shape.SFalse => inner
      | CalculusCheck::Shape.SUnit => inner
      | CalculusCheck::Shape.SLt => inner
      | CalculusCheck::Shape.SEq => inner
      | CalculusCheck::Shape.SGt => inner
      | CalculusCheck::Shape.SAdt d0 => inner))
    | CalculusCheck::Expr.ECall function operands => (let inner : Reach := (reachTerm program found operands scope (ctor CalculusCheck::Ty.TNil)); (record Reach [] {own := (project inner own), calls := (if (natMember (project inner calls) function) then (project inner calls) else (cons function (project inner calls))), applies := (project inner applies)}))
    | CalculusCheck::Expr.EClosure function captures => (reachTerm program found captures scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.EApply target operands => (let one : Reach := (reachTerm program found target scope (ctor CalculusCheck::Ty.TNil)); (let two : Reach := (reachTerm program one operands scope (ctor CalculusCheck::Ty.TNil)); (record Reach [] {own := (project two own), calls := (project two calls), applies := (tyAddNew (project two applies) (inferTy program target scope))})))
    | CalculusCheck::Expr.EPrim operation operands => (let inner : Reach := (reachTerm program found operands scope (ctor CalculusCheck::Ty.TNil)); (record Reach [] {own := (if (project inner own) then true else (primFails operation)), calls := (project inner calls), applies := (project inner applies)}))
    | CalculusCheck::Expr.EFirst value => (reachTerm program found value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.ESecond value => (reachTerm program found value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.EField value index => (reachTerm program found value scope (ctor CalculusCheck::Ty.TNil))
    | CalculusCheck::Expr.ENil => found
    | CalculusCheck::Expr.ECons head tail => (let inner : Reach := (reachTerm program found head scope subject); (reachTerm program inner tail scope subject))
    | CalculusCheck::Expr.EArm shape binders body => (reachTerm program found body (CalculusCheck::bindAll binders (armFields program shape subject) scope) (ctor CalculusCheck::Ty.TNil))),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: expr
}

definition{
  name: reachFunctions,
  parameters: [program : CalculusCheck::Program, bodies : CalculusCheck::Expr, index : Nat],
  result: Facts,
  body: (match bodies
    | CalculusCheck::Expr.EValue d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EVar d0 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ELet d0 d1 d2 d3 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ECond d0 d1 d2 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EMatch d0 d1 d2 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EBuild d0 d1 d2 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ECall d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EClosure d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EApply d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EPrim d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EFirst d0 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ESecond d0 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.EField d0 d1 => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ENil => (ctor Facts.FactsNil)
    | CalculusCheck::Expr.ECons body rest => (let found : Reach := (reachTerm program (record Reach [] {own := false, calls := (nil Nat), applies := (ctor CalculusCheck::Ty.TNil)}) body (CalculusCheck::bindAll (CalculusCheck::functionNames program index) (CalculusCheck::functionTypesOr program index) (nil (Nat * CalculusCheck::Ty))) (ctor CalculusCheck::Ty.TNil)); (ctor Facts.FactsCons (project found own) (project found calls) (project found applies) (reachFunctions program rest (index + 1))))
    | CalculusCheck::Expr.EArm d0 d1 d2 => (ctor Facts.FactsNil)),
  axioms: [Quot.sound, propext],
  executable: true,
  recursive_argument: bodies
}

definition{
  name: inPlace,
  parameters: [known : CalculusCheck::Ty, held : CalculusCheck::Ty, ty : CalculusCheck::Ty],
  result: CalculusCheck::Ty,
  body: (match ty
    | CalculusCheck::Ty.TUnit => held
    | CalculusCheck::Ty.TBool => held
    | CalculusCheck::Ty.TNat => held
    | CalculusCheck::Ty.TInt => held
    | CalculusCheck::Ty.TFixed d0 => held
    | CalculusCheck::Ty.TString => held
    | CalculusCheck::Ty.TBytes => held
    | CalculusCheck::Ty.TOrdering => held
    | CalculusCheck::Ty.TOption value => (inPlace known held value)
    | CalculusCheck::Ty.TResult ok error => (inPlace known (inPlace known held ok) error)
    | CalculusCheck::Ty.TList d0 => held
    | CalculusCheck::Ty.TPair left right => (inPlace known (inPlace known held left) right)
    | CalculusCheck::Ty.TAdt index => (tyAddNew held ty)
    | CalculusCheck::Ty.TFn parameters result => (if (tyMember known ty) then (tyAddNew held ty) else held)
    | CalculusCheck::Ty.TNil => held
    | CalculusCheck::Ty.TCons d0 d1 => held),
  executable: true,
  recursive_argument: ty
}

definition{
  name: edgeHas,
  parameters: [edges : Edges, from : CalculusCheck::Ty, to : CalculusCheck::Ty],
  result: Bool,
  body: (match edges
    | Edges.EdgeNil => false
    | Edges.EdgeCons left right rest => (if (CalculusCheck::tyEq left from) then (if (CalculusCheck::tyEq right to) then true else (edgeHas rest from to)) else (edgeHas rest from to))),
  executable: true,
  recursive_argument: edges
}

definition{
  name: edgeCount,
  parameters: [edges : Edges],
  result: Nat,
  body: (match edges
    | Edges.EdgeNil => 0
    | Edges.EdgeCons left right rest => (1 + (edgeCount rest))),
  executable: true,
  recursive_argument: edges
}

definition{
  name: edgeAdd,
  parameters: [edges : Edges, from : CalculusCheck::Ty, to : CalculusCheck::Ty],
  result: Edges,
  body: (if (edgeHas edges from to) then edges else (ctor Edges.EdgeCons from to edges)),
  executable: true
}

definition{
  name: edgesToHeld,
  parameters: [owner : CalculusCheck::Ty, held : CalculusCheck::Ty, edges : Edges],
  result: Edges,
  body: (match held
    | CalculusCheck::Ty.TUnit => edges
    | CalculusCheck::Ty.TBool => edges
    | CalculusCheck::Ty.TNat => edges
