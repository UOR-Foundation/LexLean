\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\importmodule{Combinators}
\title{Natural number addition}
\begin{semanticmodule}
definition{
  name: addAll,
  parameters: [offset : Nat, values : (List Nat)],
  result: (List Nat),
  body: (Combinators::mapList [Nat, Nat] (fun [offset] (value : Nat) => (value + offset)) values),
  executable: true
}

definition{
  name: total,
  parameters: [values : (List Nat)],
  result: Nat,
  body: (Combinators::foldList [Nat, Nat] (fun (sum : Nat) (value : Nat) => (sum + value)) 0 values),
  executable: true
}

definition{
  name: bumpAll,
  parameters: [values : (List Nat)],
  result: (List Nat),
  body: (Combinators::mapList [Nat, Nat] (ref Combinators::increment) values),
  executable: true
}

definition{
  name: addThenDouble,
  parameters: [],
  result: (Fn (Nat) Nat),
  body: (Combinators::compose [Nat, Nat, Nat] (fun (value : Nat) => (value + value)) (ref Combinators::increment))
}

definition{
  name: evaluator,
  parameters: [],
  result: Combinators::Visitor,
  body: (record Combinators::Visitor {onLiteral := (fun (value : Nat) => value), onPlus := (fun (left : Nat) (right : Nat) => (left + right)), onTwice := (fun (value : Nat) => (value + value))})
}

theorem{
  name: add_all,
  parameters: [],
  statement: ((addAll 10 (cons 1 (cons 2 (cons 3 (nil Nat))))) = (cons 11 (cons 12 (cons 13 (nil Nat))))),
  proof: reflexivity{}
}

theorem{
  name: total_sum,
  parameters: [],
  statement: ((total (cons 1 (cons 2 (cons 3 (cons 4 (nil Nat)))))) = 10),
  proof: reflexivity{}
}

theorem{
  name: bump_all,
  parameters: [],
  statement: ((bumpAll (cons 0 (cons 4 (nil Nat)))) = (cons 1 (cons 5 (nil Nat)))),
  proof: reflexivity{}
}

theorem{
  name: composed,
  parameters: [],
  statement: ((apply (addThenDouble) 4) = 10),
  proof: reflexivity{}
}

theorem{
  name: visitor_agrees,
  parameters: [],
  statement: ((Combinators::visit (evaluator) (ctor Combinators::Arith.plus (ctor Combinators::Arith.literal 2) (ctor Combinators::Arith.twice (ctor Combinators::Arith.literal 3)))) = (Combinators::evaluate (ctor Combinators::Arith.plus (ctor Combinators::Arith.literal 2) (ctor Combinators::Arith.twice (ctor Combinators::Arith.literal 3))))),
  proof: reflexivity{}
}

theorem{
  name: map_identity,
  type_parameters: [Input],
  parameters: [values : (List 'Input)],
  statement: ((Combinators::mapList ['Input, 'Input] (fun (value : 'Input) => value) values) = values),
  proof: induction{
    branches: [
      {proof: simplify{definitions: [Combinators::mapList]}, binders: [], constructor: nil},
      {
        proof: simplify{definitions: [hypothesis, Combinators::mapList]},
        binders: [head, tail, hypothesis],
        constructor: cons
      }
    ],
    scrutinee: values
  },
  axioms: [propext]
}

theorem{
  name: map_identity_numbers,
  parameters: [],
  statement: ((Combinators::mapList [Nat, Nat] (fun (value : Nat) => value) (cons 7 (cons 8 (nil Nat)))) = (cons 7 (cons 8 (nil Nat)))),
  proof: apply{
    arguments: [(cons 7 (cons 8 (nil Nat)))],
    theorem: map_identity,
    type_arguments: [Nat]
  },
  axioms: [propext]
}
\end{semanticmodule}
\end{lexlean}
