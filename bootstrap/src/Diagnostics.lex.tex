definition{

definition{

definition{

  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: labelText,
  parameters: [label : Label],
  result: Bytes,
  body: (objectOf (cons (field (lit "message") (jsonString (project label message))) (cons (field (lit "span") (spanText (project label span))) (nil Bytes)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: noteText,
  parameters: [note : Note],
  result: Bytes,
  body: (match (project note span)
    | Option.some span => (objectOf (cons (field (lit "message") (jsonString (project note message))) (cons (field (lit "span") (spanText span)) (nil Bytes))))
    | Option.none => (objectOf (cons (field (lit "message") (jsonString (project note message))) (nil Bytes)))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: strings,
  parameters: [items : (List Bytes)],
  result: Bytes,
  body: (arrayOf (prim list_fold (fun (acc : (List Bytes)) (item : Bytes) => (cons (jsonString item) acc)) (nil Bytes) (LexLeanStd.Order::reverseList [Bytes] items) : (List Bytes))),
  axioms: [Quot.sound, propext],
  executable: true

definition{

  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: notesText,
  parameters: [items : (List Note)],
  result: Bytes,
  body: (arrayOf (prim list_fold (fun (acc : (List Bytes)) (item : Note) => (cons (noteText item) acc)) (nil Bytes) (LexLeanStd.Order::reverseList [Note] items) : (List Bytes))),
  axioms: [Quot.sound, propext],
  executable: true

definition{
  name: diagnosticText,
  parameters: [given : Diagnostic],
  result: Bytes,
  body: (let tail : (List Bytes) := (cons (field (lit "severity") (jsonString (lit "error"))) (cons (field (lit "spec") (jsonString (lit "lexlean/diagnostic/1"))) (nil Bytes))); (let located : (List Bytes) := (match (project given primary)
    | Option.some span => (cons (field (lit "primary") (spanText span)) tail)
    | Option.none => tail); (objectOf (cons (field (lit "causes") (strings (project given causes))) (cons (field (lit "code") (jsonString (project given code))) (cons (field (lit "help") (strings (project given help))) (cons (field (lit "labels") (labelsText (project given labels))) (cons (field (lit "message") (jsonString (project given message))) (cons (field (lit "notes") (notesText (project given notes))) located))))))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}

definition{
  name: byteText,
  parameters: [value : Nat],
  result: Bytes,
  body: (LexLeanStd.Numeral::pick (LexLeanStd.Numeral::byteValues) value),
  executable: true
}

definition{
  name: beStep,
  parameters: [state : (Nat * (Nat * Bytes))],
  result: (Option (Nat * (Nat * Bytes))),
  body: (if ((first state) =? 0) then (ctor Option.none [(Nat * (Nat * Bytes))]) else (ctor Option.some [(Nat * (Nat * Bytes))] ((prim subtract (first state) 1 : Nat), ((prim quotient (first (second state)) 256 0 : Nat), (cat (byteText (prim remainder (first (second state)) 256 0 : Nat)) (second (second state))))))),
  executable: true
}

definition{
  name: bigEndian,
  parameters: [value : Nat],
  result: Bytes,
  body: (second (second (first (prim iterate_until (fun (state : (Nat * (Nat * Bytes))) => (beStep state)) 9 (8, (value, 0x)) : ((Nat * (Nat * Bytes)) * Bool))))),
  executable: true
}

definition{
  name: orderStep,
  parameters: [path : Bytes, n : Nat, state : (Nat * Bytes)],
  result: (Option (Nat * Bytes)),
  body: (let position : Nat := (first state); (if (position <? n) then (match (LexLeanStd.Text::byteAt path position)
    | Option.some byte => (if (prim equal byte (0 : UInt8) : Bool) then (ctor Option.some [(Nat * Bytes)] ((position + 1), (cat (second state) 0x0001))) else (ctor Option.some [(Nat * Bytes)] ((position + 1), (cat (second state) (byteText (prim to_nat byte : Nat))))))
    | Option.none => (ctor Option.none [(Nat * Bytes)])) else (ctor Option.none [(Nat * Bytes)]))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: orderedPath,
  parameters: [path : Bytes],
  result: Bytes,
  body: (let n : Nat := (prim length path : Nat); (cat (second (first (prim iterate_until (fun [n, path] (state : (Nat * Bytes)) => (orderStep path n state)) (n + 1) (0, 0x) : ((Nat * Bytes) * Bool)))) 0x0000)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: sortKey,
  parameters: [given : Diagnostic],
  result: Bytes,
  body: (let place : (Bytes * Nat) := (match (project given primary)
    | Option.some span => ((project span path), (project span byteStart))
    | Option.none => (0x, 0)); (cat (cat (cat (orderedPath (first place)) (bigEndian (second place))) (project given code)) (project given message))),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: keyed,
  parameters: [given : Diagnostic],
  result: (Bytes * Diagnostic),
  body: ((sortKey given), given),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: sortDiagnostics,
  parameters: [items : (List Diagnostic)],
  result: (List Diagnostic),
  body: (prim list_fold (fun (acc : (List Diagnostic)) (entry : (Bytes * Diagnostic)) => (cons (second entry) acc)) (nil Diagnostic) (LexLeanStd.Order::reverseList [(Bytes * Diagnostic)] (LexLeanStd.Order::sortByKey [Diagnostic] (LexLeanStd.Order::reverseList [(Bytes * Diagnostic)] (prim list_fold (fun (acc : (List (Bytes * Diagnostic))) (item : Diagnostic) => (cons (keyed item) acc)) (nil (Bytes * Diagnostic)) items : (List (Bytes * Diagnostic)))))) : (List Diagnostic)),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}

definition{
  name: isEnvironmentCode,
  parameters: [code : Bytes],
  result: Bool,
  body: (prim list_fold (fun [code] (found : Bool) (listed : Bytes) => (if found then true else (prim equal code listed : Bool))) false (cons (lit "LLB6004") (cons (lit "LLV7001") (cons (lit "LLV7007") (cons (lit "LLV7008") (cons (lit "LLV7009") (cons (lit "LLV7010") (cons (lit "LLV7012") (cons (lit "LLV7014") (cons (lit "LLV7017") (cons (lit "LLV7018") (nil Bytes))))))))))) : Bool),
  executable: true
}

definition{
  name: codeExit,
  parameters: [code : Bytes],
  result: Nat,
  body: (if (isEnvironmentCode code) then 3 else (match (LexLeanStd.Text::byteAt code 2)
    | Option.some letter => (let value : Nat := (prim to_nat letter : Nat); (if (value =? 67) then 2 else (if (value =? 83) then 4 else (if (value =? 73) then 70 else 1))))
    | Option.none => 1)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: worstExit,
  parameters: [items : (List Diagnostic), empty : Nat],
  result: Nat,
  body: (match items
    | List.nil => empty
    | List.cons head rest => (prim list_fold (fun (worst : Nat) (item : Diagnostic) => (let rank : Nat := (codeExit (project item code)); (if (worst <? rank) then rank else worst))) 0 items : Nat)),
  axioms: [Quot.sound, propext],
  executable: true
}

definition{
  name: takeStep,
  parameters: [state : (Nat * ((List Diagnostic) * (List Diagnostic)))],
  result: (Option (Nat * ((List Diagnostic) * (List Diagnostic)))),
  body: (if ((first state) =? 0) then (ctor Option.none [(Nat * ((List Diagnostic) * (List Diagnostic)))]) else (match (first (second state))
    | List.nil => (ctor Option.none [(Nat * ((List Diagnostic) * (List Diagnostic)))])
    | List.cons head rest => (ctor Option.some [(Nat * ((List Diagnostic) * (List Diagnostic)))] ((prim subtract (first state) 1 : Nat), (rest, (cons head (second (second state)))))))),
  executable: true
}

definition{
  name: firstOf,
  parameters: [items : (List Diagnostic), count : Nat],
  result: (List Diagnostic),
  body: (LexLeanStd.Order::reverseList [Diagnostic] (second (second (first (prim iterate_until (fun (state : (Nat * ((List Diagnostic) * (List Diagnostic)))) => (takeStep state)) (count + 1) (count, (items, (nil Diagnostic))) : ((Nat * ((List Diagnostic) * (List Diagnostic))) * Bool)))))),
  executable: true
}

definition{
  name: exceeded,
  parameters: [phase : Bytes, limit : Nat, observed : Nat],
  result: Diagnostic,
  body: (record Diagnostic {code := (lit "LLS8002"), message := (cat (cat (cat (cat (cat (cat (cat (cat (lit "max_diagnostics exceeded in phase ") phase) (lit ": configured ")) (num limit)) (lit ", observed ")) (num observed)) (lit "; the first ")) (num limit)) (lit " diagnostics are reported")), primary := (ctor Option.none [Span]), labels := (nil Label), notes := (nil Note), help := (nil Bytes), causes := (nil Bytes)}),
  executable: true
}

definition{
  name: reportDiagnostics,
  parameters: [items : (List Diagnostic), limit : Nat, phase : Bytes],
  result: (Nat * (List Diagnostic)),
  body: (let sorted : (List Diagnostic) := (sortDiagnostics items); (let observed : Nat := (prim length items : Nat); (if (observed <=? limit) then ((worstExit sorted 70), sorted) else (let kept : (List Diagnostic) := (LexLeanStd.Order::reverseList [Diagnostic] (cons (exceeded phase limit observed) (LexLeanStd.Order::reverseList [Diagnostic] (firstOf sorted limit)))); ((worstExit kept 70), kept))))),
  axioms: [Quot.sound, propext],
  executable: true,
  production: {effects: [allocation, overflow], targets: [rust-std]}
}
