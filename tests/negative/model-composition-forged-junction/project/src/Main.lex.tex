\begin{lexlean}{Main}
\useglossary{lexlean.std.nat@1.2.0}
\title{Natural number addition}
\begin{semanticmodule}
definition{name: Positive, parameters: [x : Nat], result: Prop, body: (1 <= x)}

definition{
  name: positiveCheck,
  parameters: [x : Nat],
  result: Bool,
  body: (1 <=? x),
  executable: true
}

theorem{
  name: positive_check_sound,
  parameters: [x : Nat],
  statement: (((positiveCheck x) = true) -> (Positive x)),
  proof: linear_arithmetic{definitions: [Positive, positiveCheck]}
}

definition{name: Doubles, parameters: [x : Nat, y : Nat], result: Prop, body: (y = (x + x))}

contract{
  name: DoubleContract,
  input: {name: x, type: Nat},
  output: {name: y, type: Nat},
  postcondition: Doubles,
  precondition: Positive,
  validators: [{predicate: precondition, sound: positive_check_sound, validator: positiveCheck}]
}

realization{
  name: Doubler,
  descriptor: deterministic{body: (x + x)},
  executable: true,
  input: {name: x, type: Nat},
  output: Nat
}

theorem{
  name: doubler_doubles,
  parameters: [x : Nat],
  statement: ((Positive x) -> (Doubles x (Doubler x))),
  proof: linear_arithmetic{definitions: [Doubler, Doubles, Positive]}
}

evidence{
  name: DoubleEvidence,
  claims: [satisfies_contract{theorem: doubler_doubles}],
  contract: {member: DoubleContract},
  realization: {member: Doubler}
}

model{
  name: DoubleModel,
  contract: {member: DoubleContract},
  evidence: [{member: DoubleEvidence}],
  realization: {member: Doubler}
}

definition{
  name: run,
  parameters: [x : Nat],
  result: (Result Nat ContractViolation),
  body: (checked DoubleModel x with [precondition]),
  executable: true
}

theorem{
  name: forged_junction,
  parameters: [x : Nat],
  statement: ((Positive x) -> (0 <= x)),
  proof: linear_arithmetic{}
}

definition{name: Small, parameters: [y : Nat], result: Prop, body: (y <= 100)}

definition{
  name: smallCheck,
  parameters: [y : Nat],
  result: Bool,
  body: (y <=? 100),
  executable: true
}

theorem{
  name: small_check_sound,
  parameters: [y : Nat],
  statement: (((smallCheck y) = true) -> (Small y)),
  proof: linear_arithmetic{definitions: [Small, smallCheck]}
}

contract{
  name: HalfContract,
  input: {name: y, type: Nat},
  output: {name: z, type: Nat},
  precondition: Small,
  validators: [{predicate: precondition, sound: small_check_sound, validator: smallCheck}]
}

realization{
  name: Halver,
  descriptor: deterministic{body: (prim quotient y 2 0 : Nat)},
  executable: true,
  input: {name: y, type: Nat},
  output: Nat
}

model{name: HalfModel, contract: {member: HalfContract}, realization: {member: Halver}}

realization{
  name: Pipeline,
  descriptor: composite{
    form: sequence{
      junctions: [proved{evidence: forged_junction}],
      stages: [{member: DoubleModel}, {member: HalfModel}]
    }
  },
  executable: true,
  input: {name: x, type: Nat},
  output: Nat
}
\end{semanticmodule}
\end{lexlean}
